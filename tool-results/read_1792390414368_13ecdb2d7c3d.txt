     1→// Alpaca Trading API client (server-side only): paper + live order routing,
     2→// positions, bracket/OCO exits, order replacement for SL/TP drag updates.
     3→//
     4→// Verified against the paper API (2026-10):
     5→// - Stocks support order_class=bracket (entry + take_profit/stop_loss legs).
     6→// - Crypto rejects ALL advanced order classes (order_class=oco -> 40010001
     7→//   "invalid order type for crypto order"; bracket -> 42210000) AND plain
     8→//   `stop` type. Sell exits also CANNOT be reserved against a PENDING buy
     9→//   (40310000 "insufficient balance" — only settled position qty counts),
    10→//   so crypto complex orders are EMULATED: the entry carries its intended
    11→//   TP/SL in the client_order_id (`ttb-…`) and the OCO supervisor below
    12→//   attaches the exit legs (`tte-…`) the moment the entry FILLS, canceling
    13→//   any surviving leg once the position is gone.
    14→// - PATCH /v2/orders/{id} creates a REPLACEMENT order with a NEW id.
    15→// - Position symbols drop the crypto slash (BTC/USD -> BTCUSD); ORDER
    16→//   symbols keep it (BTC/USD).
    17→import { db } from './db'

    40→export async function resolveTradeMode(userId: string | undefined, requested?: unknown): Promise<TradeMode> {
    41→  if (isTradeMode(requested)) return requested
    42→  if (userId) {
    43→    const user = await db.user.findUnique({ where: { id: userId } })

    45→    if (isTradeMode(s?.tradeMode)) return s.tradeMode

    47→  return 'paper'

    55→interface TradeFetchResult<T> {
    56→  data: T | null
    57→  status: number
    58→  error?: string
    59→}
    60→
    61→async function tradeFetch<T>(
    62→  userId: string,
    63→  mode: TradeMode,
    64→  path: string,
    65→  init: RequestInit = {},
    66→): Promise<TradeFetchResult<T>> {
    67→  const keys = await getAlpacaKeys(userId)
    68→  if (!keys) return { data: null, status: 0, error: 'no-keys' }
    69→  const controller = new AbortController()
    70→  const timer = setTimeout(() => controller.abort(), 9000)
    71→  try {
    72→    const res = await fetch(`${TRADE_BASE[mode]}${path}`, {
    73→      ...init,
    74→      headers: {
    75→        'APCA-API-KEY-ID': keys.keyId,
    76→        'APCA-API-SECRET-KEY': keys.secret,
    77→        accept: 'application/json',
    78→        ...(init.body ? { 'content-type': 'application/json' } : {}),
    79→        ...(init.headers || {}),
    80→      },
    81→      signal: controller.signal,
    82→      cache: 'no-store',
    83→    })
    84→    const text = await res.text()
    85→    let parsed: unknown = null
    86→    try {
    87→      parsed = text ? JSON.parse(text) : null
    88→    } catch {
    89→      parsed = null
    90→    }
    91→    if (!res.ok) {
    92→      const msg =
    93→        (parsed as { message?: string } | null)?.message ||
    94→        `Trading API HTTP ${res.status}`
    95→      console.error(`Trading ${mode} ${path} -> ${res.status}: ${msg}`)
    96→      return { data: parsed as T, status: res.status, error: msg }
    97→    }
    98→    return { data: parsed as T, status: res.status }
    99→  } catch (e) {
   100→    console.error(`Trading ${mode} ${path} failed:`, e instanceof Error ? e.message : e)
   101→    return { data: null, status: 0, error: e instanceof Error ? e.message : 'network error' }
   102→  } finally {
   103→    clearTimeout(timer)
   104→  }
   105→}

   109→/**
   110→ * Alpaca rejects prices with more than 9 decimal places ("limit price exceeds
   111→ * maximum precision of 9 decimal places") on CRYPTO, and sub-penny increments
   112→ * on STOCKS ("sub-penny increment does not fulfill minimum pricing criteria").
   113→ * Chart drag prices are raw JS floats (65800.22400000001) that blow past both —
   114→ * first the direct PATCH fails, then the cancel+re-place fallback fails too,
   115→ * LEAVING THE EXIT ORDER CANCELED AND LOST. Round every outgoing price string:
   116→ * crypto 6dp (>=1) / 8dp (<1), stocks penny 2dp (>=1) / 4dp (<1).
   117→ */
   118→export function fmtPx(v: number, crypto = false): string {
   119→  if (!Number.isFinite(v) || v <= 0) return String(v)
   120→  const dp = crypto ? (v >= 1 ? 6 : 8) : v >= 1 ? 2 : 4
   121→  return String(parseFloat(v.toFixed(dp)))
   122→}

   241→/**
   242→ * Place an entry order with optional exits.
   243→ * Stocks: single bracket request when TP/SL present (Alpaca attaches legs).
   244→ * Crypto: advanced classes are rejected — the entry is placed ALONE carrying
   245→ * the intended TP/SL in its client_order_id (`ttb-…`); the OCO supervisor
   246→ * (ensureCryptoOcoSupervisor) attaches the exit legs when the entry fills.
   247→ */
   248→export async function placeOrderWithExits(
   249→  userId: string,
   250→  mode: TradeMode,
   251→  body: PlaceOrderBody,
   252→): Promise<{ results: TradingOrder[]; errors: string[]; splitNote?: string }> {
   253→  const isCrypto = body.symbol.includes('/')
   254→  const type = body.type || 'market'
   255→  const tif = body.time_in_force || (isCrypto ? 'gtc' : type === 'market' ? 'day' : 'gtc')
   256→  const entry: Record<string, unknown> = {
   257→    symbol: tradeSymbol(body.symbol),
   258→    qty: String(body.qty),
   259→    side: body.side,
   260→    type,
   261→    time_in_force: tif,
   262→  }
   263→  if (type === 'limit') {
   264→    if (!body.limit_price) return { results: [], errors: ['Limit price required for limit orders'] }
   265→    entry.limit_price = String(body.limit_price)
   266→  }
   267→  if (body.client_order_id) entry.client_order_id = body.client_order_id
   268→
   269→  const errors: string[] = []
   270→  const results: TradingOrder[] = []
   271→
   272→  if (!isCrypto) {
   273→    // Stocks: bracket handles entry + legs atomically.
   274→    const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   275→    const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   276→    if (hasTp || hasSl) {
   277→      entry.order_class = 'bracket'
   278→      if (hasTp) entry.take_profit = { limit_price: body.take_profit!.toFixed(2) }
   279→      if (hasSl) entry.stop_loss = { stop_price: body.stop_loss!.toFixed(2) }
   280→      // encode the plan for the chart's pending display: Alpaca materializes
   281→      // bracket LEGS only when the entry activates, so a queued (weekend /
   282→      // pre-market) bracket would otherwise render without its TP/SL lines
   283→      entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   284→    }
   285→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   286→      method: 'POST',
   287→      body: JSON.stringify(entry),
   288→    })
   289→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   290→    else errors.push(r.error || 'Entry order failed')
   291→    return { results, errors }
   292→  }
   293→
   294→  // ---- Crypto: complex orders are EMULATED (no native bracket/OCO) ----
   295→  const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   296→  const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   297→  if (hasTp || hasSl) {
   298→    // encode the planned exits so the chart can render them while the entry
   299→    // is pending AND the supervisor can attach the legs without extra state
   300→    entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   301→  }
   302→  const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   303→    method: 'POST',
   304→    body: JSON.stringify(entry),
   305→  })
   306→  if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   307→  else {
   308→    errors.push(r.error || 'Entry order failed')
   309→    return { results, errors }
   310→  }
   311→  if (!hasTp && !hasSl) return { results, errors }
   312→
   313→  // Fast path: a MARKET entry usually fills instantly — poll briefly and
   314→  // attach the exits in-line so protection starts immediately. Limit entries
   315→  // return right away (the supervisor attaches on fill within ~6s).
   316→  if (type === 'market') {
   317→    let cur: TradingOrder | null = r.data
   318→    for (let i = 0; i < 4 && cur && cur.status !== 'filled'; i++) {
   319→      await new Promise((res) => setTimeout(res, 600))
   320→      const s = await tradeFetch<TradingOrder>(userId, mode, `/v2/orders/${(r.data as TradingOrder).id}`)
   321→      cur = s.data ?? cur
   322→      if (cur && ['canceled', 'cancelled', 'rejected', 'expired'].includes(cur.status)) break
   323→    }
   324→    if (cur?.status === 'filled') {
   325→      const legErrs = await placeCryptoExitLegs(
   326→        userId,
   327→        mode,
   328→        cur,
   329→        hasTp ? body.take_profit : undefined,
   330→        hasSl ? body.stop_loss : undefined,
   331→      )
   332→      errors.push(...legErrs)
   333→      return { results, errors }
   334→    }
   335→  }
   336→  return {
   337→    results,
   338→    errors,
   339→    splitNote: 'TP/SL attach automatically when the entry fills (OCO-managed)',
   340→  }
   341→}

   343→// ---------- crypto complex-order emulation (OCO supervisor) ----------
   344→
   345→/** Price -> client_order_id-safe digits (6dp fixed point). */
   346→function encP(v: number): string {
   347→  return String(Math.round(v * 1e6))
   348→}
   349→
   350→/** client_order_id prefix identifying a MANAGED crypto exit leg. */
   351→function exitLegClientId(entryOrderId: string, kind: 'tp' | 'sl'): string {
   352→  return `tte-${entryOrderId}-${kind}`
   353→}
   354→
   355→/** Encode the OCO/bracket plan (mode + TP/SL prices) into the entry's client_order_id. */
   356→function encodePlanClientId(mode: TradeMode, tp: number | null, sl: number | null): string {
   357→  const rand = Math.random().toString(36).slice(2, 8)
   358→  return `ttb-${mode[0]}-tp${tp != null && tp > 0 ? encP(tp) : '0'}-sl${sl != null && sl > 0 ? encP(sl) : '0'}-${rand}`
   359→}
   360→
   361→/** Decode the plan from a `ttb-…` client_order_id. Returns null when not ours. */
   362→function parseCryptoPlan(clientOrderId: string | undefined | null): { tp: number | null; sl: number | null } | null {
   363→  const m = /^ttb-[a-z](?:-tp(\d+))?(?:-sl(\d+))?-/.exec(clientOrderId || '')
   364→  if (!m) return null
   365→  const dec = (s: string | undefined) => (s ? parseInt(s, 10) / 1e6 : 0)
   366→  const tp = dec(m[1])
   367→  const sl = dec(m[2])
   368→  return { tp: tp > 0 ? tp : null, sl: sl > 0 ? sl : null }
   369→}
   370→
   371→/**
   372→ * Place the exit legs for a FILLED crypto entry (50/50 qty split when both
   373→ * legs are requested — Alpaca reserves balance per open sell order). Leg ids
   374→ * are deterministic (`tte-<entryId>-tp|-sl`) so the supervisor is idempotent.
   375→ * Crypto buys receive qty minus a trading fee, so legs apply a 0.99 safety
   376→ * factor to avoid 403 "insufficient balance".
   377→ */
   378→async function placeCryptoExitLegs(
   379→  userId: string,
   380→  mode: TradeMode,
   381→  entry: TradingOrder,
   382→  tp?: number,
   383→  sl?: number,
   384→): Promise<string[]> {
   385→  const errors: string[] = []
   386→  const filledQty = parseFloat(entry.filled_qty) || parseFloat(entry.qty) || 0
   387→  if (!(filledQty > 0)) return ['entry has no filled qty']
   388→  const exitSide = entry.side === 'buy' ? 'sell' : 'buy'
   389→  const both = tp != null && sl != null
   390→  const legQty = ((both ? filledQty / 2 : filledQty) * 0.99).toFixed(8).replace(/0+$/, '').replace(/\.$/, '')
   391→  if (tp != null) {
   392→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   393→      method: 'POST',
   394→      body: JSON.stringify({
   395→        symbol: entry.symbol,
   396→        qty: legQty,
   397→        side: exitSide,
   398→        type: 'limit',
   399→        time_in_force: 'gtc',
   400→        limit_price: tp.toFixed(tp < 1 ? 6 : 2),
   401→        client_order_id: exitLegClientId(entry.id, 'tp'),
   402→      }),
   403→    })
   404→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`TP: ${r.error || 'failed'}`)
   405→  }
   406→  if (sl != null) {
   407→    // stop-limit sell: limit slightly below the stop so it clears the book
   408→    const limP = exitSide === 'sell' ? sl * 0.999 : sl * 1.001
   409→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   410→      method: 'POST',
   411→      body: JSON.stringify({
   412→        symbol: entry.symbol,
   413→        qty: legQty,
   414→        side: exitSide,
   415→        type: 'stop_limit',
   416→        time_in_force: 'gtc',
   417→        stop_price: sl.toFixed(sl < 1 ? 6 : 2),
   418→        limit_price: limP.toFixed(limP < 1 ? 6 : 2),
   419→        client_order_id: exitLegClientId(entry.id, 'sl'),
   420→      }),
   421→    })
   422→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`SL: ${r.error || 'failed'}`)
   423→  }
   424→  return errors
   425→}
   426→
   427→const OCO_CYCLE_MS = 6000
   428→const OCO_OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   429→// last attach-failure signature per entry id — the supervisor retries every
   430→// cycle, but transient rejections (e.g. Alpaca's "potential wash trade
   431→// detected" while a crossing pending entry rests) must not spam the log
   432→const attachWarnSig = new Map<string, string>()
   433→
   434→/** Supervised (user,mode) pairs — survives dev hot reloads via globalThis. */
   435→function supervisedSet(): Set<string> {
   436→  const g = globalThis as { __ttOcoSupervised?: Set<string> }
   437→  if (!g.__ttOcoSupervised) g.__ttOcoSupervised = new Set<string>()
   438→  return g.__ttOcoSupervised
   439→}
   440→
   441→/**
   442→ * Start the OCO supervisor for a (user, mode) pair (idempotent). One cycle
   443→ * every 6s: (a) cancel managed exit legs whose position is gone (OCO
   444→ * semantics — when the position closes, the surviving leg dies),
   445→ * (b) attach exit legs to managed entries that have FILLED.
   446→ */
   447→export function ensureCryptoOcoSupervisor(userId: string, mode: TradeMode): void {
   448→  const key = `${userId}|${mode}`
   449→  if (supervisedSet().has(key)) return
   450→  supervisedSet().add(key)
   451→  const timer = setInterval(() => {
   452→    void runOcoCycle(userId, mode)
   453→  }, OCO_CYCLE_MS)
   454→  // never keep the process alive just for the supervisor
   455→  if (typeof timer === 'object' && timer !== null && 'unref' in timer) {
   456→    ;(timer as { unref: () => void }).unref()
   457→  }
   458→}
   459→
   460→async function runOcoCycle(userId: string, mode: TradeMode): Promise<void> {
   461→  try {
   462→    const [ordRes, posRes] = await Promise.all([
   463→      tradeFetch<TradingOrder[]>(userId, mode, '/v2/orders?status=all&limit=500&direction=desc'),
   464→      tradeFetch<TradingPosition[]>(userId, mode, '/v2/positions'),
   465→    ])
   466→    const orders: TradingOrder[] = Array.isArray(ordRes.data) ? ordRes.data : []
   467→    const positions: TradingPosition[] = Array.isArray(posRes.data) ? posRes.data : []
   468→    const norm = (s: string) => s.replace('/', '').toUpperCase()
   469→    const posSet = new Set(positions.map((p) => norm(p.symbol)))
   470→    const knownCoids = new Set(orders.map((o) => o.client_order_id || ''))
   471→    const isCrypto = (o: TradingOrder) => o.asset_class === 'crypto' || o.symbol.includes('/')
   472→
   473→    // (a) OCO cleanup: a managed exit leg dies with its position
   474→    for (const o of orders) {
   475→      if (!isCrypto(o) || !OCO_OPEN_STATUSES.includes(o.status)) continue
   476→      if (!(o.client_order_id || '').startsWith('tte-')) continue
   477→      if (!posSet.has(norm(o.symbol))) {
   478→        const r = await cancelOrder(userId, mode, o.id)
   479→        console.log(`[oco] canceled orphaned exit ${o.client_order_id} (${o.symbol}) -> ${r.status}`)
   480→      }
   481→    }
   482→
   483→    // (b) attach exit legs to managed entries that have FILLED
   484→    for (const o of orders) {
   485→      if (!isCrypto(o) || o.status !== 'filled') continue
   486→      const plan = parseCryptoPlan(o.client_order_id)
   487→      if (!plan || (plan.tp == null && plan.sl == null)) continue
   488→      const wantTp = plan.tp != null && !knownCoids.has(exitLegClientId(o.id, 'tp'))
   489→      const wantSl = plan.sl != null && !knownCoids.has(exitLegClientId(o.id, 'sl'))
   490→      if (!wantTp && !wantSl) continue
   491→      const errs = await placeCryptoExitLegs(
   492→        userId,
   493→        mode,
   494→        o,
   495→        wantTp ? plan.tp! : undefined,
   496→        wantSl ? plan.sl! : undefined,
   497→      )
   498→      if (errs.length) {
   499→        const sig = errs.join('; ')
   500→        if (attachWarnSig.get(o.id) !== sig) {
   501→          attachWarnSig.set(o.id, sig)
   502→          console.warn(`[oco] exit attach failed for ${o.id} (${o.symbol}):`, sig)
   503→        }
   504→      } else {
   505→        attachWarnSig.delete(o.id)
   506→        console.log(`[oco] attached exits to filled entry ${o.id} (${o.symbol}) tp=${plan.tp} sl=${plan.sl}`)
   507→      }
   508→    }
   509→  } catch (e) {

   511→  }

   533→/** Rebuild an open exit order body from an existing order (used by the
   534→ * cancel+re-place fallback and OCO-sibling regeneration). */
   535→function exitBodyFrom(
   536→  orig: TradingOrder,
   537→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number },
   538→): Record<string, unknown> | null {
   539→  const isCrypto = orig.asset_class === 'crypto' || orig.symbol.includes('/')
   540→  const num = (v: string | null | undefined): number | null => {
   541→    if (v == null) return null
   542→    const n = parseFloat(v)
   543→    return Number.isFinite(n) && n > 0 ? n : null
   544→  }
   545→  const newStop = patch.stop_price ?? num(orig.stop_price)
   546→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   547→  const newTrail = patch.trail_percent ?? num(orig.trail_percent)
   548→  const body: Record<string, unknown> = {
   549→    symbol: orig.symbol,
   550→    qty: orig.qty,
   551→    side: orig.side,
   552→    type: orig.type,
   553→    time_in_force: orig.time_in_force || 'gtc',
   554→  }
   555→  if (orig.type === 'stop') {
   556→    if (newStop == null) return null
   557→    body.stop_price = fmtPx(newStop, isCrypto)
   558→  } else if (orig.type === 'stop_limit') {
   559→    if (newStop == null) return null
   560→    body.stop_price = fmtPx(newStop, isCrypto)
   561→    if (newLimit != null) {
   562→      body.limit_price = fmtPx(newLimit, isCrypto)
   563→    } else {
   564→      // preserve the original stop→limit offset when only the stop moves
   565→      const os = num(orig.stop_price)
   566→      const ol = num(orig.limit_price)
   567→      const ratio = os && ol ? ol / os : orig.side === 'sell' ? 0.999 : 1.001
   568→      body.limit_price = fmtPx(newStop * ratio, isCrypto)
   569→    }
   570→  } else if (orig.type === 'limit') {
   571→    if (newLimit == null) return null
   572→    body.limit_price = fmtPx(newLimit, isCrypto)
   573→  } else if (orig.type === 'trailing_stop') {
   574→    if (newTrail == null) return null
   575→    body.trail_percent = String(newTrail)
   576→  } else {
   577→    return null // market / unknown — never re-placed blindly
   578→  }
   579→  return body
   580→}

   591→/**
   592→ * Replace SL/TP prices on an open exit order — the chart line-drag path.
   593→ *
   594→ * Verified against the paper API (2026-10):
   595→ * - A PATCH on a STANDALONE exit (crypto split exits) replaces cleanly:
   596→ *   type/time_in_force/sibling prices are preserved and a NEW order id is
   597→ *   returned.
   598→ * - The chart drag used to 404 when the order lived on the OTHER host: the
   599→ *   client did not send `mode`, so the PATCH went to the stored trade-mode
   600→ *   preference — paper order ids do not exist on live and vice versa, the
   601→ *   update silently never landed on Alpaca.
   602→ * - Bracket OCO legs may refuse direct replacement; cancel+re-place is the
   603→ *   universal fallback (canceling one OCO leg can cascade-cancel the sibling,
   604→ *   which we detect and regenerate).
   605→ */
   606→export async function replaceExitOrderSmart(
   607→  userId: string,
   608→  mode: TradeMode,
   609→  orderId: string,
   610→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number; crypto?: boolean },
   611→): Promise<ReplaceExitResult> {
   612→  const patchBody: Record<string, unknown> = {}
   613→  if (patch.stop_price != null) patchBody.stop_price = fmtPx(patch.stop_price, patch.crypto)
   614→  if (patch.limit_price != null) patchBody.limit_price = fmtPx(patch.limit_price, patch.crypto)
   615→  if (patch.trail_percent != null) patchBody.trail_percent = String(parseFloat(patch.trail_percent.toFixed(4)))
   616→
   617→  // 1) PATCH on the requested host; 2) retry the other host on 404 —
   618→  // both hosts share the same key pair, so a wrong-host id self-heals here.
   619→  const order: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   620→  let last: TradeFetchResult<TradingOrder> | null = null
   621→  let errHost: TradeMode = mode
   622→  for (const m of order) {
   623→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`, {
   624→      method: 'PATCH',
   625→      body: JSON.stringify(patchBody),
   626→    })
   627→    if (r.data && (r.data as TradingOrder).id) {
   628→      return { data: r.data, status: r.status, mode: m, method: 'replace' }
   629→    }
   630→    last = r
   631→    // 422 "order parameters are not changed" — the order ALREADY carries the
   632→    // requested value (e.g. a second drag to the same level). That IS the
   633→    // desired end state: return the current order instead of churning a
   634→    // cancel+re-place cycle.
   635→    if (r.status === 422 && String(r.error || '').includes('not changed')) {
   636→      const cur = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   637→      if (cur.data) {
   638→        return { data: cur.data, status: 200, mode: m, method: 'replace', note: 'Order already at this value' }
   639→      }
   640→    }
   641→    if (r.status !== 404) {
   642→      // real rejection — the order lives on THIS host; run the fallback there
   643→      errHost = m
   644→      break
   645→    }
   646→  }
   647→
   648→  // 3) The id may have JUST been replaced (fast double-drag: Alpaca's
   649→  // cancel/replace is async and 404s for a beat) — follow the replaced_by
   650→  // chain and retry the direct PATCH on the live successor first.
   651→  const OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   652→  let orig: TradingOrder | null = null
   653→  let targetId = orderId
   654→  for (let hop = 0; hop < 4; hop++) {
   655→    const o = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`)
   656→    if (!o.data) break
   657→    const od = o.data
   658→    if (OPEN_STATUSES.includes(String(od.status))) {
   659→      orig = od
   660→      if (targetId !== orderId) {
   661→        // the open successor exists — retry the direct PATCH on it
   662→        const retry = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`, {
   663→          method: 'PATCH',
   664→          body: JSON.stringify(patchBody),
   665→        })
   666→        if (retry.data && (retry.data as TradingOrder).id) {
   667→          return { data: retry.data, status: 200, mode: errHost, method: 'replace', note: 'Applied to the replacement order' }
   668→        }
   669→        if (retry.status === 422 && String(retry.error || '').includes('not changed')) {
   670→          return { data: od, status: 200, mode: errHost, method: 'replace', note: 'Order already at this value' }
   671→        }
   672→        last = retry
   673→      }
   674→      break
   675→    }
   676→    const rb = od.replaced_by
   677→    if (typeof rb === 'string' && rb && rb !== targetId) {
   678→      targetId = rb
   679→      continue
   680→    }
   681→    orig = od // terminal status without a successor
   682→    break
   683→  }
   684→  if (!orig) {
   685→    return {
   686→      data: null,
   687→      status: last?.status || 502,
   688→      error: last?.error || 'Order not found on paper or live',
   689→      mode: errHost,
   690→      method: 'cancel-replace',
   691→    }
   692→  }
   693→  if (!OPEN_STATUSES.includes(String(orig.status))) {
   694→    return {
   695→      data: null,
   696→      status: 409,
   697→      error: `Order is ${orig.status} on ${errHost} — nothing to replace (refresh the chart)`,
   698→      mode: errHost,
   699→      method: 'cancel-replace',
   700→    }
   701→  }
   702→
   703→  // snapshot the OCO sibling BEFORE cancelling (limit TP vs stop-ish SL of
   704→  // the same symbol + same side) so a cascade-cancel can be healed
   705→  const openRes = await tradeFetch<TradingOrder[]>(userId, errHost, "/v2/orders?status=open&limit=500")
   706→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   707→  const isStopish = (t: string) => t === 'stop' || t === 'stop_limit' || t === 'trailing_stop'
   708→  const sibling = (openRes.data || []).find(
   709→    (o) =>
   710→      o.id !== orderId &&
   711→      norm(o.symbol) === norm(orig.symbol) &&
   712→      o.side === orig.side &&
   713→      ((isStopish(orig.type) && o.type === 'limit') || (orig.type === 'limit' && isStopish(o.type))),
   714→  )
   715→
   716→  const cancelRes = await cancelOrder(userId, errHost, targetId)
   717→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
   718→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode: errHost, method: 'cancel-replace' }
   719→  }
   720→
   721→  const newBody = exitBodyFrom(orig, patch)
   722→  if (!newBody) {
   723→    return { data: null, status: 400, error: `Unsupported exit type ${orig.type} for re-place`, mode: errHost, method: 'cancel-replace' }
   724→  }
   725→  // re-place on the host the order actually lives on (`mode` can differ when
   726→  // the PATCH self-healed across paper/live)
   727→  const r2 = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   728→    method: 'POST',
   729→    body: JSON.stringify(newBody),
   730→  })
   731→  if (!(r2.data && (r2.data as TradingOrder).id)) {
   732→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode: errHost, method: 'cancel-replace' }
   733→  }
   734→
   735→  // heal the sibling if the cancel cascaded (OCO group cancellation)
   736→  let note: string | undefined
   737→  if (sibling) {
   738→    const stillOpen = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${sibling.id}`)
   739→    const gone = !stillOpen.data || ['canceled', 'cancelled', 'expired', 'rejected'].includes(String(stillOpen.data.status))
   740→    if (gone) {
   741→      const sibBody = exitBodyFrom(sibling, {})
   742→      if (sibBody) {
   743→        const rs = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   744→          method: 'POST',
   745→          body: JSON.stringify(sibBody),
   746→        })
   747→        note = rs.data ? 'OCO sibling re-created' : `Warning: sibling exit lost (${rs.error || 're-place failed'})`
   748→      }
   749→    }
   750→  }
   751→  return { data: r2.data, status: r2.status, mode: errHost, method: 'cancel-replace', note }
   752→}

   767→/** Poll the order until the async cancel lands (buying power frees up). */
   768→async function waitUntilCanceled(userId: string, host: TradeMode, orderId: string): Promise<void> {
   769→  for (let i = 0; i < 4; i++) {
   770→    await new Promise((res) => setTimeout(res, 350))
   771→    const r = await tradeFetch<TradingOrder>(userId, host, `/v2/orders/${orderId}`)
   772→    const s = String(r.data?.status || '')
   773→    if (['canceled', 'cancelled', 'expired', 'rejected'].includes(s)) return
   774→    if (!r.data) return // gone / unreachable — best effort
   775→  }
   776→}

   778→/**
   779→ * Change the entry limit price and/or the planned TP/SL of a PENDING
   780→ * (unfilled) entry order — the chart drag on the dashed entry / (pend)
   781→ * TP / SL lines lands here.
   782→ *
   783→ * - STOCK bracket parents: a direct PATCH accepts limit_price plus
   784→ *   take_profit/stop_loss bodies — both legs' prices are always sent so the
   785→ *   untouched leg keeps its level.
   786→ * - CRYPTO managed entries (`ttb-` plan in client_order_id): ALWAYS cancel →
   787→ *   wait → re-place with a freshly encoded plan. A direct PATCH replaces the
   788→ *   order WITHOUT carrying the client_order_id over (verified live: the
   789→ *   replacement lost the plan and the OCO supervisor would never attach the
   790→ *   exits), and client_order_id cannot be safely re-sent (uniqueness).
   791→ *   Re-placing keeps the REMAINING qty (a partially filled entry keeps its
   792→ *   fill).
   793→ * - Plain limit entries (no plan): direct PATCH on limit_price.
   794→ * - Any direct-PATCH rejection falls back to cancel+re-place with a full
   795→ *   body (mirrors replaceExitOrderSmart's strategy).
   796→ */
   797→export async function replacePendingEntrySmart(
   798→  userId: string,
   799→  mode: TradeMode,
   800→  orderId: string,
   801→  patch: { limit_price?: number; tp?: number; sl?: number; crypto?: boolean },
   802→): Promise<ReplaceEntryResult> {
   803→  // 1) locate the order (host fallback, same id spaces as exits)
   804→  const hosts: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   805→  let host: TradeMode = mode
   806→  let orig: TradingOrder | null = null
   807→  let last: TradeFetchResult<TradingOrder> | null = null
   808→  for (const m of hosts) {
   809→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   810→    if (r.data && (r.data as TradingOrder).id) {
   811→      orig = r.data
   812→      host = m
   813→      break
   814→    }
   815→    last = r
   816→  }
   817→  if (!orig) {
   818→    return { data: null, status: last?.status || 404, error: last?.error || 'Order not found on paper or live', mode }
   819→  }
   820→  const statusStr = String(orig.status)
   821→  if (!ENTRY_OPEN_STATUSES.includes(statusStr)) {
   822→    return { data: null, status: 409, error: `Order is ${statusStr} — nothing to change (refresh the chart)`, mode: host }
   823→  }
   824→
   825→  const isCrypto = orig.asset_class === 'crypto' || orig.symbol.includes('/')
   826→  const crypto = Boolean(patch.crypto) || isCrypto
   827→  const plan = parseCryptoPlan(orig.client_order_id)
   828→  const exitLegs = (orig.legs || []).filter((l) => l.side !== orig.side)
   829→  const num = (v: string | null | undefined): number | null => {
   830→    if (v == null) return null
   831→    const n = parseFloat(v)
   832→    return Number.isFinite(n) && n > 0 ? n : null
   833→  }
   834→  // current TP/SL: requested patch wins, else the parent's legs, else the plan
   835→  const curTp =
   836→    patch.tp ??
   837→    num(exitLegs.find((l) => l.type === 'limit')?.limit_price) ??
   838→    plan?.tp ??
   839→    null
   840→  const curSl =
   841→    patch.sl ??
   842→    num(exitLegs.find((l) => l.type === 'stop' || l.type === 'stop_limit')?.stop_price) ??
   843→    plan?.sl ??
   844→    null
   845→  const origHasTp = curTp != null
   846→  const origHasSl = curSl != null
   847→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   848→  // a PATCH cannot ADD a bracket leg the order does not have — full re-place
   849→  const addsLeg = (patch.tp != null && !origHasTp) || (patch.sl != null && !origHasSl)
   850→  // crypto plan orders ALWAYS re-place: a direct PATCH drops the ttb-
   851→  // client_order_id (verified live), losing the TP/SL plan entirely
   852→  const managedCrypto = Boolean(plan)
   853→
   854→  // 2) direct PATCH (limit change and/or leg price updates, no new legs)
   855→  if (!addsLeg && !managedCrypto) {
   856→    const body: Record<string, unknown> = {}
   857→    if (patch.limit_price != null && newLimit != null) body.limit_price = fmtPx(newLimit, crypto)
   858→    if (origHasTp) body.take_profit = { limit_price: fmtPx(curTp!, crypto) }
   859→    if (origHasSl) body.stop_loss = { stop_price: fmtPx(curSl!, crypto) }
   860→    if (Object.keys(body).length > 0) {
   861→      const r = await tradeFetch<TradingOrder>(userId, host, `/v2/orders/${orderId}`, {
   862→        method: 'PATCH',
   863→        body: JSON.stringify(body),
   864→      })
   865→      if (r.data && (r.data as TradingOrder).id) {
   866→        return { data: r.data, status: r.status, mode: host, method: 'replace' }
   867→      }
   868→      if (r.status === 422 && String(r.error || '').includes('not changed')) {
   869→        return { data: orig, status: 200, mode: host, method: 'replace', note: 'Order already at this value' }
   870→      }
   871→      // real rejection → cancel+re-place below
   872→    } else {
   873→      return { data: null, status: 400, error: 'nothing to change', mode: host }
   874→    }
   875→  }
   876→
   877→  // 3) cancel + re-place (plan change on crypto, new leg, or rejected PATCH)
   878→  const cancelRes = await cancelOrder(userId, host, orderId)
   879→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
   880→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode: host, method: 'cancel-replace' }
   881→  }
   882→  if (cancelRes.status !== 404) await waitUntilCanceled(userId, host, orderId)
   883→
   884→  const remaining = (parseFloat(orig.qty) || 0) - (parseFloat(orig.filled_qty) || 0)
   885→  if (!(remaining > 0)) {
   886→    return { data: null, status: 200, mode: host, method: 'cancel-replace', note: 'Entry filled — nothing re-placed' }
   887→  }
   888→  const body: Record<string, unknown> = {
   889→    symbol: orig.symbol,
   890→    qty: String(remaining),
   891→    side: orig.side,
   892→    type: orig.type,
   893→    time_in_force: orig.time_in_force || 'gtc',
   894→  }
   895→  if (newLimit != null && (orig.type === 'limit' || orig.type === 'stop_limit')) {
   896→    body.limit_price = fmtPx(newLimit, crypto)
   897→  }
   898→  if (orig.type === 'stop' && curSl != null) body.stop_price = fmtPx(curSl, crypto)
   899→  const wantTp = patch.tp ?? curTp
   900→  const wantSl = patch.sl ?? curSl
   901→  const hadPlan = Boolean(plan) || (orig.legs?.length ?? 0) > 0 || patch.tp != null || patch.sl != null
   902→  if (hadPlan) {
   903→    if (isCrypto) {
   904→      // re-encode the plan — the supervisor attaches legs from it on fill
   905→      body.client_order_id = encodePlanClientId(host, wantTp ?? null, wantSl ?? null)
   906→    } else {
   907→      body.order_class = 'bracket'
   908→      if (wantTp != null) body.take_profit = { limit_price: fmtPx(wantTp, false) }
   909→      if (wantSl != null) body.stop_loss = { stop_price: fmtPx(wantSl, false) }
   910→    }
   911→  }
   912→  const r2 = await tradeFetch<TradingOrder>(userId, host, '/v2/orders', {
   913→    method: 'POST',
   914→    body: JSON.stringify(body),
   915→  })
   916→  if (!(r2.data && (r2.data as TradingOrder).id)) {
   917→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode: host, method: 'cancel-replace' }
   918→  }
   919→  return { data: r2.data, status: r2.status, mode: host, method: 'cancel-replace' }
   920→}
   921→
   922→/**
   923→ * Market-close a position. Crypto balances are reserved per open exit order,
   924→ * so lingering SL/TP legs would starve the liquidation (Alpaca sells only
   925→ * qty_available) — cancel the symbol's opposite-side (exit) orders FIRST,
   926→ * then close. Returns the close order and how many exits were cancelled.
   927→ */
   928→export async function closePositionFull(
   929→  userId: string,
   930→  mode: TradeMode,
   931→  symbol: string,
   932→): Promise<{ close: TradeFetchResult<TradingOrder>; canceledExits: number }> {
   933→  const tSym = tradeSymbol(symbol)
   934→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   935→  // snapshot the position side BEFORE closing (long -> sell exits, short -> buy)
   936→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
   937→  const posSide = posRes.data?.side
   938→  let canceledExits = 0
   939→  // fetch ALL open orders and filter here — Alpaca's `symbols=` query wants
   940→  // the slash form for crypto (BTC/USD) while callers pass either form
   941→  const openRes = await getOpenOrders(userId, mode)
   942→  const orders: TradingOrder[] = (Array.isArray(openRes.data) ? openRes.data : [])
   943→    .filter((o) => norm(o.symbol) === norm(tSym))
   944→  const exitSide = posSide === 'long' ? 'sell' : posSide === 'short' ? 'buy' : null
   945→  for (const o of orders) {
   946→    if (exitSide && o.side !== exitSide) continue // keep same-side (entry) orders
   947→    if (exitSide == null && o.type === 'market') continue // no position info: keep entries
   948→    const c = await cancelOrder(userId, mode, o.id)
   949→    if (c.status === 204 || c.status === 200) canceledExits += 1
   950→  }
   951→  const close = await closePosition(userId, mode, symbol)
   952→  return { close, canceledExits }
   953→}

   967→/**
   968→ * Attach a TP and/or SL to an open position — the chart's "+TP"/"+SL" chip
   969→ * drag and the Positions-tab add buttons land here.
   970→ *
   971→ * Stocks (verified against the paper API 2026-10):
   972→ * - order_class=oco REQUIRES take_profit.limit_price AND stop_loss.stop_price
   973→ *   AND an existing position to exit (without one: "oco orders must be exit
   974→ *   orders"; parent must be type=limit). With both legs requested we send the
   975→ *   documented OCO exit body and FALL BACK to separate legs on rejection.
   976→ * - A single leg is a plain limit (TP) / stop (SL) order on the full position.
   977→ * Crypto: ALL advanced order classes are rejected — place split exit legs
   978→ * (50/50 qty, 0.99 fee factor) directly against the position, tagged with a
   979→ * `ttp-` client_order_id (ignored by the OCO supervisor's `tte-` cleanup).
   980→ */
   981→export async function attachExitsToPosition(
   982→  userId: string,
   983→  mode: TradeMode,
   984→  symbol: string,
   985→  exits: { tp?: number; sl?: number },
   986→): Promise<AttachExitsResult> {
   987→  const tSym = tradeSymbol(symbol)
   988→  const hasTp = typeof exits.tp === 'number' && exits.tp > 0
   989→  const hasSl = typeof exits.sl === 'number' && exits.sl > 0
   990→  if (!hasTp && !hasSl) return { results: [], errors: ['no TP/SL price given'] }
   991→
   992→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
   993→  if (!posRes.data) {
   994→    return { results: [], errors: [posRes.error && posRes.error !== 'not found' ? posRes.error : 'Position not found'] }
   995→  }
   996→  const pos = posRes.data
   997→  // crypto-ness comes from the POSITION (callers pass either symbol form —
   998→  // the chart sends the trade symbol BTCUSD, whose crypto-ness the route
   999→  // cannot infer from a slash)
  1000→  const isCrypto = pos.asset_class === 'crypto' || pos.symbol.includes('/')
  1001→  // orders use the position's own symbol form (crypto keeps the slash)
  1002→  const orderSymbol = pos.symbol
  1003→  // long position -> sell exits; short position -> buy exits
  1004→  const exitSide = pos.side === 'short' ? 'buy' : 'sell'
  1005→  const qtyAvail = parseFloat(pos.qty_available) || parseFloat(pos.qty) || 0
  1006→  if (!(qtyAvail > 0)) return { results: [], errors: ['position has no available qty'] }
  1007→
  1008→  const results: TradingOrder[] = []
  1009→  const errors: string[] = []
  1010→  const post = (body: Record<string, unknown>) =>
  1011→    tradeFetch<TradingOrder>(userId, mode, '/v2/orders', { method: 'POST', body: JSON.stringify(body) })
  1012→  const legQty = (n: number) => String(parseFloat((n * (isCrypto ? 0.99 : 1)).toFixed(8)))
  1013→
  1014→  if (!isCrypto) {
  1015→    if (hasTp && hasSl) {
  1016→      const r = await post({
  1017→        symbol: orderSymbol,
  1018→        qty: String(qtyAvail),
  1019→        side: exitSide,
  1020→        type: 'limit',
  1021→        time_in_force: 'gtc',
  1022→        limit_price: fmtPx(exits.tp!),
  1023→        order_class: 'oco',
  1024→        take_profit: { limit_price: fmtPx(exits.tp!) },
  1025→        stop_loss: { stop_price: fmtPx(exits.sl!) },
  1026→      })
  1027→      if (r.data && (r.data as TradingOrder).id) {
  1028→        return { results: [r.data], errors, oco: true }
  1029→      }
  1030→      errors.push(`OCO rejected, placing legs separately: ${r.error || 'failed'}`)
  1031→    }
  1032→    if (hasTp) {
  1033→      const r = await post({
  1034→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'limit',
  1035→        time_in_force: 'gtc', limit_price: fmtPx(exits.tp!),
  1036→      })
  1037→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1038→      else errors.push(`TP: ${r.error || 'failed'}`)
  1039→    }
  1040→    if (hasSl) {
  1041→      const r = await post({
  1042→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'stop',
  1043→        time_in_force: 'gtc', stop_price: fmtPx(exits.sl!),
  1044→      })
  1045→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1046→      else errors.push(`SL: ${r.error || 'failed'}`)
  1047→    }
  1048→    return { results, errors }
  1049→  }
  1050→
  1051→  // ---- crypto: emulated split legs (balance is reserved per open sell) ----
  1052→  // Each attached leg takes HALF the position (the two legs together cover
  1053→  // it), capped by what is still available — Alpaca reserves every open sell
  1054→  // against qty_available, so a full-size first leg would make attaching the
  1055→  // second one impossible (502 insufficient balance). Legs may be attached
  1056→  // independently in any order.
  1057→  const posQty = parseFloat(pos.qty) || qtyAvail
  1058→  const qty = legQty(Math.min(qtyAvail, posQty / 2))
  1059→  const coidBase = `ttp-${Math.random().toString(36).slice(2, 8)}`
  1060→  if (hasTp) {
  1061→    const r = await post({
  1062→      symbol: orderSymbol, qty, side: exitSide, type: 'limit', time_in_force: 'gtc',
  1063→      limit_price: fmtPx(exits.tp!, true),
  1064→      client_order_id: `${coidBase}-tp`,
  1065→    })
  1066→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1067→    else errors.push(`TP: ${r.error || 'failed'}`)
  1068→  }
  1069→  if (hasSl) {
  1070→    // stop-limit: book-clearing limit offset beyond the stop
  1071→    const limP = exitSide === 'sell' ? exits.sl! * 0.999 : exits.sl! * 1.001
  1072→    const r = await post({
  1073→      symbol: orderSymbol, qty, side: exitSide, type: 'stop_limit', time_in_force: 'gtc',
  1074→      stop_price: fmtPx(exits.sl!, true),
  1075→      limit_price: fmtPx(limP, true),
  1076→      client_order_id: `${coidBase}-sl`,
  1077→    })
  1078→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1079→    else errors.push(`SL: ${r.error || 'failed'}`)
  1080→  }
  1081→  return { results, errors }
  1082→}

  1084→export async function closePosition(userId: string, mode: TradeMode, symbol: string) {
  1085→  return tradeFetch<TradingOrder>(userId, mode, `/v2/positions/${encodeURIComponent(tradeSymbol(symbol))}`, {
  1086→    method: 'DELETE',
  1087→  })
  1088→}
