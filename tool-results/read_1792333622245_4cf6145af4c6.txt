   492→   492→  function segPath(ctx: CanvasRenderingContext2D, p1: { x: number; y: number }, p2: { x: number; y: number }, w: number, extendRight: boolean) {
   493→   493→    ctx.beginPath()
   494→   494→    ctx.moveTo(p1.x, p1.y)
   495→   495→    if (extendRight) {
   496→   496→      const dx = p2.x - p1.x
   497→   497→      const dy = p2.y - p1.y
   498→   498→      const farX = dx !== 0 ? (w - p1.x) / dx : Infinity
   499→   499→      const ex = farX >= 0 ? w : p1.x - dx * 100
   500→   500→      const ey = p1.y + dy * (farX >= 0 ? farX : -100)
   501→   501→      ctx.lineTo(ex, ey)
   502→   502→    } else {
   503→   503→      ctx.lineTo(p2.x, p2.y)
   504→   504→    }
   505→   505→    ctx.stroke()
   506→   506→  }

   509→   509→    ctx.save()
   510→   510→    ctx.strokeStyle = d.color
   511→   511→    ctx.lineWidth = d.width + (highlighted ? 1 : 0)
   512→   512→    ctx.globalAlpha = 1
   513→   513→
   514→   514→    if (d.kind === 'hline') {
   515→   515→      const y = yOf(d.points[0]?.price ?? 0)
   516→   516→      if (y != null) {
   517→   517→        hlinePath(ctx, y, w)
   518→   518→        drawChip(ctx, w - 74, y, fmtPrice(d.points[0].price), d.color, highlighted)
   519→   519→        // move grip dots at center
   520→   520→        dot(ctx, w / 2, y, d.color, highlighted)
   521→   521→      }
   522→   522→    } else if (d.kind === 'vline') {
   523→   523→      const x = xOf(d.points[0]?.time ?? null)
   524→   524→      if (x != null) {
   525→   525→        vlinePath(ctx, x, h)
   526→   526→        dot(ctx, x, h / 2, d.color, highlighted)
   527→   527→      }
   528→   528→    } else if (d.kind === 'trend' || d.kind === 'fib') {
   529→   529→      const p1 = pixelPoint(d.points[0])
   530→   530→      const p2 = pixelPoint(d.points[1])
   531→   531→      if (p1 && p2) {
   532→   532→        if (d.kind === 'trend') {
   533→   533→          segPath(ctx, p1, p2, w, false)
   534→   534→        } else {
   535→   535→          // fib: levels between p1..p2 extended to right edge
   536→   536→          const xLeft = Math.min(p1.x, p2.x)
   537→   537→          const price1 = d.points[0].price
   538→   538→          const price2 = d.points[1].price
   539→   539→          ctx.font = '10px ui-monospace, monospace'
   540→   540→          for (const lv of FIB_LEVELS) {
   541→   541→            const priceLv = price2 + (price1 - price2) * lv
   542→   542→            const y = yOf(priceLv)
   543→   543→            if (y == null) continue
   544→   544→            ctx.globalAlpha = lv === 0 || lv === 1 ? 0.9 : 0.6
   545→   545→            ctx.setLineDash(lv === 0.5 ? [6, 3] : [])
   546→   546→            ctx.beginPath()
   547→   547→            ctx.moveTo(xLeft, Math.round(y) + 0.5)
   548→   548→            ctx.lineTo(w, Math.round(y) + 0.5)
   549→   549→            ctx.stroke()
   550→   550→            ctx.globalAlpha = 0.95
   551→   551→            ctx.fillStyle = d.color
   552→   552→            const label = `${lv.toFixed(3)}  ${fmtPrice(priceLv)}`
   553→   553→            ctx.fillText(label, xLeft + 4, y - 3)
   554→   554→          }
   555→   555→        }
   556→   556→        dot(ctx, p1.x, p1.y, d.color, highlighted)
   557→   557→        dot(ctx, p2.x, p2.y, d.color, highlighted)
   558→   558→      }
   559→   559→    }
   560→   560→    ctx.restore()
   561→   561→  }

   637→   637→  const distToSeg = (px: number, py: number, a: { x: number; y: number }, b: { x: number; y: number }) => {
   638→   638→    const dx = b.x - a.x
   639→   639→    const dy = b.y - a.y
   640→   640→    const len2 = dx * dx + dy * dy
   641→   641→    if (len2 === 0) return Math.hypot(px - a.x, py - a.y)
   642→   642→    let t = ((px - a.x) * dx + (py - a.y) * dy) / len2
   643→   643→    t = Math.max(0, Math.min(1, t))
   644→   644→    return Math.hypot(px - (a.x + t * dx), py - (a.y + t * dy))
   645→   645→  }

   656→   656→    if (myExits?.sl && (myExits.sl.stopPrice != null || myExits.sl.limitPrice != null)) {

   658→   658→      if (myExits.sl.type !== 'trailing_stop') {
