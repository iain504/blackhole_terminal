   255→   255→// Ticking local clock for the footer (updates every second).
   256→   256→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   257→   257→function LiveClock() {
   258→   258→  const [now, setNow] = useState<Date | null>(null)
   259→   259→  useEffect(() => {
   260→   260→    setNow(new Date())
   261→   261→    const timer = setInterval(() => setNow(new Date()), 1000)
   262→   262→    return () => clearInterval(timer)
   263→   263→  }, [])
   264→   264→  const tz = useMemo(() => {
   265→   265→    try {
   266→   266→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   267→   267→    } catch {
   268→   268→      return 'Local'
   269→   269→    }
   270→   270→  }, [])
   271→   271→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   272→   272→  const time = now
   273→   273→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   274→   274→    : '--:--:--'
   275→   275→  return (
   276→   276→    <span
   277→   277→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   278→   278→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   279→   279→    >
   280→   280→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   281→   281→      <span>{time}</span>
   282→   282→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   283→   283→    </span>
   284→   284→  )
   285→   285→}

   355→   355→function alertSummary(a: Alert): string {
   356→   356→  if (a.kind === 'rsi') {
   357→   357→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   358→   358→  }
   359→   359→  if (a.kind === 'macd') {
   360→   360→    const p = a.params ?? {}
   361→   361→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   362→   362→    const cond =
   363→   363→      a.condition === 'bull_cross' ? 'bullish crossover'
   364→   364→      : a.condition === 'bear_cross' ? 'bearish crossover'
   365→   365→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   366→   366→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   367→   367→      : 'crossover'
   368→   368→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   369→   369→  }
   370→   370→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   371→   371→}

   425→   425→          : kind === 'rsi'
   426→   426→            ? {
   427→   427→                symbol: sym, kind, price: parseFloat(rsiLevel),
   428→   428→                direction: dir, timeframe: tf,
   429→   429→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   430→   430→              }
   431→   431→            : {
   432→   432→                symbol: sym, kind, condition: cond, timeframe: tf,
   433→   433→                price: parseFloat(macdLevel) || 0,
   434→   434→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   435→   435→                color: '#f59e0b',
   436→   436→              }
   437→   437→      const res = await fetch('/api/alerts', {
   438→   438→        method: 'POST',
   439→   439→        headers: { 'content-type': 'application/json' },
   440→   440→        body: JSON.stringify(body),
   441→   441→      })
   442→   442→      const json = await res.json().catch(() => null)
   443→   443→      if (res.ok && json?.alert) {
   444→   444→        addAlertLocal(json.alert as Alert)
   445→   445→      } else {
   446→   446→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   447→   447→      }
   448→   448→    } catch { /* offline */ } finally {
   449→   449→      setCreating(false)
   450→   450→    }
   451→   451→  }
   452→   452→
   453→   453→  const toggle = async (id: string, status: string) => {
   454→   454→    const next = status === 'paused' ? 'active' : 'paused'
   455→   455→    patchAlertLocal(id, { status: next })
   456→   456→    await fetch(`/api/alerts/${id}`, {
   457→   457→      method: 'PATCH',
   458→   458→      headers: { 'content-type': 'application/json' },
   459→   459→      body: JSON.stringify({ status: next }),
   460→   460→    })
   461→   461→  }
   462→   462→  const remove = async (id: string) => {
   463→   463→    removeAlertLocal(id)
   464→   464→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   465→   465→  }
   466→   466→
   467→   467→  const formValid =
   468→   468→    kind === 'price'
   469→   469→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   470→   470→      : kind === 'rsi'
   471→   471→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   472→   472→        : cond === 'cross_above' || cond === 'cross_below'
   473→   473→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   474→   474→          : symbol.trim().length > 0
   475→   475→
   476→   476→  return (
   477→   477→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   478→   478→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   479→   479→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   480→   480→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   481→   481→          <span className="flex items-center gap-2">
   482→   482→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   483→   483→          </span>
   484→   484→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   485→   485→            Close
   486→   486→          </Button>
   487→   487→        </div>
   488→   488→
   489→   489→        {/* ---- new alarm form ---- */}
   490→   490→        <div className="border-b border-zinc-800 p-2.5">
   491→   491→          <div className="mb-2 flex items-center gap-1.5">
   492→   492→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   493→   493→              <button
   494→   494→                key={k}
   495→   495→                onClick={() => setKind(k)}
   496→   496→                aria-pressed={kind === k}
   497→   497→                className={cn(
   498→   498→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   499→   499→                  kind === k
   500→   500→                    ? k === 'rsi'
   501→   501→                      ? 'bg-pink-500/15 text-pink-300'
   502→   502→                      : k === 'macd'
   503→   503→                        ? 'bg-amber-500/15 text-amber-300'
   504→   504→                        : 'bg-emerald-500/15 text-emerald-300'
   505→   505→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   506→   506→                )}
   507→   507→              >
   508→   508→                {k}
   509→   509→              </button>
   510→   510→            ))}
   511→   511→            <Input
   512→   512→              value={symbol}
   513→   513→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   514→   514→              placeholder="SYMBOL"
   515→   515→              aria-label="Alarm symbol"
   516→   516→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   517→   517→            />
   518→   518→          </div>
   519→   519→
   520→   520→          {kind === 'price' && (
   521→   521→            <div className="flex items-center gap-1.5">
   522→   522→              <select
   523→   523→                value={dir}
   524→   524→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   525→   525→                aria-label="Trigger direction"
   526→   526→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   527→   527→              >
   528→   528→                <option value="above">crosses above</option>
   529→   529→                <option value="below">falls below</option>
   530→   530→              </select>
   531→   531→              <Input
   532→   532→                type="number" step="any" value={price}
   533→   533→                onChange={(e) => setPrice(e.target.value)}
   534→   534→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   535→   535→                placeholder="Price" aria-label="Alert price"
   536→   536→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   537→   537→              />
   538→   538→            </div>
   539→   539→          )}
   540→   540→
   541→   541→          {kind === 'rsi' && (
   542→   542→            <div className="space-y-1.5">
   543→   543→              <div className="flex items-center gap-1.5">
   544→   544→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>

   547→   547→                  onChange={(e) => setRsiPeriod(e.target.value)}
   548→   548→                  aria-label="RSI period"
   549→   549→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   550→   550→                />
   551→   551→                <select
   552→   552→                  value={dir}
   553→   553→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   554→   554→                  aria-label="RSI trigger direction"
   555→   555→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   556→   556→                >
   557→   557→                  <option value="above">crosses above</option>
   558→   558→                  <option value="below">crosses below</option>
   559→   559→                </select>
   560→   560→                <Input
   561→   561→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   562→   562→                  onChange={(e) => setRsiLevel(e.target.value)}
   563→   563→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   564→   564→                  aria-label="RSI level"
   565→   565→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   566→   566→                />
   567→   567→              </div>
   568→   568→              <TfRow tf={tf} setTf={setTf} />
   569→   569→            </div>
   570→   570→          )}
   571→   571→
   572→   572→          {kind === 'macd' && (
   573→   573→            <div className="space-y-1.5">
   574→   574→              <div className="flex items-center gap-1.5">
   575→   575→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   576→   576→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   577→   577→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   578→   578→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   579→   579→                <select
   580→   580→                  value={cond}

   582→   582→                  aria-label="MACD condition"
   583→   583→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"

   594→   594→                    type="number" step="any" value={macdLevel}
   595→   595→                    onChange={(e) => setMacdLevel(e.target.value)}

   604→   604→

   620→   620→          {alerts.map((a) => (
   621→   621→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">

   624→   624→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   625→   625→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   626→   626→                <Badge
   627→   627→                  variant="outline"
   628→   628→                  className={
   629→   629→                    a.status === 'triggered'
   630→   630→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   631→   631→                      : a.status === 'paused'
   632→   632→                        ? 'border-zinc-600 text-[10px] text-zinc-400'

   636→   636→                  {a.status}

   642→   642→                </Button>
   643→   643→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   644→   644→                  ✕
   645→   645→                </Button>

   647→   647→            </div>

   649→   649→        </div>
   650→   650→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   651→   651→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings

   653→   653→            <Settings2 className="h-3 w-3" />
   654→   654→          </span>
   655→   655→        </div>
   656→   656→      </div>
   657→   657→    </div>
   658→   658→  )
   659→   659→}
