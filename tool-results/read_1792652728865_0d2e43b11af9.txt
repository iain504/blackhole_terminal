    56→/** Resolves the pane list for a tab (grid config or the single legacy pane). */
    57→export function tabPanes(tab: Tab): Pane[] {
    58→  if (tab.grid?.panes?.length) return tab.grid.panes
    59→  return [{ symbol: tab.symbol, timeframe: tab.timeframe }]
    60→}
    61→
    62→/** Resolves the active grid layout for a tab ('1' when unset). */
    63→export function tabLayout(tab: Tab): GridLayoutId {
    64→  return tab.grid?.layout ?? '1'
    65→}

    67→/** Pane count for a layout id. */
    68→export function layoutPaneCount(id: GridLayoutId): number {
    69→  return GRID_LAYOUTS.find((l) => l.id === id)?.panes ?? 1

   301→export const useTerminal = create<TerminalState>((set, get) => ({
   302→  user: null,
   303→  hydrated: false,
   304→  setUser: (u) => set({ user: u }),
   305→  setHydrated: (v) => set({ hydrated: v }),
   306→
   307→  tabs: [{ id: uid('tab'), ...DEFAULT_TAB }],
   308→  activeTabId: '',
   309→
   310→  addTab: (type) => {
   311→    const t: Tab =
   312→      type === 'chart'
   313→        ? { id: uid('tab'), type, symbol: get().tabs.find((x) => x.id === get().activeTabId)?.symbol || 'AAPL', timeframe: get().tabs.find((x) => x.id === get().activeTabId)?.timeframe || '1D' }
   314→        : { id: uid('tab'), type, symbol: 'AAPL', timeframe: '1D' }
   315→    set((s) => ({ tabs: [...s.tabs, t], activeTabId: t.id, layoutDirty: true }))
   316→  },
   317→
   318→  closeTab: (id) => {
   319→    const { tabs, activeTabId } = get()
   320→    if (tabs.length <= 1) return
   321→    const idx = tabs.findIndex((t) => t.id === id)
   322→    const remaining = tabs.filter((t) => t.id !== id)
   323→    const nextActive = activeTabId === id ? remaining[Math.max(0, idx - 1)].id : activeTabId
   324→    set({ tabs: remaining, activeTabId: nextActive, layoutDirty: true })
   325→  },
   326→
   327→  setActiveTab: (id) => set({ activeTabId: id, layoutDirty: true }),
   328→
   329→  moveTab: (dragId, toIndex) =>
   330→    set((s) => {
   331→      const from = s.tabs.findIndex((t) => t.id === dragId)
   332→      if (from < 0) return {}
   333→      // remove first, then clamp the insertion point into the shortened array
   334→      const next = s.tabs.filter((t) => t.id !== dragId)
   335→      const idx = Math.max(0, Math.min(toIndex, next.length))
   336→      next.splice(idx, 0, s.tabs[from])
   337→      if (next.every((t, i) => t.id === s.tabs[i].id)) return {} // no-op order
   338→      return { tabs: next, layoutDirty: true }
   339→    }),
   340→
   341→  setTabSymbol: (tabId, symbol) =>
   342→    set((s) => ({
   343→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, symbol } : t)),
   344→      layoutDirty: true,
   345→    })),
   346→
   347→  setTabTimeframe: (tabId, tf) =>
   348→    set((s) => ({
   349→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, timeframe: tf } : t)),
   350→      layoutDirty: true,
   351→    })),
   352→
   353→  // ---------- multi-chart grid ----------
   354→  setGridLayout: (tabId, layout) =>
   355→    set((s) => ({
   356→      tabs: s.tabs.map((t) => {
   357→        if (t.id !== tabId || t.type !== 'chart') return t
   358→        const panes = tabPanes(t)
   359→        const count = layoutPaneCount(layout)
   360→        const next = [...panes]
   361→        while (next.length < count) next.push({ ...panes[0] })
   362→        next.length = count
   363→        if (layout === '1') {
   364→          // single pane mirrors the legacy tab fields so other panels keep working
   365→          return { ...t, symbol: next[0].symbol, timeframe: next[0].timeframe, grid: { layout, panes: next } }
   366→        }
   367→        // new arrangement → start from the default proportions (stale size
   368→        // keys from the previous arrangement would fight the new groups)
   369→        return { ...t, grid: { layout, panes: next } }
   370→      }),
   371→      layoutDirty: true,
   372→    })),
   373→
   374→  setPaneSymbol: (tabId, paneIdx, symbol) =>
   375→    set((s) => ({
   376→      tabs: s.tabs.map((t) => {
   377→        if (t.id !== tabId) return t
   378→        const panes = tabPanes(t)
   379→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   380→        if (panes[i].symbol === symbol) return t
   381→        const next = panes.map((p, j) => (j === i ? { ...p, symbol } : p))
   382→        // legacy single-chart mode keeps tab fields in sync
   383→        if (!t.grid || t.grid.layout === '1') return { ...t, symbol, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   384→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   385→      }),
   386→      layoutDirty: true,
   387→    })),
   388→
   389→  setPaneTimeframe: (tabId, paneIdx, tf) =>
   390→    set((s) => ({
   391→      tabs: s.tabs.map((t) => {
   392→        if (t.id !== tabId) return t
   393→        const panes = tabPanes(t)
   394→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   395→        if (panes[i].timeframe === tf) return t
   396→        const next = panes.map((p, j) => (j === i ? { ...p, timeframe: tf } : p))
   397→        if (!t.grid || t.grid.layout === '1') return { ...t, timeframe: tf, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   398→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   399→      }),
   400→      layoutDirty: true,
   401→    })),
   402→
   403→  setPaneChartType: (tabId, paneIdx, chartType) =>
   404→    set((s) => ({
   405→      tabs: s.tabs.map((t) => {
   406→        if (t.id !== tabId) return t
   407→        const panes = tabPanes(t)
   408→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   409→        if (panes[i].chartType === chartType) return t
   410→        const next = panes.map((p, j) => (j === i ? { ...p, chartType } : p))
   411→        if (!t.grid || t.grid.layout === '1') return { ...t, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   412→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   413→      }),
   414→      layoutDirty: true,
   415→    })),
   416→
   417→  setGridSizes: (tabId, key, sizes) =>
   418→    set((s) => ({
   419→      tabs: s.tabs.map((t) => {
   420→        if (t.id !== tabId || t.type !== 'chart') return t
   421→        const clean = sizes.slice(0, 8).map((n) => (Number.isFinite(n) ? Math.min(97, Math.max(3, Math.round(n * 10) / 10)) : 50))
   422→        if (clean.length < 2) return t
   423→        const prev = t.grid?.sizes?.[key]
   424→        if (prev && prev.length === clean.length && prev.every((v, i) => Math.abs(v - clean[i]) < 0.05)) return t
   425→        return {
   426→          ...t,
   427→          grid: { layout: t.grid?.layout ?? '1', panes: tabPanes(t), sizes: { ...(t.grid?.sizes ?? {}), [key]: clean } },
   428→        }
   429→      }),
   430→      layoutDirty: true,
   431→    })),
   432→
   433→  setFocusedPane: (tabId, paneIdx) =>
   434→    set((s) => ({
   435→      tabs: s.tabs.map((t) => (t.id === tabId && t.focusedPane !== paneIdx ? { ...t, focusedPane: paneIdx } : t)),
   436→      layoutDirty: true,
   437→    })),
   438→
   439→  setSyncCrosshair: (tabId, v) =>
   440→    set((s) => ({
   441→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, syncCrosshair: v } : t)),
   442→      layoutDirty: true,
   443→    })),
   444→
   445→  selectSymbol: (symbol) => {
   446→    // Task 56: let the mobile shell know a symbol was picked from a full-screen
   447→    // tool (heat-map tile, calendar row, …) so it can switch to the chart view.
   448→    // Desktop has no listener — harmless CustomEvent.
   449→    if (typeof window !== 'undefined') {
   450→      window.dispatchEvent(new CustomEvent('tt:symbol-picked', { detail: symbol }))
   451→    }
   452→    const { activeTabId, tabs } = get()
   453→    const active = tabs.find((t) => t.id === activeTabId)
   454→    if (active && active.type === 'chart') {
   455→      // target the focused pane so multi-chart grids follow watchlist clicks
   456→      const panes = tabPanes(active)
   457→      const idx = Math.min(active.focusedPane ?? 0, panes.length - 1)
   458→      get().setPaneSymbol(active.id, idx, symbol)
   459→    } else {
   460→      const chartTab = tabs.find((t) => t.type === 'chart')
   461→      if (chartTab) {
   462→        // Activate the first chart tab and route through setPaneSymbol so the
   463→        // tab's PANES are updated too — setting only `symbol` left
   464→        // grid.panes[0] on the old symbol (tab label said XOM, chart showed
   465→        // AAPL after clicking a heatmap tile from the heatmap tab).
   466→        get().setPaneSymbol(chartTab.id, chartTab.focusedPane ?? 0, symbol)
   467→        set((s) => ({ activeTabId: chartTab.id, layoutDirty: true }))
   468→      }
   469→    }
   470→  },
   471→
   472→  panelOrder: DEFAULT_LAYOUT.sidebar.panelOrder,
   473→  panelCollapsed: {},
   474→  mainSizes: DEFAULT_LAYOUT.sidebar.mainSizes,
   475→
   476→  setPanelOrder: (order) => set({ panelOrder: order, layoutDirty: true }),
   477→

   739→  reorderWatchlistSymbol: async (watchlistId, from, to) => {
   740→    const list = get().watchlists.find((l) => l.id === watchlistId)
   741→    if (!list) return
   742→    if (from < 0 || from >= list.items.length) return
   743→    const to2 = Math.max(0, Math.min(list.items.length, to))
   744→    if (to2 === from || to2 === from + 1) return // same place — no-op
   745→    const isSep = (s: string) => s.startsWith('#')
   746→    const items = [...list.items]
   747→    if (isSep(items[from])) {
   748→      // dragging a SEPARATOR moves the whole group block (marker + its symbols)
   749→      let end = from + 1
   750→      while (end < items.length && !isSep(items[end])) end++
   751→      const block = items.splice(from, end - from)
   752→      // map the boundary into the array AFTER the block was extracted, then
   753→      // apply the same moving-down offset as single-row moves
   754→      const b = to2 > from ? Math.max(from, to2 - block.length) : to2
   755→      const ins = Math.max(0, Math.min(items.length, from < to2 ? b - 1 : b))
   756→      items.splice(ins, 0, ...block)
   757→    } else {
   758→      const [sym] = items.splice(from, 1)
   759→      // moving DOWN: removing the item first shifts the target left by one
   760→      items.splice(from < to2 ? to2 - 1 : to2, 0, sym)
   761→    }
   762→    set((s) => ({ watchlists: s.watchlists.map((l) => (l.id === watchlistId ? { ...l, items } : l)) }))
   763→    try {
   764→      const res = await fetch(`/api/watchlists/${watchlistId}`, {
   765→        method: 'PATCH',
   766→        headers: { 'content-type': 'application/json' },
   767→        body: JSON.stringify({ items }),
   768→      })
   769→      if (!res.ok) await get().loadWatchlists() // resync to source of truth
   770→    } catch {
   771→      /* offline — optimistic order stays; next action resyncs */
   772→    }
