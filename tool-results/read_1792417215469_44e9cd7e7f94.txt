     1→// Alpaca Trading API client (server-side only): paper + live order routing,
     2→// positions, bracket/OCO exits, order replacement for SL/TP drag updates.
     3→//
     4→// Verified against the paper API (2026-10):
     5→// - Stocks support order_class=bracket (entry + take_profit/stop_loss legs).
     6→// - Crypto rejects ALL advanced order classes (order_class=oco -> 40010001
     7→//   "invalid order type for crypto order"; bracket -> 42210000) AND plain
     8→//   `stop` type. Sell exits also CANNOT be reserved against a PENDING buy
     9→//   (40310000 "insufficient balance" — only settled position qty counts),
    10→//   so crypto complex orders are EMULATED: the entry carries its intended
    11→//   TP/SL in the client_order_id (`ttb-…`) and the OCO supervisor below
    12→//   attaches the exit legs (`tte-…`) the moment the entry FILLS, canceling
    13→//   any surviving leg once the position is gone.
    14→// - PATCH /v2/orders/{id} creates a REPLACEMENT order with a NEW id.
    15→// - Position symbols drop the crypto slash (BTC/USD -> BTCUSD); ORDER
    16→//   symbols keep it (BTC/USD).
    17→import { db } from './db'
    18→import { getAlpacaKeys } from './alpaca'

    40→export async function resolveTradeMode(userId: string | undefined, requested?: unknown): Promise<TradeMode> {
    41→  if (isTradeMode(requested)) return requested

    55→interface TradeFetchResult<T> {
    56→  data: T | null
    57→  status: number
    58→  error?: string
    59→}
    60→
    61→async function tradeFetch<T>(
    62→  userId: string,
    63→  mode: TradeMode,
    64→  path: string,
    65→  init: RequestInit = {},
    66→): Promise<TradeFetchResult<T>> {
    67→  const keys = await getAlpacaKeys(userId)
    68→  if (!keys) return { data: null, status: 0, error: 'no-keys' }
    69→  const controller = new AbortController()
    70→  const timer = setTimeout(() => controller.abort(), 9000)
    71→  try {
    72→    const res = await fetch(`${TRADE_BASE[mode]}${path}`, {
    73→      ...init,
    74→      headers: {
    75→        'APCA-API-KEY-ID': keys.keyId,
    76→        'APCA-API-SECRET-KEY': keys.secret,
    77→        accept: 'application/json',
    78→        ...(init.body ? { 'content-type': 'application/json' } : {}),
    79→        ...(init.headers || {}),
    80→      },
    81→      signal: controller.signal,
    82→      cache: 'no-store',
    83→    })
    84→    const text = await res.text()
    85→    let parsed: unknown = null
    86→    try {
    87→      parsed = text ? JSON.parse(text) : null
    88→    } catch {
    89→      parsed = null
    90→    }
    91→    if (!res.ok) {
    92→      const msg =
    93→        (parsed as { message?: string } | null)?.message ||
    94→        `Trading API HTTP ${res.status}`
    95→      console.error(`Trading ${mode} ${path} -> ${res.status}: ${msg}`)
    96→      return { data: parsed as T, status: res.status, error: msg }
    97→    }
    98→    return { data: parsed as T, status: res.status }
    99→  } catch (e) {
   100→    console.error(`Trading ${mode} ${path} failed:`, e instanceof Error ? e.message : e)
   101→    return { data: null, status: 0, error: e instanceof Error ? e.message : 'network error' }
   102→  } finally {
   103→    clearTimeout(timer)
   104→  }
   105→}

   109→/**
   110→ * Alpaca rejects prices with more than 9 decimal places ("limit price exceeds
   111→ * maximum precision of 9 decimal places") on CRYPTO, and sub-penny increments
   112→ * on STOCKS ("sub-penny increment does not fulfill minimum pricing criteria").
   113→ * Chart drag prices are raw JS floats (65800.22400000001) that blow past both —
   114→ * first the direct PATCH fails, then the cancel+re-place fallback fails too,
   115→ * LEAVING THE EXIT ORDER CANCELED AND LOST. Round every outgoing price string:
   116→ * crypto 6dp (>=1) / 8dp (<1), stocks penny 2dp (>=1) / 4dp (<1).
   117→ */
   118→export function fmtPx(v: number, crypto = false): string {
   119→  if (!Number.isFinite(v) || v <= 0) return String(v)
   120→  const dp = crypto ? (v >= 1 ? 6 : 8) : v >= 1 ? 2 : 4
   121→  return String(parseFloat(v.toFixed(dp)))
   122→}

   217→export async function getOpenOrders(userId: string, mode: TradeMode, symbol?: string) {
   218→  const q = symbol ? `&symbols=${encodeURIComponent(symbol)}` : ''
   219→  // Alpaca hides bracket legs from the DEFAULT (nested=false) list: an open
   220→  // bracket parent comes back with legs=null, so pending-entry TP/SL (and any
   221→  // exit mapped from legs) is invisible to callers. The nested=true variant
   222→  // nests the legs under the parent — but a FILLED parent's ACTIVE legs only
   223→  // show in the flat list. Fetch BOTH and merge by id (nested wins when it
   224→  // carries legs) so both shapes are always covered.
   225→  const [flat, nested] = await Promise.all([
   226→    tradeFetch<TradingOrder[]>(userId, mode, `/v2/orders?status=open&limit=100${q}`),
   227→    tradeFetch<TradingOrder[]>(userId, mode, `/v2/orders?status=open&limit=100&nested=true${q}`),
   228→  ])
   229→  const flatRows: TradingOrder[] = Array.isArray(flat.data) ? flat.data : []
   230→  const nestedRows: TradingOrder[] = Array.isArray(nested.data) ? nested.data : []
   231→  if (nestedRows.length === 0) return flat
   232→  const byId = new Map<string, TradingOrder>()
   233→  for (const o of flatRows) byId.set(o.id, o)
   234→  for (const o of nestedRows) {
   235→    const prev = byId.get(o.id)
   236→    byId.set(o.id, !prev || (o.legs?.length ?? 0) > 0 ? o : prev)
   237→  }
   238→  return { ...flat, data: [...byId.values()] }
   239→}

   262→/**
   263→ * Place an entry order with optional exits.
   264→ * Stocks: single bracket request when TP/SL present (Alpaca attaches legs).
   265→ * Crypto: advanced classes are rejected — the entry is placed ALONE carrying
   266→ * the intended TP/SL in its client_order_id (`ttb-…`); the OCO supervisor
   267→ * (ensureCryptoOcoSupervisor) attaches the exit legs when the entry fills.
   268→ */
   269→export async function placeOrderWithExits(
   270→  userId: string,
   271→  mode: TradeMode,
   272→  body: PlaceOrderBody,
   273→): Promise<{ results: TradingOrder[]; errors: string[]; splitNote?: string }> {
   274→  const isCrypto = body.symbol.includes('/')
   275→  const type = body.type || 'market'
   276→  const tif = body.time_in_force || (isCrypto ? 'gtc' : type === 'market' ? 'day' : 'gtc')
   277→  const entry: Record<string, unknown> = {
   278→    symbol: tradeSymbol(body.symbol),
   279→    qty: String(body.qty),
   280→    side: body.side,
   281→    type,
   282→    time_in_force: tif,
   283→  }
   284→  if (type === 'limit') {
   285→    if (!body.limit_price) return { results: [], errors: ['Limit price required for limit orders'] }
   286→    entry.limit_price = String(body.limit_price)
   287→  }
   288→  if (body.client_order_id) entry.client_order_id = body.client_order_id
   289→
   290→  const errors: string[] = []
   291→  const results: TradingOrder[] = []
   292→
   293→  if (!isCrypto) {
   294→    // Stocks: bracket handles entry + legs atomically.
   295→    const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   296→    const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   297→    if (hasTp || hasSl) {
   298→      entry.order_class = 'bracket'
   299→      if (hasTp) entry.take_profit = { limit_price: body.take_profit!.toFixed(2) }
   300→      if (hasSl) entry.stop_loss = { stop_price: body.stop_loss!.toFixed(2) }
   301→      // encode the plan for the chart's pending display: Alpaca materializes
   302→      // bracket LEGS only when the entry activates, so a queued (weekend /
   303→      // pre-market) bracket would otherwise render without its TP/SL lines
   304→      entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   305→    }
   306→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   307→      method: 'POST',
   308→      body: JSON.stringify(entry),
   309→    })
   310→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   311→    else errors.push(r.error || 'Entry order failed')
   312→    return { results, errors }
   313→  }
   314→
   315→  // ---- Crypto: complex orders are EMULATED (no native bracket/OCO) ----
   316→  const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   317→  const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   318→  if (hasTp || hasSl) {
   319→    // encode the planned exits so the chart can render them while the entry
   320→    // is pending AND the supervisor can attach the legs without extra state
   321→    entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   322→  }
   323→  const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   324→    method: 'POST',
   325→    body: JSON.stringify(entry),
   326→  })
   327→  if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   328→  else {
   329→    errors.push(r.error || 'Entry order failed')
   330→    return { results, errors }
   331→  }
   332→  if (!hasTp && !hasSl) return { results, errors }
   333→
   334→  // Fast path: a MARKET entry usually fills instantly — poll briefly and
   335→  // attach the exits in-line so protection starts immediately. Limit entries
   336→  // return right away (the supervisor attaches on fill within ~6s).
   337→  if (type === 'market') {
   338→    let cur: TradingOrder | null = r.data
   339→    for (let i = 0; i < 4 && cur && cur.status !== 'filled'; i++) {
   340→      await new Promise((res) => setTimeout(res, 600))
   341→      const s = await tradeFetch<TradingOrder>(userId, mode, `/v2/orders/${(r.data as TradingOrder).id}`)
   342→      cur = s.data ?? cur
   343→      if (cur && ['canceled', 'cancelled', 'rejected', 'expired'].includes(cur.status)) break
   344→    }
   345→    if (cur?.status === 'filled') {
   346→      const legErrs = await placeCryptoExitLegs(
   347→        userId,
   348→        mode,
   349→        cur,
   350→        hasTp ? body.take_profit : undefined,
   351→        hasSl ? body.stop_loss : undefined,
   352→      )
   353→      errors.push(...legErrs)
   354→      return { results, errors }
   355→    }
   356→  }
   357→  return {
   358→    results,
   359→    errors,
   360→    splitNote: 'TP/SL attach automatically when the entry fills (OCO-managed)',
   361→  }
   362→}

   364→// ---------- crypto complex-order emulation (OCO supervisor) ----------
   365→
   366→/** Price -> client_order_id-safe digits (6dp fixed point). */
   367→function encP(v: number): string {
   368→  return String(Math.round(v * 1e6))
   369→}
   370→
   371→/** client_order_id prefix identifying a MANAGED crypto exit leg. */
   372→function exitLegClientId(entryOrderId: string, kind: 'tp' | 'sl'): string {
   373→  return `tte-${entryOrderId}-${kind}`
   374→}
   375→
   376→/** Encode the OCO/bracket plan (mode + TP/SL prices) into the entry's client_order_id. */
   377→function encodePlanClientId(mode: TradeMode, tp: number | null, sl: number | null): string {
   378→  const rand = Math.random().toString(36).slice(2, 8)
   379→  return `ttb-${mode[0]}-tp${tp != null && tp > 0 ? encP(tp) : '0'}-sl${sl != null && sl > 0 ? encP(sl) : '0'}-${rand}`
   380→}
   381→
   382→/** Decode the plan from a `ttb-…` client_order_id. Returns null when not ours. */
   383→function parseCryptoPlan(clientOrderId: string | undefined | null): { tp: number | null; sl: number | null } | null {
   384→  const m = /^ttb-[a-z](?:-tp(\d+))?(?:-sl(\d+))?-/.exec(clientOrderId || '')
   385→  if (!m) return null
   386→  const dec = (s: string | undefined) => (s ? parseInt(s, 10) / 1e6 : 0)
   387→  const tp = dec(m[1])
   388→  const sl = dec(m[2])
   389→  return { tp: tp > 0 ? tp : null, sl: sl > 0 ? sl : null }
   390→}
   391→
   392→/**
   393→ * Place the exit legs for a FILLED crypto entry (50/50 qty split when both
   394→ * legs are requested — Alpaca reserves balance per open sell order). Leg ids
   395→ * are deterministic (`tte-<entryId>-tp|-sl`) so the supervisor is idempotent.
   396→ * Crypto buys receive qty minus a trading fee, so legs apply a 0.99 safety
   397→ * factor to avoid 403 "insufficient balance".
   398→ */
   399→async function placeCryptoExitLegs(
   400→  userId: string,
   401→  mode: TradeMode,
   402→  entry: TradingOrder,
   403→  tp?: number,
   404→  sl?: number,
   405→): Promise<string[]> {
   406→  const errors: string[] = []
   407→  const filledQty = parseFloat(entry.filled_qty) || parseFloat(entry.qty) || 0
   408→  if (!(filledQty > 0)) return ['entry has no filled qty']
   409→  const exitSide = entry.side === 'buy' ? 'sell' : 'buy'
   410→  const both = tp != null && sl != null
   411→  const legQty = ((both ? filledQty / 2 : filledQty) * 0.99).toFixed(8).replace(/0+$/, '').replace(/\.$/, '')
   412→  if (tp != null) {
   413→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   414→      method: 'POST',
   415→      body: JSON.stringify({
   416→        symbol: entry.symbol,
   417→        qty: legQty,
   418→        side: exitSide,
   419→        type: 'limit',
   420→        time_in_force: 'gtc',
   421→        limit_price: tp.toFixed(tp < 1 ? 6 : 2),
   422→        client_order_id: exitLegClientId(entry.id, 'tp'),
   423→      }),
   424→    })
   425→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`TP: ${r.error || 'failed'}`)
   426→  }
   427→  if (sl != null) {
   428→    // stop-limit sell: limit slightly below the stop so it clears the book
   429→    const limP = exitSide === 'sell' ? sl * 0.999 : sl * 1.001
   430→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   431→      method: 'POST',
   432→      body: JSON.stringify({
   433→        symbol: entry.symbol,
   434→        qty: legQty,
   435→        side: exitSide,
   436→        type: 'stop_limit',
   437→        time_in_force: 'gtc',
   438→        stop_price: sl.toFixed(sl < 1 ? 6 : 2),
   439→        limit_price: limP.toFixed(limP < 1 ? 6 : 2),
   440→        client_order_id: exitLegClientId(entry.id, 'sl'),
   441→      }),
   442→    })
   443→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`SL: ${r.error || 'failed'}`)
   444→  }
   445→  return errors
   446→}

   448→const OCO_CYCLE_MS = 6000
   449→const OCO_OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   450→// last attach-failure signature per entry id — the supervisor retries every
   451→// cycle, but transient rejections (e.g. Alpaca's "potential wash trade
   452→// detected" while a crossing pending entry rests) must not spam the log
   453→const attachWarnSig = new Map<string, string>()

   455→/** Supervised (user,mode) pairs — survives dev hot reloads via globalThis. */
   456→function supervisedSet(): Set<string> {
   457→  const g = globalThis as { __ttOcoSupervised?: Set<string> }
   458→  if (!g.__ttOcoSupervised) g.__ttOcoSupervised = new Set<string>()
   459→  return g.__ttOcoSupervised
   460→}

   462→/**
   463→ * Start the OCO supervisor for a (user, mode) pair (idempotent). One cycle
   464→ * every 6s: (a) cancel managed exit legs whose position is gone (OCO
   465→ * semantics — when the position closes, the surviving leg dies),
   466→ * (b) attach exit legs to managed entries that have FILLED.
   467→ */
   468→export function ensureCryptoOcoSupervisor(userId: string, mode: TradeMode): void {
   469→  const key = `${userId}|${mode}`
   470→  if (supervisedSet().has(key)) return
   471→  supervisedSet().add(key)
   472→  const timer = setInterval(() => {
   473→    void runOcoCycle(userId, mode)
   474→  }, OCO_CYCLE_MS)
   475→  // never keep the process alive just for the supervisor
   476→  if (typeof timer === 'object' && timer !== null && 'unref' in timer) {
   477→    ;(timer as { unref: () => void }).unref()
   478→  }
   479→}
   480→
   481→async function runOcoCycle(userId: string, mode: TradeMode): Promise<void> {
   482→  try {
   483→    const [ordRes, posRes] = await Promise.all([
   484→      tradeFetch<TradingOrder[]>(userId, mode, '/v2/orders?status=all&limit=500&direction=desc'),
   485→      tradeFetch<TradingPosition[]>(userId, mode, '/v2/positions'),
   486→    ])
   487→    const orders: TradingOrder[] = Array.isArray(ordRes.data) ? ordRes.data : []
   488→    const positions: TradingPosition[] = Array.isArray(posRes.data) ? posRes.data : []
   489→    const norm = (s: string) => s.replace('/', '').toUpperCase()
   490→    const posSet = new Set(positions.map((p) => norm(p.symbol)))
   491→    const knownCoids = new Set(orders.map((o) => o.client_order_id || ''))
   492→    const isCrypto = (o: TradingOrder) => o.asset_class === 'crypto' || o.symbol.includes('/')
   493→
   494→    // (a) OCO cleanup: a managed exit leg dies with its position
   495→    for (const o of orders) {
   496→      if (!isCrypto(o) || !OCO_OPEN_STATUSES.includes(o.status)) continue
   497→      if (!(o.client_order_id || '').startsWith('tte-')) continue
   498→      if (!posSet.has(norm(o.symbol))) {
   499→        const r = await cancelOrder(userId, mode, o.id)
   500→        console.log(`[oco] canceled orphaned exit ${o.client_order_id} (${o.symbol}) -> ${r.status}`)
   501→      }
   502→    }
   503→
   504→    // (b) attach exit legs to managed entries that have FILLED
   505→    for (const o of orders) {
   506→      if (!isCrypto(o) || o.status !== 'filled') continue
   507→      const plan = parseCryptoPlan(o.client_order_id)
   508→      if (!plan || (plan.tp == null && plan.sl == null)) continue
   509→      const wantTp = plan.tp != null && !knownCoids.has(exitLegClientId(o.id, 'tp'))
   510→      const wantSl = plan.sl != null && !knownCoids.has(exitLegClientId(o.id, 'sl'))
   511→      if (!wantTp && !wantSl) continue
   512→      const errs = await placeCryptoExitLegs(
   513→        userId,
   514→        mode,
   515→        o,
   516→        wantTp ? plan.tp! : undefined,
   517→        wantSl ? plan.sl! : undefined,
   518→      )
   519→      if (errs.length) {
   520→        const sig = errs.join('; ')
   521→        if (attachWarnSig.get(o.id) !== sig) {
   522→          attachWarnSig.set(o.id, sig)
   523→          console.warn(`[oco] exit attach failed for ${o.id} (${o.symbol}):`, sig)
   524→        }
   525→      } else {
   526→        attachWarnSig.delete(o.id)
   527→        console.log(`[oco] attached exits to filled entry ${o.id} (${o.symbol}) tp=${plan.tp} sl=${plan.sl}`)
   528→      }
   529→    }
   530→  } catch (e) {
   531→    console.warn('[oco] supervisor cycle failed:', e instanceof Error ? e.message : e)
   532→  }
   533→}

   554→/** Rebuild an open exit order body from an existing order (used by the
   555→ * cancel+re-place fallback and OCO-sibling regeneration). */
   556→function exitBodyFrom(
   557→  orig: TradingOrder,
   558→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number },
   559→): Record<string, unknown> | null {
   560→  const isCrypto = orig.asset_class === 'crypto' || orig.symbol.includes('/')
   561→  const num = (v: string | null | undefined): number | null => {
   562→    if (v == null) return null
   563→    const n = parseFloat(v)
   564→    return Number.isFinite(n) && n > 0 ? n : null
   565→  }
   566→  const newStop = patch.stop_price ?? num(orig.stop_price)
   567→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   568→  const newTrail = patch.trail_percent ?? num(orig.trail_percent)
   569→  const body: Record<string, unknown> = {
   570→    symbol: orig.symbol,
   571→    qty: orig.qty,
   572→    side: orig.side,
   573→    type: orig.type,
   574→    time_in_force: orig.time_in_force || 'gtc',
   575→  }
   576→  if (orig.type === 'stop') {
   577→    if (newStop == null) return null
   578→    body.stop_price = fmtPx(newStop, isCrypto)
   579→  } else if (orig.type === 'stop_limit') {
   580→    if (newStop == null) return null
   581→    body.stop_price = fmtPx(newStop, isCrypto)
   582→    if (newLimit != null) {
   583→      body.limit_price = fmtPx(newLimit, isCrypto)
   584→    } else {
   585→      // preserve the original stop→limit offset when only the stop moves
   586→      const os = num(orig.stop_price)
   587→      const ol = num(orig.limit_price)
   588→      const ratio = os && ol ? ol / os : orig.side === 'sell' ? 0.999 : 1.001
   589→      body.limit_price = fmtPx(newStop * ratio, isCrypto)
   590→    }
   591→  } else if (orig.type === 'limit') {
   592→    if (newLimit == null) return null
   593→    body.limit_price = fmtPx(newLimit, isCrypto)
   594→  } else if (orig.type === 'trailing_stop') {
   595→    if (newTrail == null) return null
   596→    body.trail_percent = String(newTrail)
   597→  } else {
   598→    return null // market / unknown — never re-placed blindly
   599→  }
   600→  return body
   601→}

   612→/**
   613→ * Replace SL/TP prices on an open exit order — the chart line-drag path.
   614→ *
   615→ * Verified against the paper API (2026-10):
   616→ * - A PATCH on a STANDALONE exit (crypto split exits) replaces cleanly:
   617→ *   type/time_in_force/sibling prices are preserved and a NEW order id is
   618→ *   returned.
   619→ * - The chart drag used to 404 when the order lived on the OTHER host: the
   620→ *   client did not send `mode`, so the PATCH went to the stored trade-mode
   621→ *   preference — paper order ids do not exist on live and vice versa, the
   622→ *   update silently never landed on Alpaca.
   623→ * - Bracket OCO legs may refuse direct replacement; cancel+re-place is the
   624→ *   universal fallback (canceling one OCO leg can cascade-cancel the sibling,
   625→ *   which we detect and regenerate).
   626→ */
   627→export async function replaceExitOrderSmart(
   628→  userId: string,
   629→  mode: TradeMode,
   630→  orderId: string,
   631→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number; crypto?: boolean },
   632→): Promise<ReplaceExitResult> {
   633→  const patchBody: Record<string, unknown> = {}
   634→  if (patch.stop_price != null) patchBody.stop_price = fmtPx(patch.stop_price, patch.crypto)
   635→  if (patch.limit_price != null) patchBody.limit_price = fmtPx(patch.limit_price, patch.crypto)
   636→  if (patch.trail_percent != null) patchBody.trail_percent = String(parseFloat(patch.trail_percent.toFixed(4)))
   637→
   638→  // 1) PATCH on the requested host; 2) retry the other host on 404 —
   639→  // both hosts share the same key pair, so a wrong-host id self-heals here.
   640→  const order: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   641→  let last: TradeFetchResult<TradingOrder> | null = null
   642→  let errHost: TradeMode = mode
   643→  for (const m of order) {
   644→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`, {
   645→      method: 'PATCH',
   646→      body: JSON.stringify(patchBody),
   647→    })
   648→    if (r.data && (r.data as TradingOrder).id) {
   649→      return { data: r.data, status: r.status, mode: m, method: 'replace' }
   650→    }
   651→    last = r
   652→    // 422 "order parameters are not changed" — the order ALREADY carries the
   653→    // requested value (e.g. a second drag to the same level). That IS the
   654→    // desired end state: return the current order instead of churning a
   655→    // cancel+re-place cycle.
   656→    if (r.status === 422 && String(r.error || '').includes('not changed')) {
   657→      const cur = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   658→      if (cur.data) {
   659→        return { data: cur.data, status: 200, mode: m, method: 'replace', note: 'Order already at this value' }
   660→      }
   661→    }
   662→    if (r.status !== 404) {
   663→      // real rejection — the order lives on THIS host; run the fallback there
   664→      errHost = m
   665→      break
   666→    }
   667→  }
   668→
   669→  // 3) The id may have JUST been replaced (fast double-drag: Alpaca's
   670→  // cancel/replace is async and 404s for a beat) — follow the replaced_by
   671→  // chain and retry the direct PATCH on the live successor first.
   672→  const OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   673→  let orig: TradingOrder | null = null
   674→  let targetId = orderId
   675→  for (let hop = 0; hop < 4; hop++) {
   676→    const o = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`)
   677→    if (!o.data) break
   678→    const od = o.data
   679→    if (OPEN_STATUSES.includes(String(od.status))) {
   680→      orig = od
   681→      if (targetId !== orderId) {
   682→        // the open successor exists — retry the direct PATCH on it
   683→        const retry = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`, {
   684→          method: 'PATCH',
   685→          body: JSON.stringify(patchBody),
   686→        })
   687→        if (retry.data && (retry.data as TradingOrder).id) {
   688→          return { data: retry.data, status: 200, mode: errHost, method: 'replace', note: 'Applied to the replacement order' }
   689→        }
   690→        if (retry.status === 422 && String(retry.error || '').includes('not changed')) {
   691→          return { data: od, status: 200, mode: errHost, method: 'replace', note: 'Order already at this value' }
   692→        }
   693→        last = retry
   694→      }
   695→      break
   696→    }
   697→    const rb = od.replaced_by
   698→    if (typeof rb === 'string' && rb && rb !== targetId) {
   699→      targetId = rb
   700→      continue
   701→    }
   702→    orig = od // terminal status without a successor
   703→    break
   704→  }
   705→  if (!orig) {
   706→    return {
   707→      data: null,
   708→      status: last?.status || 502,
   709→      error: last?.error || 'Order not found on paper or live',
   710→      mode: errHost,
   711→      method: 'cancel-replace',
   712→    }
   713→  }
   714→  if (!OPEN_STATUSES.includes(String(orig.status))) {
   715→    return {
   716→      data: null,
   717→      status: 409,
   718→      error: `Order is ${orig.status} on ${errHost} — nothing to replace (refresh the chart)`,
   719→      mode: errHost,
   720→      method: 'cancel-replace',
   721→    }
   722→  }
   723→
   724→  // snapshot the OCO sibling BEFORE cancelling (limit TP vs stop-ish SL of
   725→  // the same symbol + same side) so a cascade-cancel can be healed
   726→  const openRes = await tradeFetch<TradingOrder[]>(userId, errHost, "/v2/orders?status=open&limit=500")
   727→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   728→  const isStopish = (t: string) => t === 'stop' || t === 'stop_limit' || t === 'trailing_stop'
   729→  const sibling = (openRes.data || []).find(
   730→    (o) =>
   731→      o.id !== orderId &&
   732→      norm(o.symbol) === norm(orig.symbol) &&
   733→      o.side === orig.side &&
   734→      ((isStopish(orig.type) && o.type === 'limit') || (orig.type === 'limit' && isStopish(o.type))),
   735→  )
   736→
   737→  const cancelRes = await cancelOrder(userId, errHost, targetId)
   738→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
   739→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode: errHost, method: 'cancel-replace' }
   740→  }
   741→
   742→  const newBody = exitBodyFrom(orig, patch)
   743→  if (!newBody) {
   744→    return { data: null, status: 400, error: `Unsupported exit type ${orig.type} for re-place`, mode: errHost, method: 'cancel-replace' }
   745→  }
   746→  // re-place on the host the order actually lives on (`mode` can differ when
   747→  // the PATCH self-healed across paper/live)
   748→  const r2 = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   749→    method: 'POST',
   750→    body: JSON.stringify(newBody),
   751→  })
   752→  if (!(r2.data && (r2.data as TradingOrder).id)) {
   753→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode: errHost, method: 'cancel-replace' }
   754→  }
   755→
   756→  // heal the sibling if the cancel cascaded (OCO group cancellation)
   757→  let note: string | undefined
   758→  if (sibling) {
   759→    const stillOpen = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${sibling.id}`)
   760→    const gone = !stillOpen.data || ['canceled', 'cancelled', 'expired', 'rejected'].includes(String(stillOpen.data.status))
   761→    if (gone) {
   762→      const sibBody = exitBodyFrom(sibling, {})
   763→      if (sibBody) {
   764→        const rs = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   765→          method: 'POST',
   766→          body: JSON.stringify(sibBody),
   767→        })
   768→        note = rs.data ? 'OCO sibling re-created' : `Warning: sibling exit lost (${rs.error || 're-place failed'})`
   769→      }
   770→    }
   771→  }
   772→  return { data: r2.data, status: r2.status, mode: errHost, method: 'cancel-replace', note }
   773→}

   799→/**
   800→ * Change the entry limit price and/or the planned TP/SL of a PENDING
   801→ * (unfilled) entry order — the chart drag on the dashed entry / (pend)
   802→ * TP / SL lines lands here.
   803→ *
   804→ * Verified against the paper API (2026-10, the "US-stock drag kills both
   805→ * TP/SL" bug):
   806→ * - Alpaca REFUSES a PATCH on a bracket parent in `accepted` status
   807→ *   (422 "cannot replace order in accepted status") — so EVERY stock
   808→ *   pending-order drag takes the cancel+re-place fallback below.
   809→ * - Canceling the parent cascades to the WHOLE family (parent + both legs).
   810→ * - The re-place POST is then validated by Alpaca against the entry price:
   811→ *   TP must be >= base+0.01 (long) / <= base-0.01 (short), SL mirrored.
   812→ *   A drag that crossed the entry made the POST 422 AFTER the family was
   813→ *   already canceled — both TP/SL lines vanished (crypto never hits this:
   814→ *   its plan rides in the client_order_id with no Alpaca-side validation).
   815→ *
   816→ * Therefore:
   817→ * - STOCK bracket parents: pre-validate the final entry/TP/SL combo BEFORE
   818→ *   any cancel and reject crossing drags with a clear message (the family
   819→ *   stays on the chart); a direct PATCH is still attempted first when the
   820→ *   order is in a replaceable state.
   821→ * - If the re-place POST fails for ANY reason after the cancel, the ORIGINAL
   822→ *   prices are re-posted (restore) so a transient rejection can never lose
   823→ *   the exits.
   824→ * - CRYPTO managed entries (`ttb-` plan in client_order_id): ALWAYS cancel →
   825→ *   wait → re-place with a freshly encoded plan. A direct PATCH replaces the
   826→ *   order WITHOUT carrying the client_order_id over (verified live: the
   827→ *   replacement lost the plan and the OCO supervisor would never attach the
   828→ *   exits), and client_order_id cannot be safely re-sent (uniqueness).
   829→ *   Re-placing keeps the REMAINING qty (a partially filled entry keeps its
   830→ *   fill).
   831→ * - Plain limit entries (no plan): direct PATCH on limit_price.
   832→ */
   833→export async function replacePendingEntrySmart(
   834→  userId: string,
   835→  mode: TradeMode,
   836→  orderId: string,
   837→  patch: { limit_price?: number; tp?: number; sl?: number; crypto?: boolean },
   838→): Promise<ReplaceEntryResult> {
   839→  // 1) locate the order (host fallback, same id spaces as exits)
   840→  const hosts: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   841→  let host: TradeMode = mode
   842→  let orig: TradingOrder | null = null
   843→  let last: TradeFetchResult<TradingOrder> | null = null
   844→  for (const m of hosts) {
   845→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   846→    if (r.data && (r.data as TradingOrder).id) {
   847→      orig = r.data
   848→      host = m
   849→      break
   850→    }
   851→    last = r
   852→  }
   853→  if (!orig) {
   854→    return { data: null, status: last?.status || 404, error: last?.error || 'Order not found on paper or live', mode }
   855→  }
   856→  const statusStr = String(orig.status)
   857→  if (!ENTRY_OPEN_STATUSES.includes(statusStr)) {
   858→    return { data: null, status: 409, error: `Order is ${statusStr} — nothing to change (refresh the chart)`, mode: host }
   859→  }
   860→
   861→  const isCrypto = orig.asset_class === 'crypto' || orig.symbol.includes('/')
   862→  const crypto = Boolean(patch.crypto) || isCrypto
   863→  const plan = parseCryptoPlan(orig.client_order_id)
   864→  const exitLegs = (orig.legs || []).filter((l) => l.side !== orig.side)
   865→  const num = (v: string | null | undefined): number | null => {
   866→    if (v == null) return null
   867→    const n = parseFloat(v)
   868→    return Number.isFinite(n) && n > 0 ? n : null
   869→  }
   870→  // pre-drag prices (for the restore-on-failure re-post)
   871→  const origLimit0 = num(orig.limit_price)
   872→  const origTp0 = num(exitLegs.find((l) => l.type === 'limit')?.limit_price) ?? plan?.tp ?? null
   873→  const origSl0 = num(exitLegs.find((l) => l.type === 'stop' || l.type === 'stop_limit')?.stop_price) ?? plan?.sl ?? null
   874→  // the order's ACTUAL current values (pre-patch) — used to verify "not changed"
   875→  const curTp0 = origTp0
   876→  const curSl0 = origSl0
   877→  // current TP/SL: requested patch wins, else the parent's legs, else the plan
   878→  const curTp =
   879→    patch.tp ??
   880→    num(exitLegs.find((l) => l.type === 'limit')?.limit_price) ??
   881→    plan?.tp ??
   882→    null
   883→  const curSl =
   884→    patch.sl ??
   885→    num(exitLegs.find((l) => l.type === 'stop' || l.type === 'stop_limit')?.stop_price) ??
   886→    plan?.sl ??
   887→    null
   888→  const origHasTp = curTp != null
   889→  const origHasSl = curSl != null
   890→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   891→  // a PATCH cannot ADD a bracket leg the order does not have — full re-place
   892→  const addsLeg = (patch.tp != null && !origHasTp) || (patch.sl != null && !origHasSl)
   893→  // crypto plan orders ALWAYS re-place: a direct PATCH drops the ttb-
   894→  // client_order_id (verified live), losing the TP/SL plan entirely
   895→  const managedCrypto = Boolean(plan)
   896→
   897→  // ---- STOCK bracket validation (BEFORE any cancel — this is the fix for
   898→  // the "drag kills both TP/SL" bug). Alpaca validates bracket legs against
   899→  // the entry price and the cancel+re-place fallback is UNAVOIDABLE for
   900→  // stock brackets (PATCH refused on accepted-status parents), so a crossing
   901→  // drag would cancel the family and then fail the re-place — total loss.
   902→  // Reject the drag up-front with an actionable message; nothing is canceled
   903→  // and the chart refresh restores the lines at their old prices.
   904→  const basePx = newLimit ?? (orig.type === 'stop_limit' || orig.type === 'stop' ? num(orig.stop_price) : null)
   905→  if (!crypto && basePx != null && (curTp != null || curSl != null)) {
   906→    const long = orig.side === 'buy'
   907→    const tick = 0.01
   908→    const entry = fmtPx(basePx, false)
   909→    if (curTp != null) {
   910→      const okTp = long ? curTp >= basePx + tick : curTp <= basePx - tick
   911→      if (!okTp) {
   912→        return {
   913→          data: null,
   914→          status: 422,
   915→          mode: host,
   916→          error: long
   917→            ? `Take-profit must stay above the entry price (${entry}) for a long entry — drag it above the entry line`
   918→            : `Take-profit must stay below the entry price (${entry}) for a short entry — drag it below the entry line`,
   919→          method: 'replace',
   920→          note: 'Rejected before cancel — order untouched',
   921→        }
   922→      }
   923→    }
   924→    if (curSl != null) {
   925→      const okSl = long ? curSl <= basePx - tick : curSl >= basePx + tick
   926→      if (!okSl) {
   927→        return {
   928→          data: null,
   929→          status: 422,
   930→          mode: host,
   931→          error: long
   932→            ? `Stop-loss must stay below the entry price (${entry}) for a long entry — drag it below the entry line`
   933→            : `Stop-loss must stay above the entry price (${entry}) for a short entry — drag it above the entry line`,
   934→          method: 'replace',
   935→          note: 'Rejected before cancel — order untouched',
   936→        }
   937→      }
   938→    }
   939→  }
   940→
   941→  // 2) direct PATCH — ONLY for entry-field changes (or orders without legs).
   942→  // Verified live 2026-10 (queued bracket parent, market closed): Alpaca
   943→  // APPLIES limit/qty changes but SILENTLY DROPS take_profit/stop_loss
   944→  // changes sent on the parent, and a leg-ONLY patch 422s with the
   945→  // misleading "order parameters are not changed" even when the values
   946→  // differ. So any leg change on a bracket parent MUST go through the
   947→  // cancel+re-place rebuild below, or it never lands (the "second SL/TP
   948→  // drag does nothing" bug).
   949→  const touchesLegs = patch.tp != null || patch.sl != null
   950→  const isBracketParent =
   951→    String(orig.order_class || '') === 'bracket' || (orig.legs?.length ?? 0) > 0
   952→  if (!addsLeg && !managedCrypto && !(isBracketParent && touchesLegs)) {
   953→    const body: Record<string, unknown> = {}
   954→    if (patch.limit_price != null && newLimit != null) body.limit_price = fmtPx(newLimit, crypto)
   955→    if (origHasTp) body.take_profit = { limit_price: fmtPx(curTp!, crypto) }
   956→    if (origHasSl) body.stop_loss = { stop_price: fmtPx(curSl!, crypto) }
   957→    if (Object.keys(body).length > 0) {
   958→      const r = await tradeFetch<TradingOrder>(userId, host, `/v2/orders/${orderId}`, {
   959→        method: 'PATCH',
   960→        body: JSON.stringify(body),
   961→      })
   962→      if (r.data && (r.data as TradingOrder).id) {
   963→        return { data: r.data, status: r.status, mode: host, method: 'replace' }
   964→      }
   965→      if (r.status === 422 && String(r.error || '').includes('not changed')) {
   966→        // only a genuine no-op when EVERY requested value already equals the
   967→        // order's current value — Alpaca also emits this message for changes
   968→        // it refuses to apply (e.g. leg-only patch on a queued bracket),
   969→        // which must fall through to the cancel+re-place rebuild
   970→        const samePx = (a: number | null | undefined, b: number | null | undefined) =>
   971→          a != null && b != null && Math.abs(a - b) < 1e-9
   972→        const noop =
   973→          (patch.limit_price == null || samePx(newLimit, origLimit0)) &&
   974→          (patch.tp == null || samePx(patch.tp, curTp0)) &&
   975→          (patch.sl == null || samePx(patch.sl, curSl0))
   976→        if (noop) {
   977→          return { data: orig, status: 200, mode: host, method: 'replace', note: 'Order already at this value' }
   978→        }
   979→      }
   980→      // real rejection (or a fake "not changed") → cancel+re-place below
   981→    } else {
   982→      return { data: null, status: 400, error: 'nothing to change', mode: host }
   983→    }
   984→  }
   985→
   986→  // 2b) pure leg no-op on a rebuilt order (bracket parent or managed crypto
   987→  // plan): the requested TP/SL already equals the current level — don't churn
   988→  // a cancel+re-place cycle for nothing (the chart would flash its lines).
   989→  // The leg values may live in the parent's legs OR the ttb- plan.
   990→  if (touchesLegs && patch.limit_price == null && (isBracketParent || managedCrypto)) {
   991→    const samePx = (a: number | null | undefined, b: number | null | undefined) =>
   992→      a != null && b != null && Math.abs(a - b) < 1e-9
   993→    const unchanged =
   994→      (patch.tp == null || samePx(patch.tp, curTp0)) &&
   995→      (patch.sl == null || samePx(patch.sl, curSl0))
   996→    if (unchanged) {
   997→      return { data: orig, status: 200, mode: host, method: 'replace', note: 'Order already at this value' }
   998→    }
   999→  }
  1000→
  1001→  // 3) cancel + re-place (plan change on crypto, new leg, or rejected PATCH)
  1002→  const cancelRes = await cancelOrder(userId, host, orderId)
  1003→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
  1004→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode: host, method: 'cancel-replace' }
  1005→  }
  1006→  if (cancelRes.status !== 404) await waitUntilCanceled(userId, host, orderId)
  1007→
  1008→  const remaining = (parseFloat(orig.qty) || 0) - (parseFloat(orig.filled_qty) || 0)
  1009→  if (!(remaining > 0)) {
  1010→    return { data: null, status: 200, mode: host, method: 'cancel-replace', note: 'Entry filled — nothing re-placed' }
  1011→  }
  1012→  const wantTp = patch.tp ?? curTp
  1013→  const wantSl = patch.sl ?? curSl
  1014→  const hadPlan = Boolean(plan) || (orig.legs?.length ?? 0) > 0 || patch.tp != null || patch.sl != null
  1015→  const buildPlaceBody = (limit: number | null, tp: number | null, sl: number | null): Record<string, unknown> => {
  1016→    const b: Record<string, unknown> = {
  1017→      symbol: orig.symbol,
  1018→      qty: String(remaining),
  1019→      side: orig.side,
  1020→      type: orig.type,
  1021→      time_in_force: orig.time_in_force || 'gtc',
  1022→    }
  1023→    if (limit != null && (orig.type === 'limit' || orig.type === 'stop_limit')) {
  1024→      b.limit_price = fmtPx(limit, crypto)
  1025→    }
  1026→    if (orig.type === 'stop' && sl != null) b.stop_price = fmtPx(sl, crypto)
  1027→    if (hadPlan) {
  1028→      if (isCrypto) {
  1029→        // re-encode the plan — the supervisor attaches legs from it on fill
  1030→        b.client_order_id = encodePlanClientId(host, tp ?? null, sl ?? null)
  1031→      } else {
  1032→        b.order_class = 'bracket'
  1033→        if (tp != null) b.take_profit = { limit_price: fmtPx(tp, false) }
  1034→        if (sl != null) b.stop_loss = { stop_price: fmtPx(sl, false) }
  1035→      }
  1036→    }
  1037→    return b
  1038→  }
  1039→
  1040→  const r2 = await tradeFetch<TradingOrder>(userId, host, '/v2/orders', {
  1041→    method: 'POST',
  1042→    body: JSON.stringify(buildPlaceBody(newLimit, wantTp, wantSl)),
  1043→  })
  1044→  if (!(r2.data && (r2.data as TradingOrder).id)) {
  1045→    // The family is ALREADY canceled (Alpaca cascades a parent cancel to both
  1046→    // legs) — a failed re-place here used to be the "US-stock drag makes both
  1047→    // TP/SL disappear" bug. Re-post the ORIGINAL prices so a rejected
  1048→    // replacement (validation, transient error) can never lose the exits.
  1049→    const rr = await tradeFetch<TradingOrder>(userId, host, '/v2/orders', {
  1050→      method: 'POST',
  1051→      body: JSON.stringify(buildPlaceBody(origLimit0, origTp0, origSl0)),
  1052→    })
  1053→    if (rr.data && (rr.data as TradingOrder).id) {
  1054→      return {
  1055→        data: rr.data,
  1056→        status: 200,
  1057→        mode: host,
  1058→        method: 'cancel-replace',
  1059→        note: `Change rejected (${r2.error || 'unknown'}) — original order restored`,
  1060→      }
  1061→    }
  1062→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode: host, method: 'cancel-replace' }
  1063→  }
  1064→  return { data: r2.data, status: r2.status, mode: host, method: 'cancel-replace' }
  1065→}
  1066→
  1067→/**
  1068→ * Market-close a position. Crypto balances are reserved per open exit order,
  1069→ * so lingering SL/TP legs would starve the liquidation (Alpaca sells only
  1070→ * qty_available) — cancel the symbol's opposite-side (exit) orders FIRST,
  1071→ * then close. Returns the close order and how many exits were cancelled.
  1072→ */
  1073→export async function closePositionFull(
  1074→  userId: string,
  1075→  mode: TradeMode,
  1076→  symbol: string,
  1077→): Promise<{ close: TradeFetchResult<TradingOrder>; canceledExits: number }> {
  1078→  const tSym = tradeSymbol(symbol)
  1079→  const norm = (s: string) => s.replace('/', '').toUpperCase()
  1080→  // snapshot the position side BEFORE closing (long -> sell exits, short -> buy)
  1081→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
  1082→  const posSide = posRes.data?.side
  1083→  let canceledExits = 0
  1084→  // fetch ALL open orders and filter here — Alpaca's `symbols=` query wants
  1085→  // the slash form for crypto (BTC/USD) while callers pass either form
  1086→  const openRes = await getOpenOrders(userId, mode)
  1087→  const orders: TradingOrder[] = (Array.isArray(openRes.data) ? openRes.data : [])
  1088→    .filter((o) => norm(o.symbol) === norm(tSym))
  1089→  const exitSide = posSide === 'long' ? 'sell' : posSide === 'short' ? 'buy' : null
  1090→  for (const o of orders) {
  1091→    if (exitSide && o.side !== exitSide) continue // keep same-side (entry) orders
  1092→    if (exitSide == null && o.type === 'market') continue // no position info: keep entries
  1093→    // `held` legs ride a PENDING (unfilled) entry family on this symbol —
  1094→    // canceling them would strip the pending entry of its TP/SL plan (the
  1095→    // nested merge now makes them visible here). Skip them.
  1096→    if (o.status === 'held') continue
  1097→    const c = await cancelOrder(userId, mode, o.id)
  1098→    if (c.status === 204 || c.status === 200) canceledExits += 1
  1099→  }
  1100→  const close = await closePosition(userId, mode, symbol)
  1101→  return { close, canceledExits }
  1102→}

  1104→export async function cancelOrder(userId: string, mode: TradeMode, orderId: string) {
  1105→  return tradeFetch<null>(userId, mode, `/v2/orders/${orderId}`, { method: 'DELETE' })
  1106→}

  1108→// ---------- attach TP/SL to an EXISTING position (chart [TP]/[SL] drag-add) ----------
  1109→
  1110→export interface AttachExitsResult {
  1111→  results: TradingOrder[]
  1112→  errors: string[]
  1113→  oco?: boolean
  1114→}
  1115→
  1116→/**
  1117→ * Attach a TP and/or SL to an open position — the chart's "+TP"/"+SL" chip
  1118→ * drag and the Positions-tab add buttons land here.
  1119→ *
  1120→ * Stocks (verified against the paper API 2026-10):
  1121→ * - order_class=oco REQUIRES take_profit.limit_price AND stop_loss.stop_price
  1122→ *   AND an existing position to exit (without one: "oco orders must be exit
  1123→ *   orders"; parent must be type=limit). With both legs requested we send the
  1124→ *   documented OCO exit body and FALL BACK to separate legs on rejection.
  1125→ * - A single leg is a plain limit (TP) / stop (SL) order on the full position.
  1126→ * Crypto: ALL advanced order classes are rejected — place split exit legs
  1127→ * (50/50 qty, 0.99 fee factor) directly against the position, tagged with a
  1128→ * `ttp-` client_order_id (ignored by the OCO supervisor's `tte-` cleanup).
  1129→ */
  1130→export async function attachExitsToPosition(
  1131→  userId: string,
  1132→  mode: TradeMode,
  1133→  symbol: string,
  1134→  exits: { tp?: number; sl?: number },
  1135→): Promise<AttachExitsResult> {
  1136→  const tSym = tradeSymbol(symbol)
  1137→  const hasTp = typeof exits.tp === 'number' && exits.tp > 0
  1138→  const hasSl = typeof exits.sl === 'number' && exits.sl > 0
  1139→  if (!hasTp && !hasSl) return { results: [], errors: ['no TP/SL price given'] }
  1140→
  1141→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
  1142→  if (!posRes.data) {
  1143→    return { results: [], errors: [posRes.error && posRes.error !== 'not found' ? posRes.error : 'Position not found'] }
  1144→  }
  1145→  const pos = posRes.data
  1146→  // crypto-ness comes from the POSITION (callers pass either symbol form —
  1147→  // the chart sends the trade symbol BTCUSD, whose crypto-ness the route
  1148→  // cannot infer from a slash)
  1149→  const isCrypto = pos.asset_class === 'crypto' || pos.symbol.includes('/')
  1150→  // orders use the position's own symbol form (crypto keeps the slash)
  1151→  const orderSymbol = pos.symbol
  1152→  // long position -> sell exits; short position -> buy exits
  1153→  const exitSide = pos.side === 'short' ? 'buy' : 'sell'
  1154→  const qtyAvail = parseFloat(pos.qty_available) || parseFloat(pos.qty) || 0
  1155→  if (!(qtyAvail > 0)) return { results: [], errors: ['position has no available qty'] }
  1156→
  1157→  const results: TradingOrder[] = []
  1158→  const errors: string[] = []
  1159→  const post = (body: Record<string, unknown>) =>
  1160→    tradeFetch<TradingOrder>(userId, mode, '/v2/orders', { method: 'POST', body: JSON.stringify(body) })
  1161→  const legQty = (n: number) => String(parseFloat((n * (isCrypto ? 0.99 : 1)).toFixed(8)))
  1162→
  1163→  if (!isCrypto) {
  1164→    if (hasTp && hasSl) {
  1165→      const r = await post({
  1166→        symbol: orderSymbol,
  1167→        qty: String(qtyAvail),
  1168→        side: exitSide,
  1169→        type: 'limit',
  1170→        time_in_force: 'gtc',
  1171→        limit_price: fmtPx(exits.tp!),
  1172→        order_class: 'oco',
  1173→        take_profit: { limit_price: fmtPx(exits.tp!) },
  1174→        stop_loss: { stop_price: fmtPx(exits.sl!) },
  1175→      })
  1176→      if (r.data && (r.data as TradingOrder).id) {
  1177→        return { results: [r.data], errors, oco: true }
  1178→      }
  1179→      errors.push(`OCO rejected, placing legs separately: ${r.error || 'failed'}`)
  1180→    }
  1181→    if (hasTp) {
  1182→      const r = await post({
  1183→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'limit',
  1184→        time_in_force: 'gtc', limit_price: fmtPx(exits.tp!),
  1185→      })
  1186→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1187→      else errors.push(`TP: ${r.error || 'failed'}`)
  1188→    }
  1189→    if (hasSl) {
  1190→      const r = await post({
  1191→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'stop',
  1192→        time_in_force: 'gtc', stop_price: fmtPx(exits.sl!),
  1193→      })
  1194→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1195→      else errors.push(`SL: ${r.error || 'failed'}`)
  1196→    }
  1197→    return { results, errors }
  1198→  }
  1199→
  1200→  // ---- crypto: emulated split legs (balance is reserved per open sell) ----
  1201→  // Each attached leg takes HALF the position (the two legs together cover
  1202→  // it), capped by what is still available — Alpaca reserves every open sell
  1203→  // against qty_available, so a full-size first leg would make attaching the
  1204→  // second one impossible (502 insufficient balance). Legs may be attached
  1205→  // independently in any order.
  1206→  const posQty = parseFloat(pos.qty) || qtyAvail
  1207→  const qty = legQty(Math.min(qtyAvail, posQty / 2))
  1208→  const coidBase = `ttp-${Math.random().toString(36).slice(2, 8)}`
  1209→  if (hasTp) {
  1210→    const r = await post({
  1211→      symbol: orderSymbol, qty, side: exitSide, type: 'limit', time_in_force: 'gtc',
  1212→      limit_price: fmtPx(exits.tp!, true),
  1213→      client_order_id: `${coidBase}-tp`,
  1214→    })
  1215→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1216→    else errors.push(`TP: ${r.error || 'failed'}`)
  1217→  }
  1218→  if (hasSl) {
  1219→    // stop-limit: book-clearing limit offset beyond the stop
  1220→    const limP = exitSide === 'sell' ? exits.sl! * 0.999 : exits.sl! * 1.001
  1221→    const r = await post({
  1222→      symbol: orderSymbol, qty, side: exitSide, type: 'stop_limit', time_in_force: 'gtc',
  1223→      stop_price: fmtPx(exits.sl!, true),
  1224→      limit_price: fmtPx(limP, true),
  1225→      client_order_id: `${coidBase}-sl`,
  1226→    })
  1227→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
  1228→    else errors.push(`SL: ${r.error || 'failed'}`)
  1229→  }
  1230→  return { results, errors }
  1231→}
