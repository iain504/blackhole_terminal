   312→  function segPath(ctx: CanvasRenderingContext2D, p1: { x: number; y: number }, p2: { x: number; y: number }, w: number, extendRight: boolean) {
   313→    ctx.beginPath()
   314→    ctx.moveTo(p1.x, p1.y)
   315→    if (extendRight) {
   316→      const dx = p2.x - p1.x
   317→      const dy = p2.y - p1.y
   318→      const farX = dx !== 0 ? (w - p1.x) / dx : Infinity
   319→      const ex = farX >= 0 ? w : p1.x - dx * 100
   320→      const ey = p1.y + dy * (farX >= 0 ? farX : -100)
   321→      ctx.lineTo(ex, ey)
   322→    } else {
   323→      ctx.lineTo(p2.x, p2.y)
   324→    }
   325→    ctx.stroke()
   326→  }

   328→  function drawDrawing(ctx: CanvasRenderingContext2D, d: Drawing, w: number, h: number, highlighted: boolean) {
   329→    ctx.save()
   330→    ctx.strokeStyle = d.color
   331→    ctx.lineWidth = d.width + (highlighted ? 1 : 0)
   332→    ctx.globalAlpha = 1
   333→
   334→    if (d.kind === 'hline') {
   335→      const y = yOf(d.points[0]?.price ?? 0)
   336→      if (y != null) {
   337→        hlinePath(ctx, y, w)
   338→        drawChip(ctx, w - 74, y, fmtPrice(d.points[0].price), d.color, highlighted)
   339→        // move grip dots at center
   340→        dot(ctx, w / 2, y, d.color, highlighted)
   341→      }
   342→    } else if (d.kind === 'vline') {
   343→      const x = xOf(d.points[0]?.time ?? null)
   344→      if (x != null) {
   345→        vlinePath(ctx, x, h)
   346→        dot(ctx, x, h / 2, d.color, highlighted)
   347→      }
   348→    } else if (d.kind === 'trend' || d.kind === 'fib') {
   349→      const p1 = pixelPoint(d.points[0])
   350→      const p2 = pixelPoint(d.points[1])
   351→      if (p1 && p2) {
   352→        if (d.kind === 'trend') {
   353→          segPath(ctx, p1, p2, w, false)
   354→        } else {
   355→          // fib: levels between p1..p2 extended to right edge
   356→          const xLeft = Math.min(p1.x, p2.x)
   357→          const price1 = d.points[0].price
   358→          const price2 = d.points[1].price
   359→          ctx.font = '10px ui-monospace, monospace'
   360→          for (const lv of FIB_LEVELS) {
   361→            const priceLv = price2 + (price1 - price2) * lv
   362→            const y = yOf(priceLv)
   363→            if (y == null) continue
   364→            ctx.globalAlpha = lv === 0 || lv === 1 ? 0.9 : 0.6
   365→            ctx.setLineDash(lv === 0.5 ? [6, 3] : [])
   366→            ctx.beginPath()
   367→            ctx.moveTo(xLeft, Math.round(y) + 0.5)
   368→            ctx.lineTo(w, Math.round(y) + 0.5)
   369→            ctx.stroke()
   370→            ctx.globalAlpha = 0.95
   371→            ctx.fillStyle = d.color
   372→            const label = `${lv.toFixed(3)}  ${fmtPrice(priceLv)}`
   373→            ctx.fillText(label, xLeft + 4, y - 3)
   374→          }
   375→        }
   376→        dot(ctx, p1.x, p1.y, d.color, highlighted)
   377→        dot(ctx, p2.x, p2.y, d.color, highlighted)
   378→      }
   379→    }
   380→    ctx.restore()
   381→  }

   383→  function drawChip(ctx: CanvasRenderingContext2D, x: number, y: number, text: string, color: string, highlighted: boolean) {
   384→    ctx.save()
   385→    const wTxt = ctx.measureText(text).width
   386→    const bw = Math.max(46, wTxt + 12)
   387→    const bh = 16
   388→    const by = Math.round(y) - bh / 2
   389→    ctx.fillStyle = color
   390→    ctx.globalAlpha = highlighted ? 1 : 0.9
   391→    ctx.beginPath()
   392→    ctx.roundRect(x, by, bw, bh, 3)
   393→    ctx.fill()
   394→    ctx.globalAlpha = 1
   395→    ctx.fillStyle = '#09090b'
   396→    ctx.font = '10px ui-monospace, monospace'
   397→    ctx.fillText(text, x + 6, by + 11.5)
   398→    ctx.restore()
   399→  }

   443→  // ---------- hit testing ----------
   444→  const distToSeg = (px: number, py: number, a: { x: number; y: number }, b: { x: number; y: number }) => {
   445→    const dx = b.x - a.x
   446→    const dy = b.y - a.y
   447→    const len2 = dx * dx + dy * dy
   448→    if (len2 === 0) return Math.hypot(px - a.x, py - a.y)
   449→    let t = ((px - a.x) * dx + (py - a.y) * dy) / len2
   450→    t = Math.max(0, Math.min(1, t))
   451→    return Math.hypot(px - (a.x + t * dx), py - (a.y + t * dy))
   452→  }

   464→      const slPrice = myExits.sl.stopPrice ?? myExits.sl.limitPrice!
   465→      if (myExits.sl.type !== 'trailing_stop') {
   466→        const yl = yOf(slPrice)
   467→        if (yl != null && Math.abs(y - yl) < 7) {
   468→          return { type: 'exit', id: myExits.sl.orderId, part: 'sl', origPrice: slPrice, exit: myExits.sl }
   469→        }
   470→      }
   471→    }
   472→    if (myExits?.tp && myExits.tp.limitPrice != null) {
   473→      const yl = yOf(myExits.tp.limitPrice)
   474→      if (yl != null && Math.abs(y - yl) < 7) {
   475→        return { type: 'exit', id: myExits.tp.orderId, part: 'tp', origPrice: myExits.tp.limitPrice, exit: myExits.tp }
   476→      }
   477→    }
   478→    if (myPos && myPos.avgEntry > 0) {
   479→      const yl = yOf(myPos.avgEntry)
   480→      if (yl != null && Math.abs(y - yl) < 6) {
   481→        return { type: 'position', id: myPos.tradeSymbol, part: 'whole', origPrice: myPos.avgEntry }
   482→      }
   483→    }
   484→
   485→    if (showDrawings) {
   486→      for (let i = ds.length - 1; i >= 0; i--) {
   487→        const d = ds[i]
   488→        if (d.kind === 'hline') {
   489→          const yl = yOf(d.points[0]?.price ?? 0)
   490→          if (yl != null && Math.abs(y - yl) < 7) return { type: 'drawing', id: d.id, part: 'whole' }
   491→        } else if (d.kind === 'vline') {
   492→          const xl = xOf(d.points[0]?.time ?? null)
   493→          if (xl != null && Math.abs(x - xl) < 7) return { type: 'drawing', id: d.id, part: 'whole' }
   494→        } else {
   495→          const p1 = pixelPoint(d.points[0])
   496→          const p2 = pixelPoint(d.points[1])
   497→          if (p1 && Math.hypot(x - p1.x, y - p1.y) < 8) return { type: 'drawing', id: d.id, part: 'point', index: 0 }
   498→          if (p2 && Math.hypot(x - p2.x, y - p2.y) < 8) return { type: 'drawing', id: d.id, part: 'point', index: 1 }
   499→          if (d.kind === 'trend' && p1 && p2 && distToSeg(x, y, p1, p2) < 7) {
   500→            return { type: 'drawing', id: d.id, part: 'whole' }
   501→          }
   502→          if (d.kind === 'fib' && p1 && p2) {
   503→            const xLeft = Math.min(p1.x, p2.x)
   504→            if (x >= xLeft - 4) {
   505→              const price1 = d.points[0].price
   506→              const price2 = d.points[1].price
   507→              for (const lv of FIB_LEVELS) {
   508→                const yl = yOf(price2 + (price1 - price2) * lv)
   509→                if (yl != null && Math.abs(y - yl) < 6) return { type: 'drawing', id: d.id, part: 'whole' }
   510→              }
   511→            }
   512→          }
   513→        }
   514→      }
   515→    }
   516→    for (let i = als.length - 1; i >= 0; i--) {
   517→      const yl = yOf(als[i].price)
   518→      if (yl != null && Math.abs(y - yl) < 7) return { type: 'alert', id: als[i].id, part: 'whole' }
   519→    }
   520→    return null
   521→
