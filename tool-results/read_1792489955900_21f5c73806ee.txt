   293→// Ticking local clock for the footer (updates every second).
   294→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   295→function LiveClock() {
   296→  const [now, setNow] = useState<Date | null>(null)
   297→  useEffect(() => {
   298→    setNow(new Date())
   299→    const timer = setInterval(() => setNow(new Date()), 1000)
   300→    return () => clearInterval(timer)
   301→  }, [])
   302→  const tz = useMemo(() => {
   303→    try {
   304→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   305→    } catch {
   306→      return 'Local'
   307→    }
   308→  }, [])
   309→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   310→  const time = now
   311→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   312→    : '--:--:--'
   313→  return (
   314→    <span
   315→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   316→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   317→    >
   318→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   319→      <span>{time}</span>
   320→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   321→    </span>
   322→  )
   323→}

   348→function useDemoMeta(): { demo: boolean; reason?: string } {
   349→  const [meta, setMeta] = useState<{ demo: boolean; reason?: string }>({ demo: false })
   350→  useEffect(() => {
   351→    let alive = true
   352→    const check = async () => {
   353→      try {
   354→        const res = await fetch('/api/market/quote', {
   355→          method: 'POST',
   356→          headers: { 'content-type': 'application/json' },
   357→          body: JSON.stringify({ symbols: ['AAPL'] }),
   358→        })
   359→        if (!res.ok || !alive) return
   360→        const json = await res.json()
   361→        if (alive) setMeta({ demo: Boolean(json.meta?.demo), reason: json.meta?.reason })
   362→      } catch { /* offline */ }
   363→    }
   364→    check()
   365→    // Re-check right after keys are saved/changed and periodically, so the
   366→    // badge never lies after Settings are updated without a page reload.
   367→    window.addEventListener('tt:settings-changed', check)
   368→    const timer = setInterval(check, 60000)
   369→    return () => {
   370→      alive = false
   371→      window.removeEventListener('tt:settings-changed', check)
   372→      clearInterval(timer)
   373→    }
   374→  }, [])
   375→  return meta
   376→}

   392→/** One-line human summary of an alarm (any kind) for the list. */
   393→function alertSummary(a: Alert): string {
   394→  if (a.kind === 'rsi') {
   395→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   396→  }
   397→  if (a.kind === 'macd') {
   398→    const p = a.params ?? {}
   399→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   400→    const cond =
   401→      a.condition === 'bull_cross' ? 'bullish crossover'
   402→      : a.condition === 'bear_cross' ? 'bearish crossover'
   403→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   404→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   405→      : 'crossover'
   406→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   407→  }
   408→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   409→}

   417→
   418→  // ---- creation form state ----
   419→  const [kind, setKind] = useState<'price' | 'rsi' | 'macd'>('rsi')
   420→  const [symbol, setSymbol] = useState('')
   421→  const [tf, setTf] = useState('1D')
   422→  // price alarm fields
   423→  const [price, setPrice] = useState('')
   424→  const [dir, setDir] = useState<'above' | 'below'>('above')
   425→  // rsi fields
   426→  const [rsiPeriod, setRsiPeriod] = useState('14')
   427→  const [rsiLevel, setRsiLevel] = useState('70')
   428→  // macd fields
   429→  const [fast, setFast] = useState('12')
   430→  const [slow, setSlow] = useState('26')
   431→  const [signal, setSignal] = useState('9')
   432→  const [cond, setCond] = useState<'bull_cross' | 'bear_cross' | 'cross_above' | 'cross_below'>('bull_cross')
   433→  const [macdLevel, setMacdLevel] = useState('0')
   434→  const [creating, setCreating] = useState(false)
   435→  const symTouched = useRef(false)
   436→
   437→  // prefill symbol/timeframe from the active chart tab (once)
   438→  useEffect(() => {
   439→    if (symTouched.current) return
   440→    if (activeTab?.symbol) {
   441→      setSymbol(activeTab.symbol)
   442→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   443→    }
   444→  }, [activeTab?.symbol, activeTab?.timeframe])
   445→
   446→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   447→  const flipRsiLevel = (d: 'above' | 'below') => {
   448→    setDir(d)
   449→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   450→  }
   451→
   452→  const create = async () => {
   453→    const sym = symbol.trim().toUpperCase()
   454→    if (!sym) return
   455→    setCreating(true)
   456→    try {
   457→      const body: Record<string, unknown> =
   458→        kind === 'price'
   459→          ? {
   460→              symbol: sym, kind, price: parseFloat(price),
   461→              direction: dir, color: '#f59e0b',
   462→            }
   463→          : kind === 'rsi'
   464→            ? {
   465→                symbol: sym, kind, price: parseFloat(rsiLevel),
   466→                direction: dir, timeframe: tf,
   467→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   468→              }
   469→            : {
   470→                symbol: sym, kind, condition: cond, timeframe: tf,
   471→                price: parseFloat(macdLevel) || 0,
   472→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   473→                color: '#f59e0b',
   474→              }
   475→      const res = await fetch('/api/alerts', {
   476→        method: 'POST',
   477→        headers: { 'content-type': 'application/json' },
   478→        body: JSON.stringify(body),
   479→      })
   480→      const json = await res.json().catch(() => null)
   481→      if (res.ok && json?.alert) {
   482→        addAlertLocal(json.alert as Alert)
   483→      } else {
   484→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   485→      }
   486→    } catch { /* offline */ } finally {
   487→      setCreating(false)
   488→    }
   489→  }
   490→
   491→  const toggle = async (id: string, status: string) => {
   492→    const next = status === 'paused' ? 'active' : 'paused'
   493→    patchAlertLocal(id, { status: next })
   494→    await fetch(`/api/alerts/${id}`, {
   495→      method: 'PATCH',
   496→      headers: { 'content-type': 'application/json' },
   497→      body: JSON.stringify({ status: next }),
   498→    })
   499→  }
   500→  const remove = async (id: string) => {
   501→    removeAlertLocal(id)
   502→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   503→  }
   504→
   505→  const formValid =
   506→    kind === 'price'
   507→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   508→      : kind === 'rsi'
   509→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   510→        : cond === 'cross_above' || cond === 'cross_below'
   511→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   512→          : symbol.trim().length > 0
   513→
   514→  return (
   515→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   516→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   517→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   518→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   519→          <span className="flex items-center gap-2">
   520→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   521→          </span>
   522→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   523→            Close
   524→          </Button>
   525→        </div>
   526→
   527→        {/* ---- new alarm form ---- */}
   528→        <div className="border-b border-zinc-800 p-2.5">
   529→          <div className="mb-2 flex items-center gap-1.5">
   530→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   531→              <button
   532→                key={k}
   533→                onClick={() => setKind(k)}
   534→                aria-pressed={kind === k}
   535→                className={cn(
   536→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   537→                  kind === k
   538→                    ? k === 'rsi'
   539→                      ? 'bg-pink-500/15 text-pink-300'
   540→                      : k === 'macd'
   541→                        ? 'bg-amber-500/15 text-amber-300'
   542→                        : 'bg-emerald-500/15 text-emerald-300'
   543→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   544→                )}
   545→              >
   546→                {k}
   547→              </button>
   548→            ))}
   549→            <Input
   550→              value={symbol}
   551→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   552→              placeholder="SYMBOL"
   553→              aria-label="Alarm symbol"
   554→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   555→            />
   556→          </div>
   557→
   558→          {kind === 'price' && (
   559→            <div className="flex items-center gap-1.5">
   560→              <select
   561→                value={dir}
   562→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   563→                aria-label="Trigger direction"
   564→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   565→              >
   566→                <option value="above">crosses above</option>
   567→                <option value="below">falls below</option>
   568→              </select>
   569→              <Input
   570→                type="number" step="any" value={price}
   571→                onChange={(e) => setPrice(e.target.value)}
   572→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   573→                placeholder="Price" aria-label="Alert price"
   574→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   575→              />
   576→            </div>
   577→          )}
   578→
   579→          {kind === 'rsi' && (
   580→            <div className="space-y-1.5">
   581→              <div className="flex items-center gap-1.5">
   582→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>
   583→                <Input
   584→                  type="number" value={rsiPeriod} min={2} max={200}
   585→                  onChange={(e) => setRsiPeriod(e.target.value)}
   586→                  aria-label="RSI period"
   587→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   588→                />
   589→                <select
   590→                  value={dir}
   591→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   592→                  aria-label="RSI trigger direction"
   593→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   594→                >
   595→                  <option value="above">crosses above</option>
   596→                  <option value="below">crosses below</option>
   597→                </select>
   598→                <Input
   599→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   600→                  onChange={(e) => setRsiLevel(e.target.value)}
   601→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   602→                  aria-label="RSI level"
   603→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   604→                />
   605→              </div>
   606→              <TfRow tf={tf} setTf={setTf} />
   607→            </div>
   608→          )}
   609→
   610→          {kind === 'macd' && (
   611→            <div className="space-y-1.5">
   612→              <div className="flex items-center gap-1.5">
   613→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   614→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   615→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   616→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   617→                <select
   618→                  value={cond}
   619→                  onChange={(e) => setCond(e.target.value as typeof cond)}
   620→                  aria-label="MACD condition"
   621→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   622→                >
   623→                  <option value="bull_cross">bull crossover</option>
   624→                  <option value="bear_cross">bear crossover</option>
   625→                  <option value="cross_above">crosses above</option>
   626→                  <option value="cross_below">crosses below</option>
   627→                </select>
   628→              </div>
   629→              <div className="flex items-center gap-1.5">
   630→                {(cond === 'cross_above' || cond === 'cross_below') && (
   631→                  <Input
   632→                    type="number" step="any" value={macdLevel}
   633→                    onChange={(e) => setMacdLevel(e.target.value)}
   634→                    aria-label="MACD level"
   635→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   636→                  />
   637→                )}
   638→                <TfRow tf={tf} setTf={setTf} right />
   639→              </div>
   640→            </div>
   641→          )}
   642→
   643→          <Button
   644→            size="sm" disabled={!formValid || creating} onClick={() => void create()}
   645→            className="mt-2 h-7 w-full bg-amber-500 text-xs text-zinc-950 hover:bg-amber-400"
   646→          >
   647→            {creating ? <Loader2 className="h-3 w-3 animate-spin" /> : '🔔 Create alarm'}
   648→          </Button>
   649→        </div>
   650→
   651→        {/* ---- alarm list ---- */}
   652→        <div className="max-h-[45vh] overflow-y-auto p-2">
   653→          {alerts.length === 0 && (
   654→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   655→              No alarms yet — create one above or right-click the chart.
   656→            </p>
   657→          )}
   658→          {alerts.map((a) => (
   659→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   660→              <div className="flex min-w-0 items-center gap-2">
   661→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   662→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   663→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   664→                <Badge
   665→                  variant="outline"
   666→                  className={
   667→                    a.status === 'triggered'
   668→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   669→                      : a.status === 'paused'
   670→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   671→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   672→                  }
   673→                >
   674→                  {a.status}
   675→                </Badge>
   676→              </div>
   677→              <div className="flex shrink-0 items-center gap-1">
   678→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>
   679→                  {a.status === 'paused' ? 'Resume' : 'Pause'}
   680→                </Button>
   681→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   682→                  ✕
   683→                </Button>
   684→              </div>
   685→            </div>
   686→          ))}
   687→        </div>
   688→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   689→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   690→          <span className="ml-1 inline-flex items-center align-middle">
   691→            <Settings2 className="h-3 w-3" />
   692→          </span>
   693→        </div>
   694→      </div>
   695→    </div>
   696→  )
   697→}
