   342→// Ticking local clock for the footer (updates every second).
   343→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   344→function LiveClock() {
   345→  const [now, setNow] = useState<Date | null>(null)
   346→  useEffect(() => {
   347→    setNow(new Date())
   348→    const timer = setInterval(() => setNow(new Date()), 1000)
   349→    return () => clearInterval(timer)
   350→  }, [])
   351→  const tz = useMemo(() => {
   352→    try {
   353→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   354→    } catch {
   355→      return 'Local'
   356→    }
   357→  }, [])
   358→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   359→  const time = now
   360→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   361→    : '--:--:--'
   362→  return (
   363→    <span
   364→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   365→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   366→    >
   367→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   368→      <span>{time}</span>
   369→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   370→    </span>
   371→  )
   372→}

   408→        if (!res.ok || !alive) return
   409→        const json = await res.json()
   410→        if (alive) setMeta({ demo: Boolean(json.meta?.demo), reason: json.meta?.reason })
   411→      } catch { /* offline */ }

   414→    // Re-check right after keys are saved/changed and periodically, so the

   420→      window.removeEventListener('tt:settings-changed', check)
   421→      clearInterval(timer)

   442→const TF_LABEL: Record<string, string> = { '1m': '1m', '5m': '5m', '15m': '15m', '1h': '1H', '4h': '4H', '1D': '1D', '1W': '1W', '1M': '1M' }

   444→/** One-line human summary of an alarm (any kind) for the list. */
   445→function alertSummary(a: Alert): string {
   446→  if (a.kind === 'rsi') {
   447→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   448→  }
   449→  if (a.kind === 'macd') {
   450→    const p = a.params ?? {}
   451→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   452→    const cond =
   453→      a.condition === 'bull_cross' ? 'bullish crossover'
   454→      : a.condition === 'bear_cross' ? 'bearish crossover'
   455→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   456→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   457→      : 'crossover'
   458→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   459→  }
   460→  if (a.kind === 'trend') {
   461→    // the level MOVES with the bound trend line — show the crossing direction
   462→    // plus the level the line had when the alarm was created
   463→    return `Trend line · crosses ${a.direction === 'above' ? 'up' : 'down'} · level was ${fmtPrice(a.price)}`
   464→  }
   465→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   466→}

   495→  useEffect(() => {
   496→    if (symTouched.current) return
   497→    if (activeTab?.symbol) {
   498→      setSymbol(activeTab.symbol)
   499→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   500→    }
   501→  }, [activeTab?.symbol, activeTab?.timeframe])
   502→
   503→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   504→  const flipRsiLevel = (d: 'above' | 'below') => {
   505→    setDir(d)
   506→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   507→  }
   508→
   509→  const create = async () => {
   510→    const sym = symbol.trim().toUpperCase()
   511→    if (!sym) return
   512→    setCreating(true)
   513→    try {
   514→      const body: Record<string, unknown> =
   515→        kind === 'price'
   516→          ? {
   517→              symbol: sym, kind, price: parseFloat(price),
   518→              direction: dir, color: '#f59e0b',
   519→            }
   520→          : kind === 'rsi'
   521→            ? {
   522→                symbol: sym, kind, price: parseFloat(rsiLevel),
   523→                direction: dir, timeframe: tf,
   524→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   525→              }
   526→            : {
   527→                symbol: sym, kind, condition: cond, timeframe: tf,
   528→                price: parseFloat(macdLevel) || 0,
   529→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   530→                color: '#f59e0b',
   531→              }
   532→      const res = await fetch('/api/alerts', {
   533→        method: 'POST',
   534→        headers: { 'content-type': 'application/json' },
   535→        body: JSON.stringify(body),
   536→      })
   537→      const json = await res.json().catch(() => null)
   538→      if (res.ok && json?.alert) {
   539→        addAlertLocal(json.alert as Alert)
   540→      } else {
   541→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   542→      }
   543→    } catch { /* offline */ } finally {
   544→      setCreating(false)
   545→    }
   546→  }
   547→
   548→  const toggle = async (id: string, status: string) => {
   549→    const next = status === 'paused' ? 'active' : 'paused'
   550→    patchAlertLocal(id, { status: next })
   551→    await fetch(`/api/alerts/${id}`, {
   552→      method: 'PATCH',
   553→      headers: { 'content-type': 'application/json' },
   554→      body: JSON.stringify({ status: next }),
   555→    })
   556→  }
   557→  const remove = async (id: string) => {
   558→    removeAlertLocal(id)
   559→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   560→  }
   561→
   562→  const formValid =
   563→    kind === 'price'
   564→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   565→      : kind === 'rsi'
   566→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   567→        : cond === 'cross_above' || cond === 'cross_below'
   568→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   569→          : symbol.trim().length > 0
   570→
   571→  return (
   572→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   573→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   574→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   575→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   576→          <span className="flex items-center gap-2">
   577→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   578→          </span>
   579→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   580→            Close
   581→          </Button>
   582→        </div>
   583→
   584→        {/* ---- new alarm form ---- */}
   585→        <div className="border-b border-zinc-800 p-2.5">
   586→          <div className="mb-2 flex items-center gap-1.5">
   587→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   588→              <button
   589→                key={k}
   590→                onClick={() => setKind(k)}
   591→                aria-pressed={kind === k}
   592→                className={cn(
   593→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   594→                  kind === k
   595→                    ? k === 'rsi'
   596→                      ? 'bg-pink-500/15 text-pink-300'
   597→                      : k === 'macd'
   598→                        ? 'bg-amber-500/15 text-amber-300'
   599→                        : 'bg-emerald-500/15 text-emerald-300'
   600→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   601→                )}
   602→              >
   603→                {k}
   604→              </button>
   605→            ))}
   606→            <Input
   607→              value={symbol}
   608→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   609→              placeholder="SYMBOL"
   610→              aria-label="Alarm symbol"
   611→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   612→            />
   613→          </div>
   614→
   615→          {kind === 'price' && (
   616→            <div className="flex items-center gap-1.5">
   617→              <select
   618→                value={dir}
   619→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   620→                aria-label="Trigger direction"
   621→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   622→              >
   623→                <option value="above">crosses above</option>
   624→                <option value="below">falls below</option>
   625→              </select>
   626→              <Input
   627→                type="number" step="any" value={price}
   628→                onChange={(e) => setPrice(e.target.value)}
   629→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   630→                placeholder="Price" aria-label="Alert price"
   631→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   632→              />
   633→            </div>
   634→          )}
   635→
   636→          {kind === 'rsi' && (
   637→            <div className="space-y-1.5">
   638→              <div className="flex items-center gap-1.5">
   639→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>
   640→                <Input
   641→                  type="number" value={rsiPeriod} min={2} max={200}
   642→                  onChange={(e) => setRsiPeriod(e.target.value)}
   643→                  aria-label="RSI period"
   644→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   645→                />
   646→                <select
   647→                  value={dir}
   648→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   649→                  aria-label="RSI trigger direction"
   650→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   651→                >
   652→                  <option value="above">crosses above</option>
   653→                  <option value="below">crosses below</option>
   654→                </select>
   655→                <Input
   656→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   657→                  onChange={(e) => setRsiLevel(e.target.value)}
   658→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   659→                  aria-label="RSI level"
   660→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   661→                />
   662→              </div>
   663→              <TfRow tf={tf} setTf={setTf} />
   664→            </div>
   665→          )}
   666→
   667→          {kind === 'macd' && (
   668→            <div className="space-y-1.5">
   669→              <div className="flex items-center gap-1.5">
   670→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   671→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   672→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   673→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   674→                <select
   675→                  value={cond}
   676→                  onChange={(e) => setCond(e.target.value as typeof cond)}
   677→                  aria-label="MACD condition"
   678→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   679→                >
   680→                  <option value="bull_cross">bull crossover</option>
   681→                  <option value="bear_cross">bear crossover</option>
   682→                  <option value="cross_above">crosses above</option>
   683→                  <option value="cross_below">crosses below</option>
   684→                </select>
   685→              </div>
   686→              <div className="flex items-center gap-1.5">
   687→                {(cond === 'cross_above' || cond === 'cross_below') && (
   688→                  <Input
   689→                    type="number" step="any" value={macdLevel}
   690→                    onChange={(e) => setMacdLevel(e.target.value)}
   691→                    aria-label="MACD level"
   692→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   693→                  />
   694→                )}
   695→                <TfRow tf={tf} setTf={setTf} right />
   696→              </div>
   697→            </div>
   698→          )}
   699→
   700→          <Button
   701→            size="sm" disabled={!formValid || creating} onClick={() => void create()}
   702→            className="mt-2 h-7 w-full bg-amber-500 text-xs text-zinc-950 hover:bg-amber-400"
   703→          >
   704→            {creating ? <Loader2 className="h-3 w-3 animate-spin" /> : '🔔 Create alarm'}
   705→          </Button>
   706→        </div>
   707→
   708→        {/* ---- alarm list ---- */}
   709→        <div className="max-h-[45vh] overflow-y-auto p-2">
   710→          {alerts.length === 0 && (
   711→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   712→              No alarms yet — create one above or right-click the chart.
   713→            </p>
   714→          )}
   715→          {alerts.map((a) => (
   716→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   717→              <div className="flex min-w-0 items-center gap-2">
   718→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   719→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   720→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   721→                <Badge
   722→                  variant="outline"
   723→                  className={
   724→                    a.status === 'triggered'
   725→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   726→                      : a.status === 'paused'
   727→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   728→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   729→                  }
   730→                >
   731→                  {a.status}
   732→                </Badge>
   733→              </div>
   734→              <div className="flex shrink-0 items-center gap-1">
   735→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>
   736→                  {a.status === 'paused' ? 'Resume' : 'Pause'}
   737→                </Button>
   738→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   739→                  ✕
   740→                </Button>
   741→              </div>
   742→            </div>
   743→          ))}
   744→        </div>
   745→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   746→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   747→          <span className="ml-1 inline-flex items-center align-middle">
   748→            <Settings2 className="h-3 w-3" />
   749→          </span>
   750→        </div>
   751→      </div>
   752→    </div>
   753→  )
   754→}

   756→/** Timeframe selector row for indicator alarms. */
   757→function TfRow({ tf, setTf, right }: { tf: string; setTf: (v: string) => void; right?: boolean }) {

   760→      <label className="text-[10px] uppercase tracking-wider text-zinc-500">TF</label>
   761→      <div className="flex overflow-hidden rounded border border-zinc-700" role="radiogroup" aria-label="Timeframe">
   762→        {ALERT_TFS.map((t) => (
   763→          <button
   764→            key={t}
   765→            onClick={() => setTf(t)}
   766→            aria-pressed={tf === t}
   767→            className={cn(
   768→              'px-1.5 py-0.5 text-[10px] font-medium transition-colors',
   769→              tf === t ? 'bg-zinc-700 text-zinc-100' : 'bg-zinc-950 text-zinc-500 hover:text-zinc-300',
   770→            )}
   771→          >
   772→            {TF_LABEL[t]}
   773→          </button>
   774→        ))}
   775→      </div>
   776→    </div>
   777→  )
