   302→   302→// Ticking local clock for the footer (updates every second).
   303→   303→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   304→   304→function LiveClock() {
   305→   305→  const [now, setNow] = useState<Date | null>(null)
   306→   306→  useEffect(() => {
   307→   307→    setNow(new Date())
   308→   308→    const timer = setInterval(() => setNow(new Date()), 1000)
   309→   309→    return () => clearInterval(timer)
   310→   310→  }, [])
   311→   311→  const tz = useMemo(() => {
   312→   312→    try {
   313→   313→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   314→   314→    } catch {
   315→   315→      return 'Local'
   316→   316→    }
   317→   317→  }, [])
   318→   318→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   319→   319→  const time = now
   320→   320→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   321→   321→    : '--:--:--'
   322→   322→  return (
   323→   323→    <span
   324→   324→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   325→   325→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   326→   326→    >
   327→   327→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   328→   328→      <span>{time}</span>
   329→   329→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   330→   330→    </span>
   331→   331→  )
   332→   332→}

   402→   402→/** One-line human summary of an alarm (any kind) for the list. */
   403→   403→function alertSummary(a: Alert): string {
   404→   404→  if (a.kind === 'rsi') {
   405→   405→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   406→   406→  }
   407→   407→  if (a.kind === 'macd') {
   408→   408→    const p = a.params ?? {}
   409→   409→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   410→   410→    const cond =
   411→   411→      a.condition === 'bull_cross' ? 'bullish crossover'
   412→   412→      : a.condition === 'bear_cross' ? 'bearish crossover'
   413→   413→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   414→   414→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   415→   415→      : 'crossover'
   416→   416→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   417→   417→  }
   418→   418→  if (a.kind === 'trend') {
   419→   419→    // the level MOVES with the bound trend line — show the crossing direction
   420→   420→    // plus the level the line had when the alarm was created
   421→   421→    return `Trend line · crosses ${a.direction === 'above' ? 'up' : 'down'} · level was ${fmtPrice(a.price)}`
   422→   422→  }
   423→   423→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   424→   424→}

   427→   427→  const alerts = useTerminal((s) => s.alerts)
   428→   428→  const addAlertLocal = useTerminal((s) => s.addAlertLocal)
   429→   429→  const patchAlertLocal = useTerminal((s) => s.patchAlertLocal)
   430→   430→  const removeAlertLocal = useTerminal((s) => s.removeAlertLocal)
   431→   431→  const activeTab = useTerminal((s) => s.tabs.find((t) => t.id === s.activeTabId) ?? s.tabs[0])
   432→   432→
   433→   433→  // ---- creation form state ----
   434→   434→  const [kind, setKind] = useState<'price' | 'rsi' | 'macd'>('rsi')
   435→   435→  const [symbol, setSymbol] = useState('')
   436→   436→  const [tf, setTf] = useState('1D')
   437→   437→  // price alarm fields
   438→   438→  const [price, setPrice] = useState('')
   439→   439→  const [dir, setDir] = useState<'above' | 'below'>('above')
   440→   440→  // rsi fields
   441→   441→  const [rsiPeriod, setRsiPeriod] = useState('14')
   442→   442→  const [rsiLevel, setRsiLevel] = useState('70')
   443→   443→  // macd fields
   444→   444→  const [fast, setFast] = useState('12')
   445→   445→  const [slow, setSlow] = useState('26')
   446→   446→  const [signal, setSignal] = useState('9')
   447→   447→  const [cond, setCond] = useState<'bull_cross' | 'bear_cross' | 'cross_above' | 'cross_below'>('bull_cross')
   448→   448→  const [macdLevel, setMacdLevel] = useState('0')
   449→   449→  const [creating, setCreating] = useState(false)
   450→   450→  const symTouched = useRef(false)
   451→   451→
   452→   452→  // prefill symbol/timeframe from the active chart tab (once)
   453→   453→  useEffect(() => {
   454→   454→    if (symTouched.current) return
   455→   455→    if (activeTab?.symbol) {
   456→   456→      setSymbol(activeTab.symbol)
   457→   457→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   458→   458→    }
   459→   459→  }, [activeTab?.symbol, activeTab?.timeframe])
   460→   460→
   461→   461→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   462→   462→  const flipRsiLevel = (d: 'above' | 'below') => {
   463→   463→    setDir(d)
   464→   464→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   465→   465→  }
   466→   466→
   467→   467→  const create = async () => {
   468→   468→    const sym = symbol.trim().toUpperCase()
   469→   469→    if (!sym) return
   470→   470→    setCreating(true)
   471→   471→    try {
   472→   472→      const body: Record<string, unknown> =
   473→   473→        kind === 'price'
   474→   474→          ? {
   475→   475→              symbol: sym, kind, price: parseFloat(price),
   476→   476→              direction: dir, color: '#f59e0b',
   477→   477→            }
   478→   478→          : kind === 'rsi'
   479→   479→            ? {
   480→   480→                symbol: sym, kind, price: parseFloat(rsiLevel),
   481→   481→                direction: dir, timeframe: tf,
   482→   482→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   483→   483→              }
   484→   484→            : {
   485→   485→                symbol: sym, kind, condition: cond, timeframe: tf,
   486→   486→                price: parseFloat(macdLevel) || 0,
   487→   487→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   488→   488→                color: '#f59e0b',
   489→   489→              }
   490→   490→      const res = await fetch('/api/alerts', {
   491→   491→        method: 'POST',
   492→   492→        headers: { 'content-type': 'application/json' },
   493→   493→        body: JSON.stringify(body),
   494→   494→      })
   495→   495→      const json = await res.json().catch(() => null)
   496→   496→      if (res.ok && json?.alert) {
   497→   497→        addAlertLocal(json.alert as Alert)
   498→   498→      } else {
   499→   499→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   500→   500→      }
   501→   501→    } catch { /* offline */ } finally {
   502→   502→      setCreating(false)
   503→   503→    }
   504→   504→  }
   505→   505→
   506→   506→  const toggle = async (id: string, status: string) => {
   507→   507→    const next = status === 'paused' ? 'active' : 'paused'
   508→   508→    patchAlertLocal(id, { status: next })
   509→   509→    await fetch(`/api/alerts/${id}`, {
   510→   510→      method: 'PATCH',
   511→   511→      headers: { 'content-type': 'application/json' },
   512→   512→      body: JSON.stringify({ status: next }),
   513→   513→    })
   514→   514→  }
   515→   515→  const remove = async (id: string) => {
   516→   516→    removeAlertLocal(id)
   517→   517→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   518→   518→  }
   519→   519→
   520→   520→  const formValid =
   521→   521→    kind === 'price'
   522→   522→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   523→   523→      : kind === 'rsi'
   524→   524→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   525→   525→        : cond === 'cross_above' || cond === 'cross_below'
   526→   526→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   527→   527→          : symbol.trim().length > 0
   528→   528→
   529→   529→  return (
   530→   530→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   531→   531→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   532→   532→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   533→   533→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   534→   534→          <span className="flex items-center gap-2">
   535→   535→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   536→   536→          </span>
   537→   537→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   538→   538→            Close
   539→   539→          </Button>
   540→   540→        </div>
   541→   541→
   542→   542→        {/* ---- new alarm form ---- */}
   543→   543→        <div className="border-b border-zinc-800 p-2.5">
   544→   544→          <div className="mb-2 flex items-center gap-1.5">
   545→   545→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   546→   546→              <button
   547→   547→                key={k}
   548→   548→                onClick={() => setKind(k)}
   549→   549→                aria-pressed={kind === k}
   550→   550→                className={cn(
   551→   551→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   552→   552→                  kind === k
   553→   553→                    ? k === 'rsi'
   554→   554→                      ? 'bg-pink-500/15 text-pink-300'
   555→   555→                      : k === 'macd'
   556→   556→                        ? 'bg-amber-500/15 text-amber-300'
   557→   557→                        : 'bg-emerald-500/15 text-emerald-300'
   558→   558→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   559→   559→                )}
   560→   560→              >
   561→   561→                {k}
   562→   562→              </button>
   563→   563→            ))}
   564→   564→            <Input
   565→   565→              value={symbol}
   566→   566→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   567→   567→              placeholder="SYMBOL"
   568→   568→              aria-label="Alarm symbol"
   569→   569→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   570→   570→            />
   571→   571→          </div>
   572→   572→
   573→   573→          {kind === 'price' && (
   574→   574→            <div className="flex items-center gap-1.5">
   575→   575→              <select
   576→   576→                value={dir}
   577→   577→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   578→   578→                aria-label="Trigger direction"
   579→   579→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   580→   580→              >
   581→   581→                <option value="above">crosses above</option>
   582→   582→                <option value="below">falls below</option>
   583→   583→              </select>
   584→   584→              <Input
   585→   585→                type="number" step="any" value={price}
   586→   586→                onChange={(e) => setPrice(e.target.value)}
   587→   587→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   588→   588→                placeholder="Price" aria-label="Alert price"
   589→   589→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   590→   590→              />
   591→   591→            </div>
   592→   592→          )}
   593→   593→
   594→   594→          {kind === 'rsi' && (
   595→   595→            <div className="space-y-1.5">
   596→   596→              <div className="flex items-center gap-1.5">
   597→   597→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>

   599→   599→                  type="number" value={rsiPeriod} min={2} max={200}
   600→   600→                  onChange={(e) => setRsiPeriod(e.target.value)}
   601→   601→                  aria-label="RSI period"
   602→   602→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   603→   603→                />
   604→   604→                <select
   605→   605→                  value={dir}
   606→   606→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   607→   607→                  aria-label="RSI trigger direction"
   608→   608→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   609→   609→                >
   610→   610→                  <option value="above">crosses above</option>
   611→   611→                  <option value="below">crosses below</option>
   612→   612→                </select>
   613→   613→                <Input
   614→   614→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   615→   615→                  onChange={(e) => setRsiLevel(e.target.value)}
   616→   616→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   617→   617→                  aria-label="RSI level"
   618→   618→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   619→   619→                />
   620→   620→              </div>
   621→   621→              <TfRow tf={tf} setTf={setTf} />

   623→   623→          )}
   624→   624→

   637→   637→                >
   638→   638→                  <option value="bull_cross">bull crossover</option>

   640→   640→                  <option value="cross_above">crosses above</option>
   641→   641→                  <option value="cross_below">crosses below</option>
   642→   642→                </select>
   643→   643→              </div>
   644→   644→              <div className="flex items-center gap-1.5">
   645→   645→                {(cond === 'cross_above' || cond === 'cross_below') && (
   646→   646→                  <Input
   647→   647→                    type="number" step="any" value={macdLevel}
   648→   648→                    onChange={(e) => setMacdLevel(e.target.value)}
   649→   649→                    aria-label="MACD level"
   650→   650→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   651→   651→                  />
   652→   652→                )}
   653→   653→                <TfRow tf={tf} setTf={setTf} right />
   654→   654→              </div>
   655→   655→            </div>

   663→   663→          </Button>
   664→   664→        </div>
   665→   665→
   666→   666→        {/* ---- alarm list ---- */}

   669→   669→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   670→   670→              No alarms yet — create one above or right-click the chart.
   671→   671→            </p>
   672→   672→          )}
   673→   673→          {alerts.map((a) => (
   674→   674→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   675→   675→              <div className="flex min-w-0 items-center gap-2">
   676→   676→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   677→   677→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   678→   678→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   679→   679→                <Badge
   680→   680→                  variant="outline"
   681→   681→                  className={
   682→   682→                    a.status === 'triggered'
   683→   683→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   684→   684→                      : a.status === 'paused'
   685→   685→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   686→   686→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   687→   687→                  }
   688→   688→                >
   689→   689→                  {a.status}

   691→   691→              </div>
   692→   692→              <div className="flex shrink-0 items-center gap-1">
   693→   693→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>

   695→   695→                </Button>
   696→   696→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   697→   697→                  ✕
   698→   698→                </Button>
   699→   699→              </div>
   700→   700→            </div>
   701→   701→          ))}
   702→   702→        </div>
   703→   703→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   704→   704→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   705→   705→          <span className="ml-1 inline-flex items-center align-middle">
   706→   706→            <Settings2 className="h-3 w-3" />
   707→   707→          </span>
   708→   708→        </div>
   709→   709→      </div>
   710→   710→    </div>
   711→   711→  )
   712→   712→}
