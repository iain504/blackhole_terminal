   303→// Ticking local clock for the footer (updates every second).
   304→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   305→function LiveClock() {
   306→  const [now, setNow] = useState<Date | null>(null)
   307→  useEffect(() => {
   308→    setNow(new Date())
   309→    const timer = setInterval(() => setNow(new Date()), 1000)
   310→    return () => clearInterval(timer)
   311→  }, [])
   312→  const tz = useMemo(() => {
   313→    try {
   314→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   315→    } catch {
   316→      return 'Local'
   317→    }
   318→  }, [])
   319→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   320→  const time = now
   321→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   322→    : '--:--:--'
   323→  return (
   324→    <span
   325→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   326→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   327→    >
   328→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   329→      <span>{time}</span>
   330→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   331→    </span>
   332→  )
   333→}

   358→function useDemoMeta(): { demo: boolean; reason?: string } {
   359→  const [meta, setMeta] = useState<{ demo: boolean; reason?: string }>({ demo: false })
   360→  useEffect(() => {
   361→    let alive = true
   362→    const check = async () => {

   369→        if (!res.ok || !alive) return
   370→        const json = await res.json()
   371→        if (alive) setMeta({ demo: Boolean(json.meta?.demo), reason: json.meta?.reason })
   372→      } catch { /* offline */ }

   375→    // Re-check right after keys are saved/changed and periodically, so the

   377→    window.addEventListener('tt:settings-changed', check)

   404→/** One-line human summary of an alarm (any kind) for the list. */
   405→function alertSummary(a: Alert): string {
   406→  if (a.kind === 'rsi') {
   407→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   408→  }
   409→  if (a.kind === 'macd') {
   410→    const p = a.params ?? {}
   411→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   412→    const cond =
   413→      a.condition === 'bull_cross' ? 'bullish crossover'
   414→      : a.condition === 'bear_cross' ? 'bearish crossover'
   415→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   416→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   417→      : 'crossover'
   418→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   419→  }
   420→  if (a.kind === 'trend') {
   421→    // the level MOVES with the bound trend line — show the crossing direction
   422→    // plus the level the line had when the alarm was created
   423→    return `Trend line · crosses ${a.direction === 'above' ? 'up' : 'down'} · level was ${fmtPrice(a.price)}`
   424→  }
   425→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   426→}

   435→  // ---- creation form state ----
   436→  const [kind, setKind] = useState<'price' | 'rsi' | 'macd'>('rsi')
   437→  const [symbol, setSymbol] = useState('')

   444→  const [rsiLevel, setRsiLevel] = useState('70')

   448→  const [signal, setSignal] = useState('9')
   449→  const [cond, setCond] = useState<'bull_cross' | 'bear_cross' | 'cross_above' | 'cross_below'>('bull_cross')
   450→  const [macdLevel, setMacdLevel] = useState('0')
   451→  const [creating, setCreating] = useState(false)
   452→  const symTouched = useRef(false)
   453→
   454→  // prefill symbol/timeframe from the active chart tab (once)
   455→  useEffect(() => {
   456→    if (symTouched.current) return
   457→    if (activeTab?.symbol) {
   458→      setSymbol(activeTab.symbol)
   459→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   460→    }
   461→  }, [activeTab?.symbol, activeTab?.timeframe])

   463→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   464→  const flipRsiLevel = (d: 'above' | 'below') => {
   465→    setDir(d)
   466→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   467→  }
   468→
   469→  const create = async () => {
   470→    const sym = symbol.trim().toUpperCase()
   471→    if (!sym) return
   472→    setCreating(true)
   473→    try {
   474→      const body: Record<string, unknown> =
   475→        kind === 'price'
   476→          ? {
   477→              symbol: sym, kind, price: parseFloat(price),
   478→              direction: dir, color: '#f59e0b',
   479→            }
   480→          : kind === 'rsi'
   481→            ? {
   482→                symbol: sym, kind, price: parseFloat(rsiLevel),
   483→                direction: dir, timeframe: tf,
   484→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   485→              }
   486→            : {
   487→                symbol: sym, kind, condition: cond, timeframe: tf,
   488→                price: parseFloat(macdLevel) || 0,
   489→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   490→                color: '#f59e0b',
   491→              }
   492→      const res = await fetch('/api/alerts', {
   493→        method: 'POST',
   494→        headers: { 'content-type': 'application/json' },
   495→        body: JSON.stringify(body),
   496→      })
   497→      const json = await res.json().catch(() => null)
   498→      if (res.ok && json?.alert) {
   499→        addAlertLocal(json.alert as Alert)
   500→      } else {
   501→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   502→      }
   503→    } catch { /* offline */ } finally {
   504→      setCreating(false)
   505→    }
   506→  }
   507→
   508→  const toggle = async (id: string, status: string) => {
   509→    const next = status === 'paused' ? 'active' : 'paused'
   510→    patchAlertLocal(id, { status: next })
   511→    await fetch(`/api/alerts/${id}`, {
   512→      method: 'PATCH',
   513→      headers: { 'content-type': 'application/json' },
   514→      body: JSON.stringify({ status: next }),
   515→    })
   516→  }
   517→  const remove = async (id: string) => {
   518→    removeAlertLocal(id)
   519→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   520→  }
   521→
   522→  const formValid =
   523→    kind === 'price'
   524→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   525→      : kind === 'rsi'
   526→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   527→        : cond === 'cross_above' || cond === 'cross_below'
   528→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   529→          : symbol.trim().length > 0
   530→
   531→  return (
   532→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   533→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   534→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   535→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   536→          <span className="flex items-center gap-2">
   537→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   538→          </span>
   539→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   540→            Close
   541→          </Button>
   542→        </div>
   543→
   544→        {/* ---- new alarm form ---- */}
   545→        <div className="border-b border-zinc-800 p-2.5">
   546→          <div className="mb-2 flex items-center gap-1.5">
   547→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   548→              <button
   549→                key={k}
   550→                onClick={() => setKind(k)}
   551→                aria-pressed={kind === k}
   552→                className={cn(
   553→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   554→                  kind === k
   555→                    ? k === 'rsi'
   556→                      ? 'bg-pink-500/15 text-pink-300'
   557→                      : k === 'macd'
   558→                        ? 'bg-amber-500/15 text-amber-300'
   559→                        : 'bg-emerald-500/15 text-emerald-300'
   560→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   561→                )}
   562→              >
   563→                {k}
   564→              </button>
   565→            ))}
   566→            <Input
   567→              value={symbol}
   568→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   569→              placeholder="SYMBOL"
   570→              aria-label="Alarm symbol"
   571→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   572→            />
   573→          </div>
   574→
   575→          {kind === 'price' && (
   576→            <div className="flex items-center gap-1.5">
   577→              <select
   578→                value={dir}
   579→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   580→                aria-label="Trigger direction"
   581→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   582→              >
   583→                <option value="above">crosses above</option>
   584→                <option value="below">falls below</option>
   585→              </select>
   586→              <Input
   587→                type="number" step="any" value={price}
   588→                onChange={(e) => setPrice(e.target.value)}
   589→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   590→                placeholder="Price" aria-label="Alert price"
   591→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   592→              />
   593→            </div>
   594→          )}
   595→
   596→          {kind === 'rsi' && (
   597→            <div className="space-y-1.5">
   598→              <div className="flex items-center gap-1.5">
   599→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>
   600→                <Input
   601→                  type="number" value={rsiPeriod} min={2} max={200}
   602→                  onChange={(e) => setRsiPeriod(e.target.value)}
   603→                  aria-label="RSI period"
   604→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   605→                />
   606→                <select
   607→                  value={dir}
   608→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   609→                  aria-label="RSI trigger direction"
   610→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   611→                >
   612→                  <option value="above">crosses above</option>
   613→                  <option value="below">crosses below</option>
   614→                </select>
   615→                <Input
   616→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   617→                  onChange={(e) => setRsiLevel(e.target.value)}
   618→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   619→                  aria-label="RSI level"
   620→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   621→                />
   622→              </div>
   623→              <TfRow tf={tf} setTf={setTf} />
   624→            </div>
   625→          )}
   626→
   627→          {kind === 'macd' && (
   628→            <div className="space-y-1.5">
   629→              <div className="flex items-center gap-1.5">
   630→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   631→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   632→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   633→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   634→                <select
   635→                  value={cond}
   636→                  onChange={(e) => setCond(e.target.value as typeof cond)}
   637→                  aria-label="MACD condition"
   638→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   639→                >
   640→                  <option value="bull_cross">bull crossover</option>
   641→                  <option value="bear_cross">bear crossover</option>
   642→                  <option value="cross_above">crosses above</option>
   643→                  <option value="cross_below">crosses below</option>
   644→                </select>
   645→              </div>
   646→              <div className="flex items-center gap-1.5">
   647→                {(cond === 'cross_above' || cond === 'cross_below') && (
   648→                  <Input
   649→                    type="number" step="any" value={macdLevel}
   650→                    onChange={(e) => setMacdLevel(e.target.value)}
   651→                    aria-label="MACD level"
   652→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   653→                  />
   654→                )}
   655→                <TfRow tf={tf} setTf={setTf} right />
   656→              </div>
   657→            </div>
   658→          )}
   659→
   660→          <Button
   661→            size="sm" disabled={!formValid || creating} onClick={() => void create()}
   662→            className="mt-2 h-7 w-full bg-amber-500 text-xs text-zinc-950 hover:bg-amber-400"
   663→          >
   664→            {creating ? <Loader2 className="h-3 w-3 animate-spin" /> : '🔔 Create alarm'}
   665→          </Button>
   666→        </div>
   667→
   668→        {/* ---- alarm list ---- */}
   669→        <div className="max-h-[45vh] overflow-y-auto p-2">
   670→          {alerts.length === 0 && (
   671→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   672→              No alarms yet — create one above or right-click the chart.
   673→            </p>
   674→          )}
   675→          {alerts.map((a) => (
   676→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   677→              <div className="flex min-w-0 items-center gap-2">
   678→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   679→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   680→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   681→                <Badge
   682→                  variant="outline"
   683→                  className={
   684→                    a.status === 'triggered'
   685→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   686→                      : a.status === 'paused'
   687→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   688→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   689→                  }
   690→                >
   691→                  {a.status}
   692→                </Badge>
   693→              </div>
   694→              <div className="flex shrink-0 items-center gap-1">
   695→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>
   696→                  {a.status === 'paused' ? 'Resume' : 'Pause'}
   697→                </Button>
   698→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   699→                  ✕
   700→                </Button>
   701→              </div>
   702→            </div>
   703→          ))}
   704→        </div>
   705→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   706→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   707→          <span className="ml-1 inline-flex items-center align-middle">
   708→            <Settings2 className="h-3 w-3" />
   709→          </span>
   710→        </div>
   711→      </div>
   712→    </div>
   713→  )
   714→}
