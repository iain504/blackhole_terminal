     1→'use client'
     2→
     3→// Positions console tab: account strip (balance, buying power, daily P&L,
     4→// equity curve Hour..Year, full account details) + every open paper + live
     5→// position in one table — symbol, market, side/qty, entry price, TP, SL,
     6→// trailing, P&L — with a per-row close button (market-close via the Alpaca
     7→// Trading API). Below it: every PENDING (unfilled) entry order with its
     8→// planned TP/SL — click any price to edit it (same PATCH the chart drags
     9→// use) — and a cancel action. Missing TP/SL cells offer a [+ TP]/[+ SL]
    10→// button that attaches the exit to the position (drag-to-price also exists
    11→// on the chart).
    12→import { useCallback, useEffect, useState } from 'react'
    13→import { History, Loader2, RefreshCw, X } from 'lucide-react'
    14→import { Button } from '@/components/ui/button'
    15→import { Badge } from '@/components/ui/badge'
    16→import { Input } from '@/components/ui/input'
    17→import { useToast } from '@/hooks/use-toast'
    18→import { fmtPrice, fmtHeldDuration, ClosedTrade } from '@/lib/market-types'
    19→import { ExitInfo, PendingOrderInfo, PositionRow, normSym } from '@/components/chart/trading-types'
    20→import AccountOverview from './AccountOverview'
    21→import { cn } from '@/lib/utils'
    22→
    23→type Mode = 'paper' | 'live'
    24→
    25→interface ModeData {
    26→  mode: Mode
    27→  positions: PositionRow[]
    28→  exits: Record<string, { sl?: ExitInfo; tp?: ExitInfo }>
    29→  pending: PendingOrderInfo[]
    30→  error?: string
    31→  loaded: boolean
    32→}
    33→
    34→interface HistoryData {
    35→  trades: ClosedTrade[]
    36→  configured: boolean
    37→  loaded: boolean
    38→  error?: string
    39→}
    40→
    41→interface PosConsoleRow {
    42→  mode: Mode
    43→  pos: PositionRow
    44→  sl?: ExitInfo
    45→  tp?: ExitInfo
    46→}
    47→
    48→const MODES: Mode[] = ['paper', 'live']
    49→
    50→export default function PositionsTab() {
    51→  const { toast } = useToast()
    52→  const [data, setData] = useState<Record<Mode, ModeData>>({
    53→    paper: { mode: 'paper', positions: [], exits: {}, pending: [], loaded: false },
    54→    live: { mode: 'live', positions: [], exits: {}, pending: [], loaded: false },
    55→  })
    56→  const [history, setHistory] = useState<Record<Mode, HistoryData>>({
    57→    paper: { trades: [], configured: true, loaded: false },
    58→    live: { trades: [], configured: true, loaded: false },
    59→  })
    60→  const [refreshing, setRefreshing] = useState(false)
    61→  const [closing, setClosing] = useState<string | null>(null) // `${mode}:${symbol}`
    62→  const [canceling, setCanceling] = useState<string | null>(null)
    63→
    64→  const refresh = useCallback(async () => {
    65→    setRefreshing(true)
    66→    const one = async (mode: Mode): Promise<ModeData> => {
    67→      try {
    68→        const res = await fetch(`/api/trading/positions?mode=${mode}`)
    69→        if (!res.ok) {
    70→          return { mode, positions: [], exits: {}, pending: [], error: `HTTP ${res.status}`, loaded: true }
    71→        }
    72→        const json = await res.json()
    73→        const errs: string[] | undefined = json.errors
    74→        return {
    75→          mode,
    76→          positions: (json.positions || []) as PositionRow[],
    77→          exits: (json.exits || {}) as Record<string, { sl?: ExitInfo; tp?: ExitInfo }>,
    78→          pending: (json.pending || []) as PendingOrderInfo[],
    79→          error: errs?.length ? errs.join(', ') : undefined,
    80→          loaded: true,
    81→        }
    82→      } catch {
    83→        return { mode, positions: [], exits: {}, pending: [], error: 'network error', loaded: true }
    84→      }
    85→    }
    86→    const hist = async (mode: Mode): Promise<HistoryData> => {
    87→      try {
    88→        const res = await fetch(`/api/trading/history?mode=${mode}&limit=150`)
    89→        if (!res.ok) return { trades: [], configured: true, loaded: true, error: `HTTP ${res.status}` }
    90→        const json = await res.json()
    91→        return { trades: (json.trades || []) as ClosedTrade[], configured: json.configured !== false, loaded: true }
    92→      } catch {
    93→        return { trades: [], configured: true, loaded: true, error: 'network error' }
    94→      }
    95→    }
    96→    const [paper, live] = await Promise.all(MODES.map(one))
    97→    const [hPaper, hLive] = await Promise.all(MODES.map(hist))
    98→    setData({ paper, live })
    99→    setHistory({ paper: hPaper, live: hLive })
   100→    setRefreshing(false)
   101→  }, [])
   102→
   103→  useEffect(() => {
   104→    void refresh()
   105→    const timer = setInterval(() => void refresh(), 10000)
   106→    const onChanged = () => void refresh()
   107→    window.addEventListener('tt:positions-changed', onChanged)
   108→    window.addEventListener('tt:settings-changed', onChanged)
   109→    return () => {
   110→      clearInterval(timer)
   111→      window.removeEventListener('tt:positions-changed', onChanged)
   112→      window.removeEventListener('tt:settings-changed', onChanged)
   113→    }
   114→  }, [refresh])
   115→
   116→  const rows: PosConsoleRow[] = MODES.flatMap((m) =>
   117→    (data[m].positions || []).map((pos) => ({
   118→      mode: m,
   119→      pos,
   120→      sl: data[m].exits[normSym(pos.tradeSymbol)]?.sl,
   121→      tp: data[m].exits[normSym(pos.tradeSymbol)]?.tp,
   122→    })),
   123→  )
   124→  const pendingRows = MODES.flatMap((m) => (data[m].pending || []).map((p) => ({ mode: m as Mode, p })))
   125→  // realized round-trips across both modes (newest first from the API)
   126→  const historyRows = MODES.flatMap((m) =>
   127→    (history[m].trades || []).map((t) => ({ mode: m as Mode, t })),
   128→  ).sort((a, b) => new Date(b.t.exitAt).getTime() - new Date(a.t.exitAt).getTime())
   129→  const historyPnl = historyRows.reduce((acc, r) => acc + r.t.pnl, 0)
   130→  const noKeys = (m: Mode) => data[m].loaded && data[m].error === 'no-keys'
   131→
   132→  const closePosition = async (row: PosConsoleRow) => {
   133→    const key = `${row.mode}:${row.pos.tradeSymbol}`
   134→    setClosing(key)
   135→    try {
   136→      const res = await fetch(
   137→        `/api/trading/positions/${encodeURIComponent(row.pos.tradeSymbol)}?mode=${row.mode}`,
   138→        { method: 'DELETE' },
   139→      )
   140→      const json = await res.json().catch(() => null)
   141→      if (res.ok) {
   142→        toast({ title: `Closing ${row.pos.tradeSymbol} (${row.mode})`, description: 'Market close submitted' })
   143→      } else {
   144→        toast({ title: 'Close failed', description: json?.error || `HTTP ${res.status}`, variant: 'destructive' })
   145→      }
   146→    } catch {
   147→      toast({ title: 'Close failed', description: 'Network error', variant: 'destructive' })
   148→    } finally {
   149→      setClosing(null)
   150→      window.dispatchEvent(new CustomEvent('tt:positions-changed'))
   151→      void refresh()
   152→    }
   153→  }
   154→
   155→  const cancelPending = async (mode: Mode, orderId: string, label: string) => {
   156→    setCanceling(orderId)
   157→    try {
   158→      const res = await fetch(`/api/trading/orders/${orderId}?mode=${mode}`, { method: 'DELETE' })
   159→      const json = await res.json().catch(() => null)
   160→      if (res.ok) {
   161→        toast({ title: `Cancelled ${label}` })
   162→      } else {
   163→        toast({ title: 'Cancel failed', description: json?.error || `HTTP ${res.status}`, variant: 'destructive' })
   164→      }
   165→    } catch {
   166→      toast({ title: 'Cancel failed', description: 'Network error', variant: 'destructive' })
   167→    } finally {
   168→      setCanceling(null)
   169→      window.dispatchEvent(new CustomEvent('tt:positions-changed'))
   170→      void refresh()
   171→    }
   172→  }
   173→
   174→  const totals = rows.reduce(
   175→    (acc, r) => ({ upl: acc.upl + r.pos.upl, count: acc.count + 1 }),
   176→    { upl: 0, count: 0 },
   177→  )
   178→
   179→  return (
   180→    <div className="flex min-h-0 flex-1 flex-col gap-2 overflow-y-auto p-2 terminal-scroll">
   181→      {/* account strip: balance, buying power, daily P&L, equity curve, details */}
   182→      <AccountOverview />
   183→
   184→      {/* header */}
   185→      <div className="flex shrink-0 flex-wrap items-center gap-2 rounded-md border border-zinc-800/80 bg-zinc-900/70 px-3 py-2">
   186→        <span className="text-sm font-semibold text-zinc-100">Positions</span>
   187→        <Badge variant="outline" className="border-zinc-700 text-[10px] text-zinc-400">
   188→          {totals.count} open
   189→        </Badge>
   190→        <Badge variant="outline" className="border-amber-500/40 text-[10px] text-amber-400">
   191→          {pendingRows.length} pending
   192→        </Badge>
   193→        <Badge
   194→          variant="outline"
   195→          className="border-zinc-700 text-[10px] text-zinc-400"
   196→          title="Realized round-trip trades in the history group below"
   197→        >
   198→          <History className="mr-1 inline h-3 w-3" aria-hidden />
   199→          {historyRows.length} closed
   200→        </Badge>
   201→        {historyRows.length > 0 && (
   202→          <span
   203→            className={cn('text-sm font-semibold tabular-nums', historyPnl >= 0 ? 'text-emerald-400' : 'text-rose-400')}
   204→            title="Realized P&L of the trades in the history group below"
   205→          >
   206→            {historyPnl >= 0 ? '+' : '−'}${Math.abs(historyPnl).toFixed(2)} realized
   207→          </span>
   208→        )}
   209→        <span className={cn('text-sm font-semibold tabular-nums', totals.upl >= 0 ? 'text-emerald-400' : 'text-rose-400')}>
   210→          {totals.upl >= 0 ? '+' : '−'}${Math.abs(totals.upl).toFixed(2)} unrealized
   211→        </span>
   212→        <div className="ml-auto flex items-center gap-1.5">
   213→          {MODES.map((m) => (
   214→            <Badge
   215→              key={m}
   216→              variant="outline"
   217→              className={cn(
   218→                'text-[10px]',
   219→                m === 'live'
   220→                  ? 'border-amber-500/40 text-amber-400'
   221→                  : 'border-emerald-500/40 text-emerald-400',
   222→              )}
   223→              title={noKeys(m) ? `No Alpaca keys can serve ${m} — add them in Settings` : `${data[m].positions.length} position(s)`}
   224→            >
   225→              {m}
   226→              {noKeys(m) ? ' · no keys' : ''}
   227→            </Badge>
   228→          ))}
   229→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400 hover:text-zinc-100" onClick={() => void refresh()} aria-label="Refresh positions">
   230→            <RefreshCw className={cn('h-3.5 w-3.5', refreshing && 'animate-spin')} />
   231→          </Button>
   232→        </div>
   233→      </div>
   234→
   235→      {/* positions table — grows/shrinks with the available screen height
   236→          so taller screens reveal more rows (min keeps it usable on small
   237→          viewports, where the root scrolls instead) */}
   238→      <div className="flex min-h-[150px] flex-[5_1_auto] flex-col overflow-hidden rounded-md border border-zinc-800/80 bg-zinc-900/70">
   239→        <div className="min-h-0 flex-1 overflow-auto terminal-scroll">
   240→        <table className="w-full min-w-[900px] border-collapse text-xs">
   241→          <thead className="sticky top-0 z-10 bg-zinc-900 text-[10px] uppercase tracking-wider text-zinc-500">
   242→            <tr className="border-b border-zinc-800">
   243→              <th className="px-3 py-2 text-left font-semibold">Symbol</th>
   244→              <th className="px-3 py-2 text-left font-semibold">Market</th>
   245→              <th className="px-3 py-2 text-left font-semibold">Side</th>
   246→              <th className="px-3 py-2 text-right font-semibold">Qty</th>
   247→              <th className="px-3 py-2 text-right font-semibold">Entry price</th>
   248→              <th className="px-3 py-2 text-right font-semibold">Last</th>
   249→              <th className="px-3 py-2 text-right font-semibold">TP</th>
   250→              <th className="px-3 py-2 text-right font-semibold">SL</th>
   251→              <th className="px-3 py-2 text-right font-semibold">Trailing</th>
   252→              <th className="px-3 py-2 text-right font-semibold">P&amp;L</th>
   253→              <th className="px-3 py-2 text-right font-semibold">Action</th>
   254→            </tr>
   255→          </thead>
   256→          <tbody>
   257→            {rows.length === 0 && (
   258→              <tr>
   259→                <td colSpan={11} className="px-3 py-10 text-center text-zinc-500">
   260→                  {data.paper.loaded && data.live.loaded ? (
   261→                    <>
   262→                      No open positions on paper or live.
   263→                      <span className="mt-1 block text-[11px] text-zinc-600">
   264→                        Place a trade from the chart (right-click → trade ticket) and it appears here.
   265→                      </span>
   266→                    </>
   267→                  ) : (
   268→                    <span className="inline-flex items-center gap-2"><Loader2 className="h-4 w-4 animate-spin" /> Loading positions…</span>
   269→                  )}
   270→                </td>
   271→              </tr>
   272→            )}
   273→            {rows.map((r) => {
   274→              const key = `${r.mode}:${r.pos.tradeSymbol}`
   275→              const long = r.pos.side === 'long'
   276→              const up = r.pos.upl >= 0
   277→              const trailing = r.sl?.type === 'trailing_stop' ? `${r.sl.trailPercent ?? '—'}%` : null
   278→              return (
   279→                <tr key={key} className="border-b border-zinc-800/60 transition-colors hover:bg-zinc-800/40">
   280→                  <td className="px-3 py-2 font-semibold text-zinc-100">{r.pos.symbol}</td>
   281→                  <td className="px-3 py-2">
   282→                    <Badge
   283→                      variant="outline"
   284→                      className={cn(
   285→                        'text-[10px]',
   286→                        r.mode === 'live' ? 'border-amber-500/40 text-amber-400' : 'border-emerald-500/40 text-emerald-400',
   287→                      )}
   288→                    >
   289→                      {r.mode}
   290→                    </Badge>
   291→                  </td>
   292→                  <td className={cn('px-3 py-2 font-medium', long ? 'text-emerald-400' : 'text-rose-400')}>
   293→                    {long ? 'LONG' : 'SHORT'}
   294→                  </td>
   295→                  <td className="px-3 py-2 text-right tabular-nums text-zinc-300">{r.pos.qty}</td>
   296→                  <td className="px-3 py-2 text-right tabular-nums text-zinc-300">{fmtPrice(r.pos.avgEntry)}</td>
   297→                  <td className="px-3 py-2 text-right tabular-nums text-zinc-300">{fmtPrice(r.pos.current)}</td>
   298→                  <td className="px-3 py-2 text-right tabular-nums text-emerald-300">
   299→                    {r.tp?.limitPrice != null ? (
   300→                      fmtPrice(r.tp.limitPrice)
   301→                    ) : (
   302→                      <AddExitCell mode={r.mode} tradeSymbol={r.pos.tradeSymbol} kind="tp" onAdded={() => void refresh()} />
   303→                    )}
   304→                  </td>
   305→                  <td className="px-3 py-2 text-right tabular-nums text-rose-300">
   306→                    {r.sl && (r.sl.stopPrice != null || r.sl.limitPrice != null) ? (
   307→                      fmtPrice((r.sl.stopPrice ?? r.sl.limitPrice)!)
   308→                    ) : (
   309→                      <AddExitCell mode={r.mode} tradeSymbol={r.pos.tradeSymbol} kind="sl" onAdded={() => void refresh()} />
   310→                    )}
   311→                  </td>
   312→                  <td className="px-3 py-2 text-right tabular-nums text-amber-300">
   313→                    {trailing ?? <span className="text-zinc-600">—</span>}
   314→                  </td>
   315→                  <td className={cn('px-3 py-2 text-right font-semibold tabular-nums', up ? 'text-emerald-400' : 'text-rose-400')}>
   316→                    {up ? '+' : '−'}${Math.abs(r.pos.upl).toFixed(2)}
   317→                    <span className="ml-1 text-[10px] font-normal opacity-80">
   318→                      ({up ? '+' : '−'}{Math.abs(r.pos.uplPct).toFixed(2)}%)
   319→                    </span>
   320→                  </td>
   321→                  <td className="px-3 py-2 text-right">
   322→                    <Button
   323→                      size="sm"
   324→                      className="h-6 border border-rose-500/40 bg-rose-500/10 px-2 text-[11px] font-semibold text-rose-400 hover:bg-rose-500/20 hover:text-rose-300"
   325→                      disabled={closing === key}
   326→                      onClick={() => void closePosition(r)}
   327→                      aria-label={`Close ${r.mode} position ${r.pos.tradeSymbol}`}
   328→                    >
   329→                      {closing === key ? <Loader2 className="h-3 w-3 animate-spin" /> : 'Close position'}
   330→                    </Button>
   331→                  </td>
   332→                </tr>
   333→              )
   334→            })}
   335→          </tbody>
   336→        </table>
   337→        </div>
   338→      </div>
   339→
   340→      {/* pending (unfilled) entry orders — height follows the screen too */}
   341→      <div className="flex min-h-[130px] flex-[3_1_auto] flex-col rounded-md border border-zinc-800/80 bg-zinc-900/70">
   342→        <div className="flex items-center gap-2 border-b border-zinc-800/80 px-3 py-1.5">
   343→          <span className="text-xs font-semibold uppercase tracking-wider text-zinc-400">Pending orders</span>
   344→          <Badge variant="outline" className="border-zinc-700 text-[10px] text-zinc-500">{pendingRows.length}</Badge>
   345→          <span className="ml-auto text-[10px] text-zinc-600">unfilled entries · TP/SL plan shown until filled</span>
   346→        </div>
   347→        <div className="min-h-0 flex-1 overflow-auto terminal-scroll">
   348→          <table className="w-full min-w-[820px] border-collapse text-xs">
   349→            <thead className="sticky top-0 z-10 bg-zinc-900 text-[10px] uppercase tracking-wider text-zinc-500">
   350→              <tr className="border-b border-zinc-800">
   351→                <th className="px-3 py-1.5 text-left font-semibold">Symbol</th>
   352→                <th className="px-3 py-1.5 text-left font-semibold">Market</th>
   353→                <th className="px-3 py-1.5 text-left font-semibold">Side</th>
   354→                <th className="px-3 py-1.5 text-left font-semibold">Type</th>
   355→                <th className="px-3 py-1.5 text-right font-semibold">Qty</th>
   356→                <th className="px-3 py-1.5 text-right font-semibold">Filled</th>
   357→                <th className="px-3 py-1.5 text-right font-semibold">Limit</th>
   358→                <th className="px-3 py-1.5 text-right font-semibold">TP</th>
   359→                <th className="px-3 py-1.5 text-right font-semibold">SL</th>
   360→                <th className="px-3 py-1.5 text-left font-semibold">Status</th>
   361→                <th className="px-3 py-1.5 text-right font-semibold">Action</th>
   362→              </tr>
   363→            </thead>
   364→            <tbody>
   365→              {pendingRows.length === 0 && (
   366→                <tr>
   367→                  <td colSpan={11} className="px-3 py-6 text-center text-zinc-600">
   368→                    No pending orders.
   369→                  </td>
   370→                </tr>
   371→              )}
   372→              {pendingRows.map(({ mode, p }) => (
   373→                <tr key={`${mode}:${p.orderId}`} className="border-b border-zinc-800/60 transition-colors hover:bg-zinc-800/40">
   374→                  <td className="px-3 py-1.5 font-semibold text-zinc-100">{p.symbol}</td>
   375→                  <td className="px-3 py-1.5">
   376→                    <Badge
   377→                      variant="outline"
   378→                      className={cn('text-[10px]', mode === 'live' ? 'border-amber-500/40 text-amber-400' : 'border-emerald-500/40 text-emerald-400')}
   379→                    >
   380→                      {mode}
   381→                    </Badge>
   382→                  </td>
   383→                  <td className={cn('px-3 py-1.5 font-medium', p.side === 'buy' ? 'text-emerald-400' : 'text-rose-400')}>
   384→                    {p.side.toUpperCase()}
   385→                  </td>
   386→                  <td className="px-3 py-1.5 text-zinc-300">
   387→                    {p.type.toUpperCase()}
   388→                    {p.managed ? <span className="ml-1 text-[9px] text-amber-400/80">OCO</span> : null}
   389→                  </td>
   390→                  <td className="px-3 py-1.5 text-right tabular-nums text-zinc-300">{p.qty}</td>
   391→                  <td className="px-3 py-1.5 text-right tabular-nums text-zinc-500">{p.filledQty || '—'}</td>
   392→                  <td className="px-3 py-1.5 text-right tabular-nums text-zinc-300">
   393→                    {p.type === 'limit' || p.limitPrice != null ? (
   394→                      <PendingPriceCell mode={mode} p={p} field="entry" onSaved={() => void refresh()} />
   395→                    ) : (
   396→                      '—'
   397→                    )}
   398→                  </td>
   399→                  <td className="px-3 py-1.5 text-right tabular-nums text-emerald-300">
   400→                    <PendingPriceCell mode={mode} p={p} field="tp" onSaved={() => void refresh()} />
   401→                  </td>
   402→                  <td className="px-3 py-1.5 text-right tabular-nums text-rose-300">
   403→                    <PendingPriceCell mode={mode} p={p} field="sl" onSaved={() => void refresh()} />
   404→                  </td>
   405→                  <td className="px-3 py-1.5 text-zinc-400">{p.status}</td>
   406→                  <td className="px-3 py-1.5 text-right">
   407→                    <Button
   408→                      size="sm"
   409→                      className="h-5 border border-zinc-700 bg-transparent px-2 text-[10px] font-semibold text-zinc-400 hover:bg-rose-500/10 hover:text-rose-400"
   410→                      disabled={canceling === p.orderId}
   411→                      onClick={() => void cancelPending(mode, p.orderId, `${p.side} ${p.type} ${p.symbol}`)}
   412→                      aria-label={`Cancel pending ${p.side} ${p.type} ${p.symbol}`}
   413→                    >
   414→                      {canceling === p.orderId ? <Loader2 className="h-3 w-3 animate-spin" /> : 'Cancel'}
   415→                    </Button>
   416→                  </td>
   417→                </tr>
   418→              ))}
   419→            </tbody>
   420→          </table>
   421→        </div>
   422→      </div>
   423→
   424→      {/* trade history — realized round-trips (FIFO-matched closed orders),
   425→          sharing the remaining screen height with the tables above */}
   426→      <div className="flex min-h-[130px] flex-[3_1_auto] flex-col rounded-md border border-zinc-800/80 bg-zinc-900/70">
   427→        <div className="flex items-center gap-2 border-b border-zinc-800/80 px-3 py-1.5">
   428→          <History className="h-3.5 w-3.5 text-zinc-500" aria-hidden />

   521→/** Click-to-edit price cell for a pending order's entry limit / TP / SL.
   522→ * PATCHes the same endpoint the chart's line-drag uses. */
   523→function PendingPriceCell({ mode, p, field, onSaved }: {
   524→  mode: Mode
   525→  p: PendingOrderInfo
   526→  field: 'entry' | 'tp' | 'sl'
   527→  onSaved: () => void
   528→}) {
   529→  const { toast } = useToast()
   530→  const [open, setOpen] = useState(false)
   531→  const [v, setV] = useState('')
   532→  const [busy, setBusy] = useState(false)
   533→  const value = field === 'entry' ? p.limitPrice : field === 'tp' ? p.tpPrice : p.slPrice
   534→
   535→  const save = async () => {
   536→    const price = parseFloat(v)
   537→    if (!Number.isFinite(price) || price <= 0) return
   538→    setBusy(true)
   539→    try {
   540→      const body: Record<string, unknown> = { mode, crypto: p.symbol.includes('/') }
   541→      if (field === 'entry') body.entry_limit = price
   542→      else if (field === 'tp') body.tp_price = price
   543→      else body.sl_price = price
   544→      const res = await fetch(`/api/trading/orders/${p.orderId}`, {
   545→        method: 'PATCH',
   546→        headers: { 'content-type': 'application/json' },
   547→        body: JSON.stringify(body),
   548→      })
   549→      const json = await res.json().catch(() => null)
   550→      if (res.ok) {
   551→        toast({
   552→          title: `${field === 'entry' ? 'Entry' : field.toUpperCase()} updated to ${fmtPrice(price)}`,
   553→          description: json?.method === 'cancel-replace' ? 'Order re-created on Alpaca (new id)' : json?.note || undefined,
   554→        })
   555→      } else {
   556→        toast({ title: 'Update failed', description: json?.error || `HTTP ${res.status}`, variant: 'destructive' })
   557→      }
   558→    } catch {
   559→      toast({ title: 'Update failed', description: 'Network error', variant: 'destructive' })
   560→    } finally {
   561→      setBusy(false)
   562→      setOpen(false)
   563→      onSaved()
   564→      window.dispatchEvent(new CustomEvent('tt:positions-changed'))
   565→    }
   566→  }
   567→
   568→  if (!open) {
   569→    return (
   570→      <button
   571→        className="tabular-nums underline-offset-2 transition-colors hover:text-emerald-300 hover:underline decoration-dashed"
   572→        onClick={() => { setV(value != null ? String(value) : ''); setOpen(true) }}
   573→        aria-label={`Edit ${field === 'entry' ? 'entry limit' : field.toUpperCase()} of pending ${p.side} ${p.symbol}`}
   574→        title={`Click to change the ${field === 'entry' ? 'entry limit' : field.toUpperCase()} price`}
   575→      >
   576→        {value != null ? fmtPrice(value) : <span className="text-zinc-600">—</span>}
   577→      </button>
   578→    )
   579→  }
   580→  return (
   581→    <span className="inline-flex items-center gap-1">
   582→      <Input
   583→        autoFocus
   584→        type="number"
   585→        step="any"
   586→        value={v}
   587→        onChange={(e) => setV(e.target.value)}
   588→        onKeyDown={(e) => {
   589→          if (e.key === 'Enter') void save()
   590→          if (e.key === 'Escape') setOpen(false)
   591→        }}
   592→        className="h-6 w-24 border-zinc-700 bg-zinc-950 text-[11px]"
   593→        aria-label={`${field === 'entry' ? 'Entry limit' : field.toUpperCase()} price`}
   594→      />
   595→      <button
   596→        className="text-[10px] font-semibold text-emerald-400 hover:text-emerald-300"
   597→        onClick={() => void save()}
   598→        disabled={busy}
   599→        aria-label="Apply price"
   600→      >
   601→        {busy ? <Loader2 className="h-3 w-3 animate-spin" /> : 'OK'}
   602→      </button>
   603→      <button className="text-[10px] text-zinc-500 hover:text-zinc-300" onClick={() => setOpen(false)} aria-label="Cancel edit">
   604→        ✕
   605→      </button>
   606→    </span>
   607→  )
   608→}

   610→/** Small "+ TP"/"+ SL" button for a position row missing that exit — opens an
   611→ * inline price input, POSTs to /positions/[symbol]/exits. */
   612→function AddExitCell({ mode, tradeSymbol, kind, onAdded }: {
   613→  mode: Mode
   614→  tradeSymbol: string
   615→  kind: 'tp' | 'sl'
   616→  onAdded: () => void
   617→}) {

   620→  const [v, setV] = useState('')
   621→  const [busy, setBusy] = useState(false)

   624→    const price = parseFloat(v)
   625→    if (!Number.isFinite(price) || price <= 0) return
   626→    setBusy(true)

   628→      const res = await fetch(`/api/trading/positions/${encodeURIComponent(tradeSymbol)}/exits`, {

   679→      />

   688→      <button
   689→        className="text-[10px] text-zinc-500 hover:text-zinc-300"
