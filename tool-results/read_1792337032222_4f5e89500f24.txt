   170→  // ---------- coordinate helpers ----------
   171→  // Time → x. timeToCoordinate() only resolves EXACT bar times, so drawings
   172→  // captured between bars / in the left-right margins previously vanished or
   173→  // snapped to the wrong bar. Robust version: exact hit → interpolate between
   174→  // the bracketing bars (handles weekend/overnight gaps) → extrapolate into
   175→  // the margins using the per-bar pixel spacing.
   176→  const xOf = useCallback((t: number | null): number | null => {
   177→    if (t == null || !chartRef.current) return null
   178→    const ts = chartRef.current.timeScale()
   179→    const bars = barsRef.current
   180→    const exact = ts.timeToCoordinate(t as UTCTimestamp)
   181→    if (exact != null) return Number(exact)
   182→    if (bars.length === 0) return null
   183→    const last = bars.length - 1
   184→    const barX = (i: number): number | null => {
   185→      const x = ts.timeToCoordinate(bars[i].time as UTCTimestamp)
   186→      return x == null ? null : Number(x)
   187→    }
   188→    // pixels per bar from two reference bars (index→x is globally linear)
   189→    const pxPerBar = (): number | null => {
   190→      const vr = ts.getVisibleLogicalRange()
   191→      let a = vr ? Math.ceil(Number(vr.from)) : 0
   192→      let b = vr ? Math.floor(Number(vr.to)) : last
   193→      a = Math.max(0, Math.min(a, last))
   194→      b = Math.max(0, Math.min(b, last))
   195→      if (b === a) {
   196→        a = Math.max(0, a - 1)
   197→        b = Math.min(last, b + 1)
   198→      }
   199→      if (b === a) return null
   200→      const xa = barX(a)
   201→      const xb = barX(b)
   202→      if (xa == null || xb == null || xb === xa) return null
   203→      return (xb - xa) / (b - a)
   204→    }
   205→    if (t > bars[last].time) {
   206→      const x1 = barX(last)
   207→      if (x1 == null) return null
   208→      const ppb = pxPerBar()
   209→      if (ppb == null) return x1
   210→      return x1 + ((t - bars[last].time) / TIMEFRAMES[tf].seconds) * ppb
   211→    }
   212→    if (t < bars[0].time) {
   213→      const x0 = barX(0)
   214→      if (x0 == null) return null
   215→      const ppb = pxPerBar()
   216→      if (ppb == null) return x0
   217→      return x0 + ((t - bars[0].time) / TIMEFRAMES[tf].seconds) * ppb
   218→    }
   219→    // bracket: first bar with time >= t, interpolate to the previous bar
   220→    let lo = 0
   221→    let hi = last
   222→    while (lo < hi) {
   223→      const mid = (lo + hi) >> 1
   224→      if (bars[mid].time < t) lo = mid + 1
   225→      else hi = mid
   226→    }
   227→    const j = lo
   228→    const i = Math.max(0, j - 1)
   229→    const xa = barX(i)
   230→    const xb = barX(j)
   231→    if (xa == null || xb == null) return null
   232→    const dt = bars[j].time - bars[i].time
   233→    if (dt <= 0) return xa
   234→    return xa + ((t - bars[i].time) / dt) * (xb - xa)
   235→  }, [tf])
   236→

   249→  // Time under the cursor, CONTINUOUS (no snapping). lightweight-charts'
   250→  // coordinateToTime() ceils to the NEXT bar, which used to place drawn lines
   251→  // up to a full bar-width right of the arrow. We derive the exact fractional
   252→  // bar index from two reference bars (public API only) and map it to a
   253→  // continuous time: between bars → interpolate across their actual time gap,
   254→  // beyond the data → extend by the timeframe seconds. Round-trips exactly
   255→  // through xOf() above, so the committed line lands precisely under the
   256→  // cursor — matching TradingView (magnet off).
   257→  const timeAt = useCallback((x: number): number | null => {
   258→    const chart = chartRef.current
   259→    const bars = barsRef.current
   260→    if (!chart || bars.length === 0) return null
   261→    const ts = chart.timeScale()
   262→    const last = bars.length - 1
   263→    const vr = ts.getVisibleLogicalRange()
   264→    let a = vr ? Math.ceil(Number(vr.from)) : 0
   265→    let b = vr ? Math.floor(Number(vr.to)) : last
   266→    a = Math.max(0, Math.min(a, last))
   267→    b = Math.max(0, Math.min(b, last))
   268→    if (b === a) {
   269→      a = Math.max(0, a - 1)
   270→      b = Math.min(last, b + 1)
   271→    }
   272→    if (b === a) return null
   273→    const xa = ts.timeToCoordinate(bars[a].time as UTCTimestamp)
   274→    const xb = ts.timeToCoordinate(bars[b].time as UTCTimestamp)
   275→    if (xa == null || xb == null || Number(xb) === Number(xa)) return null
   276→    const fi = a + ((x - Number(xa)) / (Number(xb) - Number(xa))) * (b - a)
   277→    const tfSec = TIMEFRAMES[tf].seconds
   278→    if (fi >= last) return bars[last].time + (fi - last) * tfSec
   279→    if (fi <= 0) return bars[0].time + fi * tfSec
   280→    const i = Math.floor(fi)
   281→    const f = fi - i
   282→    const t0 = bars[i].time
   283→    const t1 = bars[i + 1].time
   284→    return t1 > t0 ? t0 + f * (t1 - t0) : t0
   285→  }, [tf])

   397→    }
   398→
   399→    // ---- y-axis event icons: earnings [E] (yfinance) + news ⚡ (TradingView-style) ----
   400→    const chartApi = chartRef.current
   401→    const mk = markersRef.current
   402→    axisIconsRef.current = []
   403→    if (chartApi && (mk.earnings.length > 0 || mk.news.length > 0) && barsRef.current.length > 0) {
   404→      const psW = chartApi.priceScale('right').width()
   405→      const tAxisH = chartApi.timeScale().height()
   406→      const bars = barsRef.current
   407→      const vis = chartApi.timeScale().getVisibleRange()
   408→      const visFrom = vis ? Number(vis.from) : null
   409→      const visTo = vis ? Number(vis.to) : null
   410→      const cx = w - psW / 2
   411→      const yMin = 13
   412→      const yMax = Math.max(yMin + 1, h - tAxisH - 10)
   413→
   414→      // bar in effect at the event moment (upcoming events → last bar)
   415→      const barAt = (ts: number): Bar | null => {
   416→        if (ts < bars[0].time) return null
   417→        if (ts >= bars[bars.length - 1].time) return bars[bars.length - 1]
   418→        let lo = 0
   419→        let hi = bars.length - 1
   420→        while (lo < hi) {
   421→          const mid = (lo + hi + 1) >> 1
   422→          if (bars[mid].time <= ts) lo = mid
   423→          else hi = mid - 1
   424→        }
   425→        return bars[lo]
   426→      }
   427→      const fmtDay = (ts: number) =>
   428→        new Date(ts * 1000).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' })
   429→
   430→      type AxisIcon = { kind: 'e' | 'n'; rawY: number; tip: string; atLast?: boolean }
   431→      const icons: AxisIcon[] = []
   432→      const lastBarTime = bars[bars.length - 1].time

   438→        if (y == null) continue

   440→          e.epsActual != null
   441→            ? ` · EPS ${e.epsActual}${e.epsEstimate != null ? ` (est ${e.epsEstimate})` : ''}`
   442→            : ''
   443→        icons.push({ kind: 'e', rawY: y, tip: `Earnings · ${fmtDay(e.ts)}${e.upcoming ? ' · upcoming' : ''}${epsBits}`, atLast: bar.time === lastBarTime })
   444→      }
   445→      for (const n of mk.news) {
   446→        const bar = barAt(n.ts)
   447→        if (!bar) continue
   448→        if (visFrom != null && visTo != null && (bar.time < visFrom || bar.time > visTo)) continue

   577→  function segPath(ctx: CanvasRenderingContext2D, p1: { x: number; y: number }, p2: { x: number; y: number }, w: number, extendRight: boolean) {
   578→    ctx.beginPath()
   579→    ctx.moveTo(p1.x, p1.y)
   580→    if (extendRight) {
   581→      const dx = p2.x - p1.x
   582→      const dy = p2.y - p1.y
   583→      const farX = dx !== 0 ? (w - p1.x) / dx : Infinity
   584→      const ex = farX >= 0 ? w : p1.x - dx * 100
   585→      const ey = p1.y + dy * (farX >= 0 ? farX : -100)
   586→      ctx.lineTo(ex, ey)
   587→    } else {
   588→      ctx.lineTo(p2.x, p2.y)
   589→    }
   590→    ctx.stroke()
   591→  }

   593→  function drawDrawing(ctx: CanvasRenderingContext2D, d: Drawing, w: number, h: number, highlighted: boolean) {
   594→    ctx.save()
   595→    ctx.strokeStyle = d.color
   596→    ctx.lineWidth = d.width + (highlighted ? 1 : 0)
   597→    ctx.globalAlpha = 1
   598→
   599→    if (d.kind === 'hline') {
   600→      const y = yOf(d.points[0]?.price ?? 0)
   601→      if (y != null) {
   602→        hlinePath(ctx, y, w)
   603→        drawChip(ctx, w - 74, y, fmtPrice(d.points[0].price), d.color, highlighted)
   604→        // move grip dots at center
   605→        dot(ctx, w / 2, y, d.color, highlighted)
   606→      }
   607→    } else if (d.kind === 'vline') {
   608→      const x = xOf(d.points[0]?.time ?? null)
   609→      if (x != null) {
   610→        vlinePath(ctx, x, h)
   611→        dot(ctx, x, h / 2, d.color, highlighted)
   612→      }
   613→    } else if (d.kind === 'trend' || d.kind === 'fib') {
   614→      const p1 = pixelPoint(d.points[0])
   615→      const p2 = pixelPoint(d.points[1])
   616→      if (p1 && p2) {
   617→        if (d.kind === 'trend') {
   618→          segPath(ctx, p1, p2, w, false)
   619→        } else {
   620→          // fib: levels between p1..p2 extended to right edge
   621→          const xLeft = Math.min(p1.x, p2.x)
   622→          const price1 = d.points[0].price
   623→          const price2 = d.points[1].price
   624→          ctx.font = '10px ui-monospace, monospace'
   625→          for (const lv of FIB_LEVELS) {
   626→            const priceLv = price2 + (price1 - price2) * lv
   627→            const y = yOf(priceLv)
   628→            if (y == null) continue
   629→            ctx.globalAlpha = lv === 0 || lv === 1 ? 0.9 : 0.6
   630→            ctx.setLineDash(lv === 0.5 ? [6, 3] : [])
   631→            ctx.beginPath()
   632→            ctx.moveTo(xLeft, Math.round(y) + 0.5)
   633→            ctx.lineTo(w, Math.round(y) + 0.5)
   634→            ctx.stroke()
   635→            ctx.globalAlpha = 0.95
   636→            ctx.fillStyle = d.color
   637→            const label = `${lv.toFixed(3)}  ${fmtPrice(priceLv)}`
   638→            ctx.fillText(label, xLeft + 4, y - 3)
   639→          }
   640→        }
   641→        dot(ctx, p1.x, p1.y, d.color, highlighted)
   642→        dot(ctx, p2.x, p2.y, d.color, highlighted)
   643→      }
   644→    }
   645→    ctx.restore()
   646→  }

   722→  const distToSeg = (px: number, py: number, a: { x: number; y: number }, b: { x: number; y: number }) => {
   723→    const dx = b.x - a.x
   724→    const dy = b.y - a.y
   725→    const len2 = dx * dx + dy * dy
   726→    if (len2 === 0) return Math.hypot(px - a.x, py - a.y)
   727→    let t = ((px - a.x) * dx + (py - a.y) * dy) / len2
   728→    t = Math.max(0, Math.min(1, t))
   729→    return Math.hypot(px - (a.x + t * dx), py - (a.y + t * dy))
   730→  }

   732→  const hitTest = useCallback((x: number, y: number): SceneHit | null => {
   733→    const st = useTerminal.getState()
   734→    const ds = st.drawingsBySymbol[symbol] || []
   735→    const als = st.alerts.filter((a) => a.symbol === symbol && a.status !== 'triggered')
   736→
   737→    // trade lines take priority (position entry / SL / TP)
   738→    const posBundle = posRef.current
   739→    const myPos = posBundle?.positions.find((p) => normSym(p.symbol) === normSym(symbol))
   740→    const myExits = posBundle?.exits[normSym(symbol)]
   741→    if (myExits?.sl && (myExits.sl.stopPrice != null || myExits.sl.limitPrice != null)) {
   742→      const slPrice = myExits.sl.stopPrice ?? myExits.sl.limitPrice!
   743→      if (myExits.sl.type !== 'trailing_stop') {
   744→        const yl = yOf(slPrice)
   745→        if (yl != null && Math.abs(y - yl) < 7) {
   746→          return { type: 'exit', id: myExits.sl.orderId, part: 'sl', origPrice: slPrice, exit: myExits.sl }
   747→        }
   748→      }
   749→    }
   750→    if (myExits?.tp && myExits.tp.limitPrice != null) {
   751→      const yl = yOf(myExits.tp.limitPrice)
   752→      if (yl != null && Math.abs(y - yl) < 7) {
   753→        return { type: 'exit', id: myExits.tp.orderId, part: 'tp', origPrice: myExits.tp.limitPrice, exit: myExits.tp }
   754→      }
   755→    }
   756→    if (myPos && myPos.avgEntry > 0) {
   757→      const yl = yOf(myPos.avgEntry)
   758→      if (yl != null && Math.abs(y - yl) < 6) {
   759→        return { type: 'position', id: myPos.tradeSymbol, part: 'whole', origPrice: myPos.avgEntry }
   760→      }
   761→    }
   762→
   763→    if (showDrawings) {
   764→      for (let i = ds.length - 1; i >= 0; i--) {
   765→        const d = ds[i]
   766→        if (d.kind === 'hline') {
   767→          const yl = yOf(d.points[0]?.price ?? 0)
   768→          if (yl != null && Math.abs(y - yl) < 7) return { type: 'drawing', id: d.id, part: 'whole' }
   769→        } else if (d.kind === 'vline') {
   770→          const xl = xOf(d.points[0]?.time ?? null)
   771→          if (xl != null && Math.abs(x - xl) < 7) return { type: 'drawing', id: d.id, part: 'whole' }
   772→        } else {
   773→          const p1 = pixelPoint(d.points[0])
   774→          const p2 = pixelPoint(d.points[1])
   775→          if (p1 && Math.hypot(x - p1.x, y - p1.y) < 8) return { type: 'drawing', id: d.id, part: 'point', index: 0 }
   776→          if (p2 && Math.hypot(x - p2.x, y - p2.y) < 8) return { type: 'drawing', id: d.id, part: 'point', index: 1 }
   777→          if (d.kind === 'trend' && p1 && p2 && distToSeg(x, y, p1, p2) < 7) {
   778→            return { type: 'drawing', id: d.id, part: 'whole' }
   779→          }
   780→          if (d.kind === 'fib' && p1 && p2) {
   781→            const xLeft = Math.min(p1.x, p2.x)
   782→            if (x >= xLeft - 4) {
   783→              const price1 = d.points[0].price
   784→              const price2 = d.points[1].price
   785→              for (const lv of FIB_LEVELS) {
   786→                const yl = yOf(price2 + (price1 - price2) * lv)
   787→                if (yl != null && Math.abs(y - yl) < 6) return { type: 'drawing', id: d.id, part: 'whole' }
   788→              }
   789→            }
   790→          }
   791→        }
   792→      }
   793→    }
   794→    for (let i = als.length - 1; i >= 0; i--) {
   795→      const yl = yOf(als[i].price)
   796→      if (yl != null && Math.abs(y - yl) < 7) return { type: 'alert', id: als[i].id, part: 'whole' }
   797→    }
   798→    return null
