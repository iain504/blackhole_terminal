     1→// Alpaca Trading API client (server-side only): paper + live order routing,
     2→// positions, bracket/OCO exits, order replacement for SL/TP drag updates.
     3→//
     4→// Verified against the paper API (2026-10):
     5→// - Stocks support order_class=bracket (entry + take_profit/stop_loss legs).
     6→// - Crypto rejects ALL advanced order classes (order_class=oco -> 40010001
     7→//   "invalid order type for crypto order"; bracket -> 42210000) AND plain
     8→//   `stop` type. Sell exits also CANNOT be reserved against a PENDING buy
     9→//   (40310000 "insufficient balance" — only settled position qty counts),
    10→//   so crypto complex orders are EMULATED: the entry carries its intended
    11→//   TP/SL in the client_order_id (`ttb-…`) and the OCO supervisor below
    12→//   attaches the exit legs (`tte-…`) the moment the entry FILLS, canceling
    13→//   any surviving leg once the position is gone.
    14→// - PATCH /v2/orders/{id} creates a REPLACEMENT order with a NEW id.
    15→// - Position symbols drop the crypto slash (BTC/USD -> BTCUSD); ORDER
    16→//   symbols keep it (BTC/USD).
    17→import { db } from './db'
    18→import { getAlpacaKeys } from './alpaca'

    39→/** Requested mode wins; else the user's stored preference; else paper. */
    40→export async function resolveTradeMode(userId: string | undefined, requested?: unknown): Promise<TradeMode> {
    41→  if (isTradeMode(requested)) return requested
    42→  if (userId) {
    43→    const user = await db.user.findUnique({ where: { id: userId } })
    44→    const s = (user?.settings as UserSettings | null) || null
    45→    if (isTradeMode(s?.tradeMode)) return s.tradeMode
    46→  }
    47→  return 'paper'
    48→}

    55→interface TradeFetchResult<T> {
    56→  data: T | null
    57→  status: number
    58→  error?: string
    59→}
    60→
    61→async function tradeFetch<T>(
    62→  userId: string,
    63→  mode: TradeMode,
    64→  path: string,
    65→  init: RequestInit = {},
    66→): Promise<TradeFetchResult<T>> {
    67→  const keys = await getAlpacaKeys(userId)
    68→  if (!keys) return { data: null, status: 0, error: 'no-keys' }
    69→  const controller = new AbortController()
    70→  const timer = setTimeout(() => controller.abort(), 9000)
    71→  try {
    72→    const res = await fetch(`${TRADE_BASE[mode]}${path}`, {
    73→      ...init,
    74→      headers: {
    75→        'APCA-API-KEY-ID': keys.keyId,
    76→        'APCA-API-SECRET-KEY': keys.secret,
    77→        accept: 'application/json',
    78→        ...(init.body ? { 'content-type': 'application/json' } : {}),
    79→        ...(init.headers || {}),
    80→      },
    81→      signal: controller.signal,
    82→      cache: 'no-store',
    83→    })
    84→    const text = await res.text()
    85→    let parsed: unknown = null
    86→    try {
    87→      parsed = text ? JSON.parse(text) : null
    88→    } catch {
    89→      parsed = null
    90→    }
    91→    if (!res.ok) {
    92→      const msg =
    93→        (parsed as { message?: string } | null)?.message ||
    94→        `Trading API HTTP ${res.status}`
    95→      console.error(`Trading ${mode} ${path} -> ${res.status}: ${msg}`)
    96→      return { data: parsed as T, status: res.status, error: msg }
    97→    }
    98→    return { data: parsed as T, status: res.status }
    99→  } catch (e) {
   100→    console.error(`Trading ${mode} ${path} failed:`, e instanceof Error ? e.message : e)
   101→    return { data: null, status: 0, error: e instanceof Error ? e.message : 'network error' }
   102→  } finally {
   103→    clearTimeout(timer)
   104→  }
   105→}

   188→/**
   189→ * Place an entry order with optional exits.
   190→ * Stocks: single bracket request when TP/SL present (Alpaca attaches legs).
   191→ * Crypto: advanced classes are rejected — the entry is placed ALONE carrying
   192→ * the intended TP/SL in its client_order_id (`ttb-…`); the OCO supervisor
   193→ * (ensureCryptoOcoSupervisor) attaches the exit legs when the entry fills.
   194→ */
   195→export async function placeOrderWithExits(
   196→  userId: string,
   197→  mode: TradeMode,
   198→  body: PlaceOrderBody,
   199→): Promise<{ results: TradingOrder[]; errors: string[]; splitNote?: string }> {
   200→  const isCrypto = body.symbol.includes('/')
   201→  const type = body.type || 'market'
   202→  const tif = body.time_in_force || (isCrypto ? 'gtc' : type === 'market' ? 'day' : 'gtc')
   203→  const entry: Record<string, unknown> = {
   204→    symbol: tradeSymbol(body.symbol),
   205→    qty: String(body.qty),
   206→    side: body.side,
   207→    type,
   208→    time_in_force: tif,
   209→  }
   210→  if (type === 'limit') {
   211→    if (!body.limit_price) return { results: [], errors: ['Limit price required for limit orders'] }
   212→    entry.limit_price = String(body.limit_price)
   213→  }
   214→  if (body.client_order_id) entry.client_order_id = body.client_order_id
   215→
   216→  const errors: string[] = []
   217→  const results: TradingOrder[] = []
   218→
   219→  if (!isCrypto) {
   220→    // Stocks: bracket handles entry + legs atomically.
   221→    const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   222→    const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   223→    if (hasTp || hasSl) {
   224→      entry.order_class = 'bracket'
   225→      if (hasTp) entry.take_profit = { limit_price: body.take_profit!.toFixed(2) }
   226→      if (hasSl) entry.stop_loss = { stop_price: body.stop_loss!.toFixed(2) }
   227→      // encode the plan for the chart's pending display: Alpaca materializes
   228→      // bracket LEGS only when the entry activates, so a queued (weekend /
   229→      // pre-market) bracket would otherwise render without its TP/SL lines
   230→      entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit : null, hasSl ? body.stop_loss : null)
   231→    }
   232→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   233→      method: 'POST',
   234→      body: JSON.stringify(entry),
   235→    })
   236→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   237→    else errors.push(r.error || 'Entry order failed')
   238→    return { results, errors }
   239→  }
   240→
   241→  // ---- Crypto: complex orders are EMULATED (no native bracket/OCO) ----
   242→  const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   243→  const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   244→  if (hasTp || hasSl) {
   245→    // encode the planned exits so the chart can render them while the entry
   246→    // is pending AND the supervisor can attach the legs without extra state
   247→    entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit : null, hasSl ? body.stop_loss : null)
   248→  }
   249→  const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   250→    method: 'POST',
   251→    body: JSON.stringify(entry),
   252→  })
   253→  if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   254→  else {
   255→    errors.push(r.error || 'Entry order failed')
   256→    return { results, errors }
   257→  }
   258→  if (!hasTp && !hasSl) return { results, errors }
   259→
   260→  // Fast path: a MARKET entry usually fills instantly — poll briefly and
   261→  // attach the exits in-line so protection starts immediately. Limit entries
   262→  // return right away (the supervisor attaches on fill within ~6s).
   263→  if (type === 'market') {
   264→    let cur: TradingOrder | null = r.data
   265→    for (let i = 0; i < 4 && cur && cur.status !== 'filled'; i++) {
   266→      await new Promise((res) => setTimeout(res, 600))
   267→      const s = await tradeFetch<TradingOrder>(userId, mode, `/v2/orders/${(r.data as TradingOrder).id}`)
   268→      cur = s.data ?? cur
   269→      if (cur && ['canceled', 'cancelled', 'rejected', 'expired'].includes(cur.status)) break
   270→    }
   271→    if (cur?.status === 'filled') {
   272→      const legErrs = await placeCryptoExitLegs(
   273→        userId,
   274→        mode,
   275→        cur,
   276→        hasTp ? body.take_profit : undefined,
   277→        hasSl ? body.stop_loss : undefined,
   278→      )
   279→      errors.push(...legErrs)
   280→      return { results, errors }
   281→    }
   282→  }
   283→  return {
   284→    results,
   285→    errors,
   286→    splitNote: 'TP/SL attach automatically when the entry fills (OCO-managed)',
   287→  }
   288→}

   290→// ---------- crypto complex-order emulation (OCO supervisor) ----------

   292→/** Price -> client_order_id-safe digits (6dp fixed point). */
   293→function encP(v: number): string {
   294→  return String(Math.round(v * 1e6))
   295→}
   296→
   297→/** client_order_id prefix identifying a MANAGED crypto exit leg. */
   298→function exitLegClientId(entryOrderId: string, kind: 'tp' | 'sl'): string {
   299→  return `tte-${entryOrderId}-${kind}`
   300→}
   301→
   302→/** Encode the OCO/bracket plan (mode + TP/SL prices) into the entry's client_order_id. */
   303→function encodePlanClientId(mode: TradeMode, tp: number | null, sl: number | null): string {
   304→  const rand = Math.random().toString(36).slice(2, 8)
   305→  return `ttb-${mode[0]}-tp${tp != null && tp > 0 ? encP(tp) : '0'}-sl${sl != null && sl > 0 ? encP(sl) : '0'}-${rand}`
   306→}
   307→
   308→/** Decode the plan from a `ttb-…` client_order_id. Returns null when not ours. */
   309→function parseCryptoPlan(clientOrderId: string | undefined | null): { tp: number | null; sl: number | null } | null {
   310→  const m = /^ttb-[a-z](?:-tp(\d+))?(?:-sl(\d+))?-/.exec(clientOrderId || '')
   311→  if (!m) return null
   312→  const dec = (s: string | undefined) => (s ? parseInt(s, 10) / 1e6 : 0)
   313→  const tp = dec(m[1])
   314→  const sl = dec(m[2])
   315→  return { tp: tp > 0 ? tp : null, sl: sl > 0 ? sl : null }
   316→}
   317→
   318→/**
   319→ * Place the exit legs for a FILLED crypto entry (50/50 qty split when both
   320→ * legs are requested — Alpaca reserves balance per open sell order). Leg ids
   321→ * are deterministic (`tte-<entryId>-tp|-sl`) so the supervisor is idempotent.
   322→ * Crypto buys receive qty minus a trading fee, so legs apply a 0.99 safety
   323→ * factor to avoid 403 "insufficient balance".
   324→ */
   325→async function placeCryptoExitLegs(
   326→  userId: string,
   327→  mode: TradeMode,
   328→  entry: TradingOrder,
   329→  tp?: number,
   330→  sl?: number,
   331→): Promise<string[]> {
   332→  const errors: string[] = []
   333→  const filledQty = parseFloat(entry.filled_qty) || parseFloat(entry.qty) || 0
   334→  if (!(filledQty > 0)) return ['entry has no filled qty']
   335→  const exitSide = entry.side === 'buy' ? 'sell' : 'buy'
   336→  const both = tp != null && sl != null
   337→  const legQty = ((both ? filledQty / 2 : filledQty) * 0.99).toFixed(8).replace(/0+$/, '').replace(/\.$/, '')
   338→  if (tp != null) {
   339→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   340→      method: 'POST',
   341→      body: JSON.stringify({
   342→        symbol: entry.symbol,
   343→        qty: legQty,
   344→        side: exitSide,
   345→        type: 'limit',
   346→        time_in_force: 'gtc',
   347→        limit_price: tp.toFixed(tp < 1 ? 6 : 2),
   348→        client_order_id: exitLegClientId(entry.id, 'tp'),
   349→      }),
   350→    })
   351→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`TP: ${r.error || 'failed'}`)
   352→  }
   353→  if (sl != null) {
   354→    // stop-limit sell: limit slightly below the stop so it clears the book
   355→    const limP = exitSide === 'sell' ? sl * 0.999 : sl * 1.001
   356→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   357→      method: 'POST',
   358→      body: JSON.stringify({
   359→        symbol: entry.symbol,
   360→        qty: legQty,
   361→        side: exitSide,
   362→        type: 'stop_limit',
   363→        time_in_force: 'gtc',
   364→        stop_price: sl.toFixed(sl < 1 ? 6 : 2),
   365→        limit_price: limP.toFixed(limP < 1 ? 6 : 2),
   366→        client_order_id: exitLegClientId(entry.id, 'sl'),
   367→      }),
   368→    })
   369→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`SL: ${r.error || 'failed'}`)
   370→  }
   371→  return errors
   372→}

   374→const OCO_CYCLE_MS = 6000
   375→const OCO_OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   376→
   377→/** Supervised (user,mode) pairs — survives dev hot reloads via globalThis. */
   378→function supervisedSet(): Set<string> {
   379→  const g = globalThis as { __ttOcoSupervised?: Set<string> }
   380→  if (!g.__ttOcoSupervised) g.__ttOcoSupervised = new Set<string>()
   381→  return g.__ttOcoSupervised
   382→}
   383→
   384→/**
   385→ * Start the OCO supervisor for a (user, mode) pair (idempotent). One cycle
   386→ * every 6s: (a) cancel managed exit legs whose position is gone (OCO
   387→ * semantics — when the position closes, the surviving leg dies),
   388→ * (b) attach exit legs to managed entries that have FILLED.
   389→ */
   390→export function ensureCryptoOcoSupervisor(userId: string, mode: TradeMode): void {
   391→  const key = `${userId}|${mode}`
   392→  if (supervisedSet().has(key)) return
   393→  supervisedSet().add(key)
   394→  const timer = setInterval(() => {
   395→    void runOcoCycle(userId, mode)
   396→  }, OCO_CYCLE_MS)
   397→  // never keep the process alive just for the supervisor
   398→  if (typeof timer === 'object' && timer !== null && 'unref' in timer) {
   399→    ;(timer as { unref: () => void }).unref()
   400→  }
   401→}
   402→
   403→async function runOcoCycle(userId: string, mode: TradeMode): Promise<void> {
   404→  try {
   405→    const [ordRes, posRes] = await Promise.all([
   406→      tradeFetch<TradingOrder[]>(userId, mode, '/v2/orders?status=all&limit=500&direction=desc'),
   407→      tradeFetch<TradingPosition[]>(userId, mode, '/v2/positions'),
   408→    ])
   409→    const orders: TradingOrder[] = Array.isArray(ordRes.data) ? ordRes.data : []
   410→    const positions: TradingPosition[] = Array.isArray(posRes.data) ? posRes.data : []
   411→    const norm = (s: string) => s.replace('/', '').toUpperCase()
   412→    const posSet = new Set(positions.map((p) => norm(p.symbol)))
   413→    const knownCoids = new Set(orders.map((o) => o.client_order_id || ''))
   414→    const isCrypto = (o: TradingOrder) => o.asset_class === 'crypto' || o.symbol.includes('/')
   415→
   416→    // (a) OCO cleanup: a managed exit leg dies with its position
   417→    for (const o of orders) {
   418→      if (!isCrypto(o) || !OCO_OPEN_STATUSES.includes(o.status)) continue
   419→      if (!(o.client_order_id || '').startsWith('tte-')) continue
   420→      if (!posSet.has(norm(o.symbol))) {
   421→        const r = await cancelOrder(userId, mode, o.id)
   422→        console.log(`[oco] canceled orphaned exit ${o.client_order_id} (${o.symbol}) -> ${r.status}`)
   423→      }
   424→    }
   425→
   426→    // (b) attach exit legs to managed entries that have FILLED
   427→    for (const o of orders) {
   428→      if (!isCrypto(o) || o.status !== 'filled') continue
   429→      const plan = parseCryptoPlan(o.client_order_id)
   430→      if (!plan || (plan.tp == null && plan.sl == null)) continue
   431→      const wantTp = plan.tp != null && !knownCoids.has(exitLegClientId(o.id, 'tp'))
   432→      const wantSl = plan.sl != null && !knownCoids.has(exitLegClientId(o.id, 'sl'))
   433→      if (!wantTp && !wantSl) continue
   434→      const errs = await placeCryptoExitLegs(
   435→        userId,
   436→        mode,
   437→        o,
   438→        wantTp ? plan.tp! : undefined,
   439→        wantSl ? plan.sl! : undefined,
   440→      )
   441→      if (errs.length) console.warn(`[oco] exit attach failed for ${o.id} (${o.symbol}):`, errs.join('; '))
   442→      else console.log(`[oco] attached exits to filled entry ${o.id} (${o.symbol}) tp=${plan.tp} sl=${plan.sl}`)
   443→    }
   444→  } catch (e) {
   445→    console.warn('[oco] supervisor cycle failed:', e instanceof Error ? e.message : e)
   446→  }
   447→}

   468→/** Rebuild an open exit order body from an existing order (used by the
   469→ * cancel+re-place fallback and OCO-sibling regeneration). */
   470→function exitBodyFrom(
   471→  orig: TradingOrder,
   472→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number },
   473→): Record<string, unknown> | null {
   474→  const num = (v: string | null | undefined): number | null => {
   475→    if (v == null) return null
   476→    const n = parseFloat(v)
   477→    return Number.isFinite(n) && n > 0 ? n : null
   478→  }
   479→  const newStop = patch.stop_price ?? num(orig.stop_price)
   480→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   481→  const newTrail = patch.trail_percent ?? num(orig.trail_percent)
   482→  const body: Record<string, unknown> = {
   483→    symbol: orig.symbol,
   484→    qty: orig.qty,
   485→    side: orig.side,
   486→    type: orig.type,
   487→    time_in_force: orig.time_in_force || 'gtc',
   488→  }
   489→  if (orig.type === 'stop') {
   490→    if (newStop == null) return null
   491→    body.stop_price = String(newStop)
   492→  } else if (orig.type === 'stop_limit') {
   493→    if (newStop == null) return null
   494→    body.stop_price = String(newStop)
   495→    if (newLimit != null) {
   496→      body.limit_price = String(newLimit)
   497→    } else {
   498→      // preserve the original stop→limit offset when only the stop moves
   499→      const os = num(orig.stop_price)
   500→      const ol = num(orig.limit_price)
   501→      const ratio = os && ol ? ol / os : orig.side === 'sell' ? 0.999 : 1.001
   502→      body.limit_price = String(+(newStop * ratio).toFixed(6))
   503→    }
   504→  } else if (orig.type === 'limit') {
   505→    if (newLimit == null) return null
   506→    body.limit_price = String(newLimit)
   507→  } else if (orig.type === 'trailing_stop') {
   508→    if (newTrail == null) return null
   509→    body.trail_percent = String(newTrail)
   510→  } else {
   511→    return null // market / unknown — never re-placed blindly
   512→  }
   513→  return body
   514→}

   517→  data: TradingOrder | null
   518→  status: number
   519→  error?: string
   520→  mode?: TradeMode
   521→  method?: 'replace' | 'cancel-replace'
   522→  note?: string
   523→}

   525→/**
   526→ * Replace SL/TP prices on an open exit order — the chart line-drag path.
   527→ *
   528→ * Verified against the paper API (2026-10):
   529→ * - A PATCH on a STANDALONE exit (crypto split exits) replaces cleanly:
   530→ *   type/time_in_force/sibling prices are preserved and a NEW order id is
   531→ *   returned.
   532→ * - The chart drag used to 404 when the order lived on the OTHER host: the
   533→ *   client did not send `mode`, so the PATCH went to the stored trade-mode
   534→ *   preference — paper order ids do not exist on live and vice versa, the
   535→ *   update silently never landed on Alpaca.
   536→ * - Bracket OCO legs may refuse direct replacement; cancel+re-place is the
   537→ *   universal fallback (canceling one OCO leg can cascade-cancel the sibling,
   538→ *   which we detect and regenerate).
   539→ */
   540→export async function replaceExitOrderSmart(
   541→  userId: string,
   542→  mode: TradeMode,
   543→  orderId: string,
   544→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number },
   545→): Promise<ReplaceExitResult> {
   546→  const patchBody: Record<string, unknown> = {}
   547→  if (patch.stop_price != null) patchBody.stop_price = String(patch.stop_price)
   548→  if (patch.limit_price != null) patchBody.limit_price = String(patch.limit_price)
   549→  if (patch.trail_percent != null) patchBody.trail_percent = String(patch.trail_percent)
   550→
   551→  // 1) PATCH on the requested host; 2) retry the other host on 404 —
   552→  // both hosts share the same key pair, so a wrong-host id self-heals here.
   553→  const order: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   554→  let last: TradeFetchResult<TradingOrder> | null = null
   555→  let errHost: TradeMode = mode
   556→  for (const m of order) {
   557→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`, {
   558→      method: 'PATCH',
   559→      body: JSON.stringify(patchBody),
   560→    })
   561→    if (r.data && (r.data as TradingOrder).id) {
   562→      return { data: r.data, status: r.status, mode: m, method: 'replace' }
   563→    }
   564→    last = r
   565→    // 422 "order parameters are not changed" — the order ALREADY carries the
   566→    // requested value (e.g. a second drag to the same level). That IS the
   567→    // desired end state: return the current order instead of churning a
   568→    // cancel+re-place cycle.
   569→    if (r.status === 422 && String(r.error || '').includes('not changed')) {
   570→      const cur = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   571→      if (cur.data) {
   572→        return { data: cur.data, status: 200, mode: m, method: 'replace', note: 'Order already at this value' }
   573→      }
   574→    }
   575→    if (r.status !== 404) {
   576→      // real rejection — the order lives on THIS host; run the fallback there
   577→      errHost = m
   578→      break
   579→    }
   580→  }
   581→
   582→  // 3) The id may have JUST been replaced (fast double-drag: Alpaca's
   583→  // cancel/replace is async and 404s for a beat) — follow the replaced_by
   584→  // chain and retry the direct PATCH on the live successor first.
   585→  const OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   586→  let orig: TradingOrder | null = null
   587→  let targetId = orderId
   588→  for (let hop = 0; hop < 4; hop++) {
   589→    const o = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`)
   590→    if (!o.data) break
   591→    const od = o.data
   592→    if (OPEN_STATUSES.includes(String(od.status))) {
   593→      orig = od
   594→      if (targetId !== orderId) {
   595→        // the open successor exists — retry the direct PATCH on it
   596→        const retry = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`, {
   597→          method: 'PATCH',
   598→          body: JSON.stringify(patchBody),
   599→        })
   600→        if (retry.data && (retry.data as TradingOrder).id) {
   601→          return { data: retry.data, status: 200, mode: errHost, method: 'replace', note: 'Applied to the replacement order' }
   602→        }
   603→        if (retry.status === 422 && String(retry.error || '').includes('not changed')) {
   604→          return { data: od, status: 200, mode: errHost, method: 'replace', note: 'Order already at this value' }
   605→        }
   606→        last = retry
   607→      }
   608→      break
   609→    }
   610→    const rb = od.replaced_by
   611→    if (typeof rb === 'string' && rb && rb !== targetId) {
   612→      targetId = rb
   613→      continue
   614→    }
   615→    orig = od // terminal status without a successor
   616→    break
   617→  }
   618→  if (!orig) {
   619→    return {
   620→      data: null,
   621→      status: last?.status || 502,
   622→      error: last?.error || 'Order not found on paper or live',
   623→      mode: errHost,
   624→      method: 'cancel-replace',
   625→    }
   626→  }
   627→  if (!OPEN_STATUSES.includes(String(orig.status))) {
   628→    return {
   629→      data: null,
   630→      status: 409,
   631→      error: `Order is ${orig.status} on ${errHost} — nothing to replace (refresh the chart)`,
   632→      mode: errHost,
   633→      method: 'cancel-replace',
   634→    }
   635→  }
   636→
   637→  // snapshot the OCO sibling BEFORE cancelling (limit TP vs stop-ish SL of
   638→  // the same symbol + same side) so a cascade-cancel can be healed
   639→  const openRes = await tradeFetch<TradingOrder[]>(userId, errHost, "/v2/orders?status=open&limit=500")
   640→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   641→  const isStopish = (t: string) => t === 'stop' || t === 'stop_limit' || t === 'trailing_stop'
   642→  const sibling = (openRes.data || []).find(
   643→    (o) =>
   644→      o.id !== orderId &&
   645→      norm(o.symbol) === norm(orig.symbol) &&
   646→      o.side === orig.side &&
   647→      ((isStopish(orig.type) && o.type === 'limit') || (orig.type === 'limit' && isStopish(o.type))),
   648→  )
   649→
   650→  const cancelRes = await cancelOrder(userId, mode, targetId)
   651→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
   652→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode, method: 'cancel-replace' }
   653→  }
   654→
   655→  const newBody = exitBodyFrom(orig, patch)
   656→  if (!newBody) {
   657→    return { data: null, status: 400, error: `Unsupported exit type ${orig.type} for re-place`, mode, method: 'cancel-replace' }
   658→  }
   659→  const r2 = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   660→    method: 'POST',
   661→    body: JSON.stringify(newBody),
   662→  })
   663→  if (!(r2.data && (r2.data as TradingOrder).id)) {
   664→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode, method: 'cancel-replace' }
   665→  }
   666→
   667→  // heal the sibling if the cancel cascaded (OCO group cancellation)
   668→  let note: string | undefined
   669→  if (sibling) {
   670→    const stillOpen = await tradeFetch<TradingOrder>(userId, mode, `/v2/orders/${sibling.id}`)
   671→    const gone = !stillOpen.data || ['canceled', 'cancelled', 'expired', 'rejected'].includes(String(stillOpen.data.status))
   672→    if (gone) {
   673→      const sibBody = exitBodyFrom(sibling, {})
   674→      if (sibBody) {
   675→        const rs = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   676→          method: 'POST',
   677→          body: JSON.stringify(sibBody),
   678→        })
   679→        note = rs.data ? 'OCO sibling re-created' : `Warning: sibling exit lost (${rs.error || 're-place failed'})`
   680→      }
   681→    }
   682→  }
   683→  return { data: r2.data, status: r2.status, mode, method: 'cancel-replace', note }
   684→}
   685→
   686→/**
   687→ * Market-close a position. Crypto balances are reserved per open exit order,
   688→ * so lingering SL/TP legs would starve the liquidation (Alpaca sells only
   689→ * qty_available) — cancel the symbol's opposite-side (exit) orders FIRST,
   690→ * then close. Returns the close order and how many exits were cancelled.
   691→ */
   692→export async function closePositionFull(
   693→  userId: string,
   694→  mode: TradeMode,
   695→  symbol: string,
   696→): Promise<{ close: TradeFetchResult<TradingOrder>; canceledExits: number }> {
   697→  const tSym = tradeSymbol(symbol)
   698→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   699→  // snapshot the position side BEFORE closing (long -> sell exits, short -> buy)
   700→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
   701→  const posSide = posRes.data?.side
   702→  let canceledExits = 0
   703→  // fetch ALL open orders and filter here — Alpaca's `symbols=` query wants
   704→  // the slash form for crypto (BTC/USD) while callers pass either form
   705→  const openRes = await getOpenOrders(userId, mode)
   706→  const orders: TradingOrder[] = (Array.isArray(openRes.data) ? openRes.data : [])
   707→    .filter((o) => norm(o.symbol) === norm(tSym))
   708→  const exitSide = posSide === 'long' ? 'sell' : posSide === 'short' ? 'buy' : null
   709→  for (const o of orders) {
   710→    if (exitSide && o.side !== exitSide) continue // keep same-side (entry) orders
   711→    if (exitSide == null && o.type === 'market') continue // no position info: keep entries
   712→    const c = await cancelOrder(userId, mode, o.id)
   713→    if (c.status === 204 || c.status === 200) canceledExits += 1
   714→  }
   715→  const close = await closePosition(userId, mode, symbol)
   716→  return { close, canceledExits }
   717→}

   720→  return tradeFetch<null>(userId, mode, `/v2/orders/${orderId}`, { method: 'DELETE' })
   721→}

   725→    method: 'DELETE',
   726→  })
