     1→// Alpaca Trading API client (server-side only): paper + live order routing,
     2→// positions, bracket/OCO exits, order replacement for SL/TP drag updates.
     3→//
     4→// Verified against the paper API (2026-10):
     5→// - Stocks support order_class=bracket (entry + take_profit/stop_loss legs).
     6→// - Crypto rejects ALL advanced order classes (order_class=oco -> 40010001
     7→//   "invalid order type for crypto order"; bracket -> 42210000) AND plain
     8→//   `stop` type. Sell exits also CANNOT be reserved against a PENDING buy
     9→//   (40310000 "insufficient balance" — only settled position qty counts),
    10→//   so crypto complex orders are EMULATED: the entry carries its intended
    11→//   TP/SL in the client_order_id (`ttb-…`) and the OCO supervisor below
    12→//   attaches the exit legs (`tte-…`) the moment the entry FILLS, canceling
    13→//   any surviving leg once the position is gone.
    14→// - PATCH /v2/orders/{id} creates a REPLACEMENT order with a NEW id.
    15→// - Position symbols drop the crypto slash (BTC/USD -> BTCUSD); ORDER
    16→//   symbols keep it (BTC/USD).
    17→import { db } from './db'
    18→import { getAlpacaKeys } from './alpaca'

    40→export async function resolveTradeMode(userId: string | undefined, requested?: unknown): Promise<TradeMode> {
    41→  if (isTradeMode(requested)) return requested
    42→  if (userId) {
    43→    const user = await db.user.findUnique({ where: { id: userId } })

    55→interface TradeFetchResult<T> {
    56→  data: T | null
    57→  status: number
    58→  error?: string
    59→}
    60→
    61→async function tradeFetch<T>(
    62→  userId: string,
    63→  mode: TradeMode,
    64→  path: string,
    65→  init: RequestInit = {},
    66→): Promise<TradeFetchResult<T>> {
    67→  const keys = await getAlpacaKeys(userId)
    68→  if (!keys) return { data: null, status: 0, error: 'no-keys' }
    69→  const controller = new AbortController()
    70→  const timer = setTimeout(() => controller.abort(), 9000)
    71→  try {
    72→    const res = await fetch(`${TRADE_BASE[mode]}${path}`, {
    73→      ...init,
    74→      headers: {
    75→        'APCA-API-KEY-ID': keys.keyId,
    76→        'APCA-API-SECRET-KEY': keys.secret,
    77→        accept: 'application/json',
    78→        ...(init.body ? { 'content-type': 'application/json' } : {}),
    79→        ...(init.headers || {}),
    80→      },
    81→      signal: controller.signal,
    82→      cache: 'no-store',
    83→    })
    84→    const text = await res.text()
    85→    let parsed: unknown = null
    86→    try {
    87→      parsed = text ? JSON.parse(text) : null
    88→    } catch {
    89→      parsed = null
    90→    }
    91→    if (!res.ok) {
    92→      const msg =
    93→        (parsed as { message?: string } | null)?.message ||
    94→        `Trading API HTTP ${res.status}`
    95→      console.error(`Trading ${mode} ${path} -> ${res.status}: ${msg}`)
    96→      return { data: parsed as T, status: res.status, error: msg }
    97→    }
    98→    return { data: parsed as T, status: res.status }
    99→  } catch (e) {
   100→    console.error(`Trading ${mode} ${path} failed:`, e instanceof Error ? e.message : e)
   101→    return { data: null, status: 0, error: e instanceof Error ? e.message : 'network error' }
   102→  } finally {
   103→    clearTimeout(timer)
   104→  }
   105→}

   109→/**
   110→ * Alpaca rejects prices with more than 9 decimal places ("limit price exceeds
   111→ * maximum precision of 9 decimal places") on CRYPTO, and sub-penny increments
   112→ * on STOCKS ("sub-penny increment does not fulfill minimum pricing criteria").
   113→ * Chart drag prices are raw JS floats (65800.22400000001) that blow past both —
   114→ * first the direct PATCH fails, then the cancel+re-place fallback fails too,
   115→ * LEAVING THE EXIT ORDER CANCELED AND LOST. Round every outgoing price string:
   116→ * crypto 6dp (>=1) / 8dp (<1), stocks penny 2dp (>=1) / 4dp (<1).
   117→ */
   118→export function fmtPx(v: number, crypto = false): string {
   119→  if (!Number.isFinite(v) || v <= 0) return String(v)
   120→  const dp = crypto ? (v >= 1 ? 6 : 8) : v >= 1 ? 2 : 4
   121→  return String(parseFloat(v.toFixed(dp)))
   122→}

   203→/**
   204→ * Place an entry order with optional exits.
   205→ * Stocks: single bracket request when TP/SL present (Alpaca attaches legs).
   206→ * Crypto: advanced classes are rejected — the entry is placed ALONE carrying
   207→ * the intended TP/SL in its client_order_id (`ttb-…`); the OCO supervisor
   208→ * (ensureCryptoOcoSupervisor) attaches the exit legs when the entry fills.
   209→ */
   210→export async function placeOrderWithExits(
   211→  userId: string,
   212→  mode: TradeMode,
   213→  body: PlaceOrderBody,
   214→): Promise<{ results: TradingOrder[]; errors: string[]; splitNote?: string }> {
   215→  const isCrypto = body.symbol.includes('/')
   216→  const type = body.type || 'market'
   217→  const tif = body.time_in_force || (isCrypto ? 'gtc' : type === 'market' ? 'day' : 'gtc')
   218→  const entry: Record<string, unknown> = {
   219→    symbol: tradeSymbol(body.symbol),
   220→    qty: String(body.qty),
   221→    side: body.side,
   222→    type,
   223→    time_in_force: tif,
   224→  }
   225→  if (type === 'limit') {
   226→    if (!body.limit_price) return { results: [], errors: ['Limit price required for limit orders'] }
   227→    entry.limit_price = String(body.limit_price)
   228→  }
   229→  if (body.client_order_id) entry.client_order_id = body.client_order_id
   230→
   231→  const errors: string[] = []
   232→  const results: TradingOrder[] = []
   233→
   234→  if (!isCrypto) {
   235→    // Stocks: bracket handles entry + legs atomically.
   236→    const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   237→    const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   238→    if (hasTp || hasSl) {
   239→      entry.order_class = 'bracket'
   240→      if (hasTp) entry.take_profit = { limit_price: body.take_profit!.toFixed(2) }
   241→      if (hasSl) entry.stop_loss = { stop_price: body.stop_loss!.toFixed(2) }
   242→      // encode the plan for the chart's pending display: Alpaca materializes
   243→      // bracket LEGS only when the entry activates, so a queued (weekend /
   244→      // pre-market) bracket would otherwise render without its TP/SL lines
   245→      entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   246→    }
   247→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   248→      method: 'POST',
   249→      body: JSON.stringify(entry),
   250→    })
   251→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   252→    else errors.push(r.error || 'Entry order failed')
   253→    return { results, errors }
   254→  }
   255→
   256→  // ---- Crypto: complex orders are EMULATED (no native bracket/OCO) ----
   257→  const hasTp = typeof body.take_profit === 'number' && body.take_profit > 0
   258→  const hasSl = typeof body.stop_loss === 'number' && body.stop_loss > 0
   259→  if (hasTp || hasSl) {
   260→    // encode the planned exits so the chart can render them while the entry
   261→    // is pending AND the supervisor can attach the legs without extra state
   262→    entry.client_order_id = encodePlanClientId(mode, hasTp ? body.take_profit! : null, hasSl ? body.stop_loss! : null)
   263→  }
   264→  const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   265→    method: 'POST',
   266→    body: JSON.stringify(entry),
   267→  })
   268→  if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   269→  else {
   270→    errors.push(r.error || 'Entry order failed')
   271→    return { results, errors }
   272→  }
   273→  if (!hasTp && !hasSl) return { results, errors }
   274→
   275→  // Fast path: a MARKET entry usually fills instantly — poll briefly and
   276→  // attach the exits in-line so protection starts immediately. Limit entries
   277→  // return right away (the supervisor attaches on fill within ~6s).
   278→  if (type === 'market') {
   279→    let cur: TradingOrder | null = r.data
   280→    for (let i = 0; i < 4 && cur && cur.status !== 'filled'; i++) {
   281→      await new Promise((res) => setTimeout(res, 600))
   282→      const s = await tradeFetch<TradingOrder>(userId, mode, `/v2/orders/${(r.data as TradingOrder).id}`)
   283→      cur = s.data ?? cur
   284→      if (cur && ['canceled', 'cancelled', 'rejected', 'expired'].includes(cur.status)) break
   285→    }
   286→    if (cur?.status === 'filled') {
   287→      const legErrs = await placeCryptoExitLegs(
   288→        userId,
   289→        mode,
   290→        cur,
   291→        hasTp ? body.take_profit : undefined,
   292→        hasSl ? body.stop_loss : undefined,
   293→      )
   294→      errors.push(...legErrs)
   295→      return { results, errors }
   296→    }
   297→  }
   298→  return {
   299→    results,
   300→    errors,
   301→    splitNote: 'TP/SL attach automatically when the entry fills (OCO-managed)',
   302→  }
   303→}

   307→/** Price -> client_order_id-safe digits (6dp fixed point). */
   308→function encP(v: number): string {
   309→  return String(Math.round(v * 1e6))
   310→}
   311→
   312→/** client_order_id prefix identifying a MANAGED crypto exit leg. */
   313→function exitLegClientId(entryOrderId: string, kind: 'tp' | 'sl'): string {
   314→  return `tte-${entryOrderId}-${kind}`
   315→}
   316→
   317→/** Encode the OCO/bracket plan (mode + TP/SL prices) into the entry's client_order_id. */
   318→function encodePlanClientId(mode: TradeMode, tp: number | null, sl: number | null): string {
   319→  const rand = Math.random().toString(36).slice(2, 8)
   320→  return `ttb-${mode[0]}-tp${tp != null && tp > 0 ? encP(tp) : '0'}-sl${sl != null && sl > 0 ? encP(sl) : '0'}-${rand}`
   321→}
   322→
   323→/** Decode the plan from a `ttb-…` client_order_id. Returns null when not ours. */
   324→function parseCryptoPlan(clientOrderId: string | undefined | null): { tp: number | null; sl: number | null } | null {
   325→  const m = /^ttb-[a-z](?:-tp(\d+))?(?:-sl(\d+))?-/.exec(clientOrderId || '')
   326→  if (!m) return null
   327→  const dec = (s: string | undefined) => (s ? parseInt(s, 10) / 1e6 : 0)
   328→  const tp = dec(m[1])
   329→  const sl = dec(m[2])
   330→  return { tp: tp > 0 ? tp : null, sl: sl > 0 ? sl : null }
   331→}
   332→
   333→/**
   334→ * Place the exit legs for a FILLED crypto entry (50/50 qty split when both
   335→ * legs are requested — Alpaca reserves balance per open sell order). Leg ids
   336→ * are deterministic (`tte-<entryId>-tp|-sl`) so the supervisor is idempotent.
   337→ * Crypto buys receive qty minus a trading fee, so legs apply a 0.99 safety
   338→ * factor to avoid 403 "insufficient balance".
   339→ */
   340→async function placeCryptoExitLegs(
   341→  userId: string,
   342→  mode: TradeMode,
   343→  entry: TradingOrder,
   344→  tp?: number,
   345→  sl?: number,
   346→): Promise<string[]> {
   347→  const errors: string[] = []
   348→  const filledQty = parseFloat(entry.filled_qty) || parseFloat(entry.qty) || 0
   349→  if (!(filledQty > 0)) return ['entry has no filled qty']
   350→  const exitSide = entry.side === 'buy' ? 'sell' : 'buy'
   351→  const both = tp != null && sl != null
   352→  const legQty = ((both ? filledQty / 2 : filledQty) * 0.99).toFixed(8).replace(/0+$/, '').replace(/\.$/, '')
   353→  if (tp != null) {
   354→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   355→      method: 'POST',
   356→      body: JSON.stringify({
   357→        symbol: entry.symbol,
   358→        qty: legQty,
   359→        side: exitSide,
   360→        type: 'limit',
   361→        time_in_force: 'gtc',
   362→        limit_price: tp.toFixed(tp < 1 ? 6 : 2),
   363→        client_order_id: exitLegClientId(entry.id, 'tp'),
   364→      }),
   365→    })
   366→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`TP: ${r.error || 'failed'}`)
   367→  }
   368→  if (sl != null) {
   369→    // stop-limit sell: limit slightly below the stop so it clears the book
   370→    const limP = exitSide === 'sell' ? sl * 0.999 : sl * 1.001
   371→    const r = await tradeFetch<TradingOrder>(userId, mode, '/v2/orders', {
   372→      method: 'POST',
   373→      body: JSON.stringify({
   374→        symbol: entry.symbol,
   375→        qty: legQty,
   376→        side: exitSide,
   377→        type: 'stop_limit',
   378→        time_in_force: 'gtc',
   379→        stop_price: sl.toFixed(sl < 1 ? 6 : 2),
   380→        limit_price: limP.toFixed(limP < 1 ? 6 : 2),
   381→        client_order_id: exitLegClientId(entry.id, 'sl'),
   382→      }),
   383→    })
   384→    if (!(r.data && (r.data as TradingOrder).id)) errors.push(`SL: ${r.error || 'failed'}`)
   385→  }
   386→  return errors
   387→}

   389→const OCO_CYCLE_MS = 6000
   390→const OCO_OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   391→
   392→/** Supervised (user,mode) pairs — survives dev hot reloads via globalThis. */
   393→function supervisedSet(): Set<string> {
   394→  const g = globalThis as { __ttOcoSupervised?: Set<string> }
   395→  if (!g.__ttOcoSupervised) g.__ttOcoSupervised = new Set<string>()
   396→  return g.__ttOcoSupervised
   397→}

   399→/**
   400→ * Start the OCO supervisor for a (user, mode) pair (idempotent). One cycle
   401→ * every 6s: (a) cancel managed exit legs whose position is gone (OCO
   402→ * semantics — when the position closes, the surviving leg dies),
   403→ * (b) attach exit legs to managed entries that have FILLED.
   404→ */
   405→export function ensureCryptoOcoSupervisor(userId: string, mode: TradeMode): void {
   406→  const key = `${userId}|${mode}`
   407→  if (supervisedSet().has(key)) return
   408→  supervisedSet().add(key)
   409→  const timer = setInterval(() => {
   410→    void runOcoCycle(userId, mode)
   411→  }, OCO_CYCLE_MS)
   412→  // never keep the process alive just for the supervisor
   413→  if (typeof timer === 'object' && timer !== null && 'unref' in timer) {
   414→    ;(timer as { unref: () => void }).unref()
   415→  }
   416→}
   417→
   418→async function runOcoCycle(userId: string, mode: TradeMode): Promise<void> {
   419→  try {
   420→    const [ordRes, posRes] = await Promise.all([
   421→      tradeFetch<TradingOrder[]>(userId, mode, '/v2/orders?status=all&limit=500&direction=desc'),
   422→      tradeFetch<TradingPosition[]>(userId, mode, '/v2/positions'),
   423→    ])
   424→    const orders: TradingOrder[] = Array.isArray(ordRes.data) ? ordRes.data : []
   425→    const positions: TradingPosition[] = Array.isArray(posRes.data) ? posRes.data : []
   426→    const norm = (s: string) => s.replace('/', '').toUpperCase()
   427→    const posSet = new Set(positions.map((p) => norm(p.symbol)))
   428→    const knownCoids = new Set(orders.map((o) => o.client_order_id || ''))
   429→    const isCrypto = (o: TradingOrder) => o.asset_class === 'crypto' || o.symbol.includes('/')
   430→
   431→    // (a) OCO cleanup: a managed exit leg dies with its position
   432→    for (const o of orders) {
   433→      if (!isCrypto(o) || !OCO_OPEN_STATUSES.includes(o.status)) continue
   434→      if (!(o.client_order_id || '').startsWith('tte-')) continue
   435→      if (!posSet.has(norm(o.symbol))) {
   436→        const r = await cancelOrder(userId, mode, o.id)
   437→        console.log(`[oco] canceled orphaned exit ${o.client_order_id} (${o.symbol}) -> ${r.status}`)
   438→      }
   439→    }
   440→
   441→    // (b) attach exit legs to managed entries that have FILLED
   442→    for (const o of orders) {
   443→      if (!isCrypto(o) || o.status !== 'filled') continue
   444→      const plan = parseCryptoPlan(o.client_order_id)
   445→      if (!plan || (plan.tp == null && plan.sl == null)) continue
   446→      const wantTp = plan.tp != null && !knownCoids.has(exitLegClientId(o.id, 'tp'))
   447→      const wantSl = plan.sl != null && !knownCoids.has(exitLegClientId(o.id, 'sl'))
   448→      if (!wantTp && !wantSl) continue
   449→      const errs = await placeCryptoExitLegs(
   450→        userId,
   451→        mode,
   452→        o,
   453→        wantTp ? plan.tp! : undefined,
   454→        wantSl ? plan.sl! : undefined,
   455→      )
   456→      if (errs.length) console.warn(`[oco] exit attach failed for ${o.id} (${o.symbol}):`, errs.join('; '))
   457→      else console.log(`[oco] attached exits to filled entry ${o.id} (${o.symbol}) tp=${plan.tp} sl=${plan.sl}`)
   458→    }
   459→  } catch (e) {
   460→    console.warn('[oco] supervisor cycle failed:', e instanceof Error ? e.message : e)
   461→  }
   462→}

   483→/** Rebuild an open exit order body from an existing order (used by the
   484→ * cancel+re-place fallback and OCO-sibling regeneration). */
   485→function exitBodyFrom(
   486→  orig: TradingOrder,
   487→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number },
   488→): Record<string, unknown> | null {
   489→  const isCrypto = orig.asset_class === 'crypto' || orig.symbol.includes('/')
   490→  const num = (v: string | null | undefined): number | null => {
   491→    if (v == null) return null
   492→    const n = parseFloat(v)
   493→    return Number.isFinite(n) && n > 0 ? n : null
   494→  }
   495→  const newStop = patch.stop_price ?? num(orig.stop_price)
   496→  const newLimit = patch.limit_price ?? num(orig.limit_price)
   497→  const newTrail = patch.trail_percent ?? num(orig.trail_percent)
   498→  const body: Record<string, unknown> = {
   499→    symbol: orig.symbol,
   500→    qty: orig.qty,
   501→    side: orig.side,
   502→    type: orig.type,
   503→    time_in_force: orig.time_in_force || 'gtc',
   504→  }
   505→  if (orig.type === 'stop') {
   506→    if (newStop == null) return null
   507→    body.stop_price = fmtPx(newStop, isCrypto)
   508→  } else if (orig.type === 'stop_limit') {
   509→    if (newStop == null) return null
   510→    body.stop_price = fmtPx(newStop, isCrypto)
   511→    if (newLimit != null) {
   512→      body.limit_price = fmtPx(newLimit, isCrypto)
   513→    } else {
   514→      // preserve the original stop→limit offset when only the stop moves
   515→      const os = num(orig.stop_price)
   516→      const ol = num(orig.limit_price)
   517→      const ratio = os && ol ? ol / os : orig.side === 'sell' ? 0.999 : 1.001
   518→      body.limit_price = fmtPx(newStop * ratio, isCrypto)
   519→    }
   520→  } else if (orig.type === 'limit') {
   521→    if (newLimit == null) return null
   522→    body.limit_price = fmtPx(newLimit, isCrypto)
   523→  } else if (orig.type === 'trailing_stop') {
   524→    if (newTrail == null) return null
   525→    body.trail_percent = String(newTrail)
   526→  } else {
   527→    return null // market / unknown — never re-placed blindly
   528→  }
   529→  return body
   530→}

   538→  note?: string

   541→/**
   542→ * Replace SL/TP prices on an open exit order — the chart line-drag path.
   543→ *
   544→ * Verified against the paper API (2026-10):
   545→ * - A PATCH on a STANDALONE exit (crypto split exits) replaces cleanly:
   546→ *   type/time_in_force/sibling prices are preserved and a NEW order id is
   547→ *   returned.
   548→ * - The chart drag used to 404 when the order lived on the OTHER host: the
   549→ *   client did not send `mode`, so the PATCH went to the stored trade-mode
   550→ *   preference — paper order ids do not exist on live and vice versa, the
   551→ *   update silently never landed on Alpaca.
   552→ * - Bracket OCO legs may refuse direct replacement; cancel+re-place is the
   553→ *   universal fallback (canceling one OCO leg can cascade-cancel the sibling,
   554→ *   which we detect and regenerate).
   555→ */
   556→export async function replaceExitOrderSmart(
   557→  userId: string,
   558→  mode: TradeMode,
   559→  orderId: string,
   560→  patch: { stop_price?: number; limit_price?: number; trail_percent?: number; crypto?: boolean },
   561→): Promise<ReplaceExitResult> {
   562→  const patchBody: Record<string, unknown> = {}
   563→  if (patch.stop_price != null) patchBody.stop_price = fmtPx(patch.stop_price, patch.crypto)
   564→  if (patch.limit_price != null) patchBody.limit_price = fmtPx(patch.limit_price, patch.crypto)
   565→  if (patch.trail_percent != null) patchBody.trail_percent = String(parseFloat(patch.trail_percent.toFixed(4)))
   566→
   567→  // 1) PATCH on the requested host; 2) retry the other host on 404 —
   568→  // both hosts share the same key pair, so a wrong-host id self-heals here.
   569→  const order: TradeMode[] = mode === 'paper' ? ['paper', 'live'] : ['live', 'paper']
   570→  let last: TradeFetchResult<TradingOrder> | null = null
   571→  let errHost: TradeMode = mode
   572→  for (const m of order) {
   573→    const r = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`, {
   574→      method: 'PATCH',
   575→      body: JSON.stringify(patchBody),
   576→    })
   577→    if (r.data && (r.data as TradingOrder).id) {
   578→      return { data: r.data, status: r.status, mode: m, method: 'replace' }
   579→    }
   580→    last = r
   581→    // 422 "order parameters are not changed" — the order ALREADY carries the
   582→    // requested value (e.g. a second drag to the same level). That IS the
   583→    // desired end state: return the current order instead of churning a
   584→    // cancel+re-place cycle.
   585→    if (r.status === 422 && String(r.error || '').includes('not changed')) {
   586→      const cur = await tradeFetch<TradingOrder>(userId, m, `/v2/orders/${orderId}`)
   587→      if (cur.data) {
   588→        return { data: cur.data, status: 200, mode: m, method: 'replace', note: 'Order already at this value' }
   589→      }
   590→    }
   591→    if (r.status !== 404) {
   592→      // real rejection — the order lives on THIS host; run the fallback there
   593→      errHost = m
   594→      break
   595→    }
   596→  }
   597→
   598→  // 3) The id may have JUST been replaced (fast double-drag: Alpaca's
   599→  // cancel/replace is async and 404s for a beat) — follow the replaced_by
   600→  // chain and retry the direct PATCH on the live successor first.
   601→  const OPEN_STATUSES = ['new', 'open', 'accepted', 'pending_new', 'held', 'partially_filled']
   602→  let orig: TradingOrder | null = null
   603→  let targetId = orderId
   604→  for (let hop = 0; hop < 4; hop++) {
   605→    const o = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`)
   606→    if (!o.data) break
   607→    const od = o.data
   608→    if (OPEN_STATUSES.includes(String(od.status))) {
   609→      orig = od
   610→      if (targetId !== orderId) {
   611→        // the open successor exists — retry the direct PATCH on it
   612→        const retry = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${targetId}`, {
   613→          method: 'PATCH',
   614→          body: JSON.stringify(patchBody),
   615→        })
   616→        if (retry.data && (retry.data as TradingOrder).id) {
   617→          return { data: retry.data, status: 200, mode: errHost, method: 'replace', note: 'Applied to the replacement order' }
   618→        }
   619→        if (retry.status === 422 && String(retry.error || '').includes('not changed')) {
   620→          return { data: od, status: 200, mode: errHost, method: 'replace', note: 'Order already at this value' }
   621→        }
   622→        last = retry
   623→      }
   624→      break
   625→    }
   626→    const rb = od.replaced_by
   627→    if (typeof rb === 'string' && rb && rb !== targetId) {
   628→      targetId = rb
   629→      continue
   630→    }
   631→    orig = od // terminal status without a successor
   632→    break
   633→  }
   634→  if (!orig) {
   635→    return {
   636→      data: null,
   637→      status: last?.status || 502,
   638→      error: last?.error || 'Order not found on paper or live',
   639→      mode: errHost,
   640→      method: 'cancel-replace',
   641→    }
   642→  }
   643→  if (!OPEN_STATUSES.includes(String(orig.status))) {
   644→    return {
   645→      data: null,
   646→      status: 409,
   647→      error: `Order is ${orig.status} on ${errHost} — nothing to replace (refresh the chart)`,
   648→      mode: errHost,
   649→      method: 'cancel-replace',
   650→    }
   651→  }
   652→
   653→  // snapshot the OCO sibling BEFORE cancelling (limit TP vs stop-ish SL of
   654→  // the same symbol + same side) so a cascade-cancel can be healed
   655→  const openRes = await tradeFetch<TradingOrder[]>(userId, errHost, "/v2/orders?status=open&limit=500")
   656→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   657→  const isStopish = (t: string) => t === 'stop' || t === 'stop_limit' || t === 'trailing_stop'
   658→  const sibling = (openRes.data || []).find(
   659→    (o) =>
   660→      o.id !== orderId &&
   661→      norm(o.symbol) === norm(orig.symbol) &&
   662→      o.side === orig.side &&
   663→      ((isStopish(orig.type) && o.type === 'limit') || (orig.type === 'limit' && isStopish(o.type))),
   664→  )
   665→
   666→  const cancelRes = await cancelOrder(userId, errHost, targetId)
   667→  if (cancelRes.status !== 204 && cancelRes.status !== 200 && cancelRes.status !== 404) {
   668→    return { data: null, status: cancelRes.status || 502, error: cancelRes.error || 'Cancel failed', mode: errHost, method: 'cancel-replace' }
   669→  }
   670→
   671→  const newBody = exitBodyFrom(orig, patch)
   672→  if (!newBody) {
   673→    return { data: null, status: 400, error: `Unsupported exit type ${orig.type} for re-place`, mode: errHost, method: 'cancel-replace' }
   674→  }
   675→  // re-place on the host the order actually lives on (`mode` can differ when
   676→  // the PATCH self-healed across paper/live)
   677→  const r2 = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   678→    method: 'POST',
   679→    body: JSON.stringify(newBody),
   680→  })
   681→  if (!(r2.data && (r2.data as TradingOrder).id)) {
   682→    return { data: null, status: r2.status || 502, error: `Re-place failed after cancel: ${r2.error || 'unknown'}`, mode: errHost, method: 'cancel-replace' }
   683→  }
   684→
   685→  // heal the sibling if the cancel cascaded (OCO group cancellation)
   686→  let note: string | undefined
   687→  if (sibling) {
   688→    const stillOpen = await tradeFetch<TradingOrder>(userId, errHost, `/v2/orders/${sibling.id}`)
   689→    const gone = !stillOpen.data || ['canceled', 'cancelled', 'expired', 'rejected'].includes(String(stillOpen.data.status))
   690→    if (gone) {
   691→      const sibBody = exitBodyFrom(sibling, {})
   692→      if (sibBody) {
   693→        const rs = await tradeFetch<TradingOrder>(userId, errHost, '/v2/orders', {
   694→          method: 'POST',
   695→          body: JSON.stringify(sibBody),
   696→        })
   697→        note = rs.data ? 'OCO sibling re-created' : `Warning: sibling exit lost (${rs.error || 're-place failed'})`
   698→      }
   699→    }
   700→  }
   701→  return { data: r2.data, status: r2.status, mode: errHost, method: 'cancel-replace', note }
   702→}
   703→
   704→/**
   705→ * Market-close a position. Crypto balances are reserved per open exit order,
   706→ * so lingering SL/TP legs would starve the liquidation (Alpaca sells only
   707→ * qty_available) — cancel the symbol's opposite-side (exit) orders FIRST,
   708→ * then close. Returns the close order and how many exits were cancelled.
   709→ */
   710→export async function closePositionFull(
   711→  userId: string,
   712→  mode: TradeMode,
   713→  symbol: string,
   714→): Promise<{ close: TradeFetchResult<TradingOrder>; canceledExits: number }> {
   715→  const tSym = tradeSymbol(symbol)
   716→  const norm = (s: string) => s.replace('/', '').toUpperCase()
   717→  // snapshot the position side BEFORE closing (long -> sell exits, short -> buy)
   718→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
   719→  const posSide = posRes.data?.side
   720→  let canceledExits = 0
   721→  // fetch ALL open orders and filter here — Alpaca's `symbols=` query wants
   722→  // the slash form for crypto (BTC/USD) while callers pass either form
   723→  const openRes = await getOpenOrders(userId, mode)
   724→  const orders: TradingOrder[] = (Array.isArray(openRes.data) ? openRes.data : [])
   725→    .filter((o) => norm(o.symbol) === norm(tSym))
   726→  const exitSide = posSide === 'long' ? 'sell' : posSide === 'short' ? 'buy' : null
   727→  for (const o of orders) {
   728→    if (exitSide && o.side !== exitSide) continue // keep same-side (entry) orders
   729→    if (exitSide == null && o.type === 'market') continue // no position info: keep entries
   730→    const c = await cancelOrder(userId, mode, o.id)
   731→    if (c.status === 204 || c.status === 200) canceledExits += 1
   732→  }
   733→  const close = await closePosition(userId, mode, symbol)
   734→  return { close, canceledExits }
   735→}

   741→// ---------- attach TP/SL to an EXISTING position (chart [TP]/[SL] drag-add) ----------
   742→
   743→export interface AttachExitsResult {
   744→  results: TradingOrder[]
   745→  errors: string[]
   746→  oco?: boolean
   747→}
   748→
   749→/**
   750→ * Attach a TP and/or SL to an open position — the chart's "+TP"/"+SL" chip
   751→ * drag and the Positions-tab add buttons land here.
   752→ *
   753→ * Stocks (verified against the paper API 2026-10):
   754→ * - order_class=oco REQUIRES take_profit.limit_price AND stop_loss.stop_price
   755→ *   AND an existing position to exit (without one: "oco orders must be exit
   756→ *   orders"; parent must be type=limit). With both legs requested we send the
   757→ *   documented OCO exit body and FALL BACK to separate legs on rejection.
   758→ * - A single leg is a plain limit (TP) / stop (SL) order on the full position.
   759→ * Crypto: ALL advanced order classes are rejected — place split exit legs
   760→ * (50/50 qty, 0.99 fee factor) directly against the position, tagged with a
   761→ * `ttp-` client_order_id (ignored by the OCO supervisor's `tte-` cleanup).
   762→ */
   763→export async function attachExitsToPosition(
   764→  userId: string,
   765→  mode: TradeMode,
   766→  symbol: string,
   767→  exits: { tp?: number; sl?: number },
   768→): Promise<AttachExitsResult> {
   769→  const tSym = tradeSymbol(symbol)
   770→  const hasTp = typeof exits.tp === 'number' && exits.tp > 0
   771→  const hasSl = typeof exits.sl === 'number' && exits.sl > 0
   772→  if (!hasTp && !hasSl) return { results: [], errors: ['no TP/SL price given'] }
   773→
   774→  const posRes = await tradeFetch<TradingPosition>(userId, mode, `/v2/positions/${encodeURIComponent(tSym)}`)
   775→  if (!posRes.data) {
   776→    return { results: [], errors: [posRes.error && posRes.error !== 'not found' ? posRes.error : 'Position not found'] }
   777→  }
   778→  const pos = posRes.data
   779→  // crypto-ness comes from the POSITION (callers pass either symbol form —
   780→  // the chart sends the trade symbol BTCUSD, whose crypto-ness the route
   781→  // cannot infer from a slash)
   782→  const isCrypto = pos.asset_class === 'crypto' || pos.symbol.includes('/')
   783→  // orders use the position's own symbol form (crypto keeps the slash)
   784→  const orderSymbol = pos.symbol
   785→  // long position -> sell exits; short position -> buy exits
   786→  const exitSide = pos.side === 'short' ? 'buy' : 'sell'
   787→  const qtyAvail = parseFloat(pos.qty_available) || parseFloat(pos.qty) || 0
   788→  if (!(qtyAvail > 0)) return { results: [], errors: ['position has no available qty'] }
   789→
   790→  const results: TradingOrder[] = []
   791→  const errors: string[] = []
   792→  const post = (body: Record<string, unknown>) =>
   793→    tradeFetch<TradingOrder>(userId, mode, '/v2/orders', { method: 'POST', body: JSON.stringify(body) })
   794→  const legQty = (n: number) => String(parseFloat((n * (isCrypto ? 0.99 : 1)).toFixed(8)))
   795→
   796→  if (!isCrypto) {
   797→    if (hasTp && hasSl) {
   798→      const r = await post({
   799→        symbol: orderSymbol,
   800→        qty: String(qtyAvail),
   801→        side: exitSide,
   802→        type: 'limit',
   803→        time_in_force: 'gtc',
   804→        limit_price: fmtPx(exits.tp!),
   805→        order_class: 'oco',
   806→        take_profit: { limit_price: fmtPx(exits.tp!) },
   807→        stop_loss: { stop_price: fmtPx(exits.sl!) },
   808→      })
   809→      if (r.data && (r.data as TradingOrder).id) {
   810→        return { results: [r.data], errors, oco: true }
   811→      }
   812→      errors.push(`OCO rejected, placing legs separately: ${r.error || 'failed'}`)
   813→    }
   814→    if (hasTp) {
   815→      const r = await post({
   816→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'limit',
   817→        time_in_force: 'gtc', limit_price: fmtPx(exits.tp!),
   818→      })
   819→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   820→      else errors.push(`TP: ${r.error || 'failed'}`)
   821→    }
   822→    if (hasSl) {
   823→      const r = await post({
   824→        symbol: orderSymbol, qty: String(qtyAvail), side: exitSide, type: 'stop',
   825→        time_in_force: 'gtc', stop_price: fmtPx(exits.sl!),
   826→      })
   827→      if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   828→      else errors.push(`SL: ${r.error || 'failed'}`)
   829→    }
   830→    return { results, errors }
   831→  }
   832→
   833→  // ---- crypto: emulated split legs (balance is reserved per open sell) ----
   834→  // Each attached leg takes HALF the position (the two legs together cover
   835→  // it), capped by what is still available — Alpaca reserves every open sell
   836→  // against qty_available, so a full-size first leg would make attaching the
   837→  // second one impossible (502 insufficient balance). Legs may be attached
   838→  // independently in any order.
   839→  const posQty = parseFloat(pos.qty) || qtyAvail
   840→  const qty = legQty(Math.min(qtyAvail, posQty / 2))
   841→  const coidBase = `ttp-${Math.random().toString(36).slice(2, 8)}`
   842→  if (hasTp) {
   843→    const r = await post({
   844→      symbol: orderSymbol, qty, side: exitSide, type: 'limit', time_in_force: 'gtc',
   845→      limit_price: fmtPx(exits.tp!, true),
   846→      client_order_id: `${coidBase}-tp`,
   847→    })
   848→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   849→    else errors.push(`TP: ${r.error || 'failed'}`)
   850→  }
   851→  if (hasSl) {
   852→    // stop-limit: book-clearing limit offset beyond the stop
   853→    const limP = exitSide === 'sell' ? exits.sl! * 0.999 : exits.sl! * 1.001
   854→    const r = await post({
   855→      symbol: orderSymbol, qty, side: exitSide, type: 'stop_limit', time_in_force: 'gtc',
   856→      stop_price: fmtPx(exits.sl!, true),
   857→      limit_price: fmtPx(limP, true),
   858→      client_order_id: `${coidBase}-sl`,
   859→    })
   860→    if (r.data && (r.data as TradingOrder).id) results.push(r.data)
   861→    else errors.push(`SL: ${r.error || 'failed'}`)
   862→  }
   863→  return { results, errors }
   864→}

   868→    method: 'DELETE',
