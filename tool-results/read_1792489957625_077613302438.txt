   293→   293→// Ticking local clock for the footer (updates every second).
   294→   294→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   295→   295→function LiveClock() {
   296→   296→  const [now, setNow] = useState<Date | null>(null)
   297→   297→  useEffect(() => {
   298→   298→    setNow(new Date())
   299→   299→    const timer = setInterval(() => setNow(new Date()), 1000)
   300→   300→    return () => clearInterval(timer)
   301→   301→  }, [])
   302→   302→  const tz = useMemo(() => {
   303→   303→    try {
   304→   304→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   305→   305→    } catch {
   306→   306→      return 'Local'
   307→   307→    }
   308→   308→  }, [])
   309→   309→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   310→   310→  const time = now
   311→   311→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   312→   312→    : '--:--:--'
   313→   313→  return (
   314→   314→    <span
   315→   315→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   316→   316→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   317→   317→    >
   318→   318→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   319→   319→      <span>{time}</span>
   320→   320→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   321→   321→    </span>
   322→   322→  )
   323→   323→}

   392→   392→/** One-line human summary of an alarm (any kind) for the list. */
   393→   393→function alertSummary(a: Alert): string {
   394→   394→  if (a.kind === 'rsi') {
   395→   395→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   396→   396→  }
   397→   397→  if (a.kind === 'macd') {
   398→   398→    const p = a.params ?? {}
   399→   399→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   400→   400→    const cond =
   401→   401→      a.condition === 'bull_cross' ? 'bullish crossover'
   402→   402→      : a.condition === 'bear_cross' ? 'bearish crossover'
   403→   403→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   404→   404→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   405→   405→      : 'crossover'
   406→   406→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   407→   407→  }
   408→   408→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   409→   409→}

   423→   423→  const [price, setPrice] = useState('')
   424→   424→  const [dir, setDir] = useState<'above' | 'below'>('above')
   425→   425→  // rsi fields
   426→   426→  const [rsiPeriod, setRsiPeriod] = useState('14')
   427→   427→  const [rsiLevel, setRsiLevel] = useState('70')
   428→   428→  // macd fields
   429→   429→  const [fast, setFast] = useState('12')
   430→   430→  const [slow, setSlow] = useState('26')
   431→   431→  const [signal, setSignal] = useState('9')
   432→   432→  const [cond, setCond] = useState<'bull_cross' | 'bear_cross' | 'cross_above' | 'cross_below'>('bull_cross')
   433→   433→  const [macdLevel, setMacdLevel] = useState('0')
   434→   434→  const [creating, setCreating] = useState(false)
   435→   435→  const symTouched = useRef(false)
   436→   436→
   437→   437→  // prefill symbol/timeframe from the active chart tab (once)
   438→   438→  useEffect(() => {
   439→   439→    if (symTouched.current) return
   440→   440→    if (activeTab?.symbol) {
   441→   441→      setSymbol(activeTab.symbol)
   442→   442→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   443→   443→    }
   444→   444→  }, [activeTab?.symbol, activeTab?.timeframe])
   445→   445→
   446→   446→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   447→   447→  const flipRsiLevel = (d: 'above' | 'below') => {
   448→   448→    setDir(d)
   449→   449→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   450→   450→  }
   451→   451→
   452→   452→  const create = async () => {
   453→   453→    const sym = symbol.trim().toUpperCase()
   454→   454→    if (!sym) return
   455→   455→    setCreating(true)
   456→   456→    try {
   457→   457→      const body: Record<string, unknown> =
   458→   458→        kind === 'price'
   459→   459→          ? {
   460→   460→              symbol: sym, kind, price: parseFloat(price),
   461→   461→              direction: dir, color: '#f59e0b',
   462→   462→            }
   463→   463→          : kind === 'rsi'
   464→   464→            ? {
   465→   465→                symbol: sym, kind, price: parseFloat(rsiLevel),
   466→   466→                direction: dir, timeframe: tf,
   467→   467→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   468→   468→              }
   469→   469→            : {
   470→   470→                symbol: sym, kind, condition: cond, timeframe: tf,
   471→   471→                price: parseFloat(macdLevel) || 0,
   472→   472→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   473→   473→                color: '#f59e0b',
   474→   474→              }
   475→   475→      const res = await fetch('/api/alerts', {
   476→   476→        method: 'POST',
   477→   477→        headers: { 'content-type': 'application/json' },
   478→   478→        body: JSON.stringify(body),
   479→   479→      })
   480→   480→      const json = await res.json().catch(() => null)
   481→   481→      if (res.ok && json?.alert) {
   482→   482→        addAlertLocal(json.alert as Alert)
   483→   483→      } else {
   484→   484→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   485→   485→      }
   486→   486→    } catch { /* offline */ } finally {
   487→   487→      setCreating(false)
   488→   488→    }
   489→   489→  }
   490→   490→
   491→   491→  const toggle = async (id: string, status: string) => {
   492→   492→    const next = status === 'paused' ? 'active' : 'paused'
   493→   493→    patchAlertLocal(id, { status: next })
   494→   494→    await fetch(`/api/alerts/${id}`, {
   495→   495→      method: 'PATCH',
   496→   496→      headers: { 'content-type': 'application/json' },
   497→   497→      body: JSON.stringify({ status: next }),
   498→   498→    })
   499→   499→  }
   500→   500→  const remove = async (id: string) => {
   501→   501→    removeAlertLocal(id)
   502→   502→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   503→   503→  }
   504→   504→
   505→   505→  const formValid =
   506→   506→    kind === 'price'
   507→   507→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   508→   508→      : kind === 'rsi'
   509→   509→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   510→   510→        : cond === 'cross_above' || cond === 'cross_below'
   511→   511→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   512→   512→          : symbol.trim().length > 0
   513→   513→
   514→   514→  return (
   515→   515→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   516→   516→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   517→   517→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   518→   518→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   519→   519→          <span className="flex items-center gap-2">
   520→   520→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   521→   521→          </span>
   522→   522→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   523→   523→            Close
   524→   524→          </Button>
   525→   525→        </div>
   526→   526→
   527→   527→        {/* ---- new alarm form ---- */}
   528→   528→        <div className="border-b border-zinc-800 p-2.5">
   529→   529→          <div className="mb-2 flex items-center gap-1.5">
   530→   530→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   531→   531→              <button
   532→   532→                key={k}
   533→   533→                onClick={() => setKind(k)}
   534→   534→                aria-pressed={kind === k}
   535→   535→                className={cn(
   536→   536→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   537→   537→                  kind === k
   538→   538→                    ? k === 'rsi'
   539→   539→                      ? 'bg-pink-500/15 text-pink-300'
   540→   540→                      : k === 'macd'
   541→   541→                        ? 'bg-amber-500/15 text-amber-300'
   542→   542→                        : 'bg-emerald-500/15 text-emerald-300'
   543→   543→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   544→   544→                )}
   545→   545→              >
   546→   546→                {k}
   547→   547→              </button>
   548→   548→            ))}
   549→   549→            <Input
   550→   550→              value={symbol}
   551→   551→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   552→   552→              placeholder="SYMBOL"
   553→   553→              aria-label="Alarm symbol"
   554→   554→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   555→   555→            />
   556→   556→          </div>
   557→   557→
   558→   558→          {kind === 'price' && (
   559→   559→            <div className="flex items-center gap-1.5">
   560→   560→              <select
   561→   561→                value={dir}
   562→   562→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   563→   563→                aria-label="Trigger direction"
   564→   564→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   565→   565→              >
   566→   566→                <option value="above">crosses above</option>
   567→   567→                <option value="below">falls below</option>
   568→   568→              </select>
   569→   569→              <Input
   570→   570→                type="number" step="any" value={price}
   571→   571→                onChange={(e) => setPrice(e.target.value)}
   572→   572→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   573→   573→                placeholder="Price" aria-label="Alert price"
   574→   574→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   575→   575→              />
   576→   576→            </div>
   577→   577→          )}
   578→   578→
   579→   579→          {kind === 'rsi' && (
   580→   580→            <div className="space-y-1.5">
   581→   581→              <div className="flex items-center gap-1.5">
   582→   582→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>
   583→   583→                <Input
   584→   584→                  type="number" value={rsiPeriod} min={2} max={200}
   585→   585→                  onChange={(e) => setRsiPeriod(e.target.value)}
   586→   586→                  aria-label="RSI period"
   587→   587→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   588→   588→                />
   589→   589→                <select
   590→   590→                  value={dir}
   591→   591→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   592→   592→                  aria-label="RSI trigger direction"
   593→   593→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   594→   594→                >
   595→   595→                  <option value="above">crosses above</option>
   596→   596→                  <option value="below">crosses below</option>
   597→   597→                </select>
   598→   598→                <Input
   599→   599→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   600→   600→                  onChange={(e) => setRsiLevel(e.target.value)}
   601→   601→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   602→   602→                  aria-label="RSI level"
   603→   603→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   604→   604→                />
   605→   605→              </div>
   606→   606→              <TfRow tf={tf} setTf={setTf} />
   607→   607→            </div>
   608→   608→          )}
   609→   609→
   610→   610→          {kind === 'macd' && (
   611→   611→            <div className="space-y-1.5">
   612→   612→              <div className="flex items-center gap-1.5">
   613→   613→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   614→   614→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   615→   615→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   616→   616→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   617→   617→                <select
   618→   618→                  value={cond}
   619→   619→                  onChange={(e) => setCond(e.target.value as typeof cond)}
   620→   620→                  aria-label="MACD condition"
   621→   621→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   622→   622→                >
   623→   623→                  <option value="bull_cross">bull crossover</option>
   624→   624→                  <option value="bear_cross">bear crossover</option>
   625→   625→                  <option value="cross_above">crosses above</option>
   626→   626→                  <option value="cross_below">crosses below</option>
   627→   627→                </select>
   628→   628→              </div>
   629→   629→              <div className="flex items-center gap-1.5">
   630→   630→                {(cond === 'cross_above' || cond === 'cross_below') && (
   631→   631→                  <Input
   632→   632→                    type="number" step="any" value={macdLevel}
   633→   633→                    onChange={(e) => setMacdLevel(e.target.value)}
   634→   634→                    aria-label="MACD level"
   635→   635→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   636→   636→                  />
   637→   637→                )}
   638→   638→                <TfRow tf={tf} setTf={setTf} right />
   639→   639→              </div>
   640→   640→            </div>
   641→   641→          )}
   642→   642→
   643→   643→          <Button
   644→   644→            size="sm" disabled={!formValid || creating} onClick={() => void create()}
   645→   645→            className="mt-2 h-7 w-full bg-amber-500 text-xs text-zinc-950 hover:bg-amber-400"
   646→   646→          >
   647→   647→            {creating ? <Loader2 className="h-3 w-3 animate-spin" /> : '🔔 Create alarm'}
   648→   648→          </Button>
   649→   649→        </div>
   650→   650→
   651→   651→        {/* ---- alarm list ---- */}
   652→   652→        <div className="max-h-[45vh] overflow-y-auto p-2">
   653→   653→          {alerts.length === 0 && (
   654→   654→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   655→   655→              No alarms yet — create one above or right-click the chart.
   656→   656→            </p>
   657→   657→          )}
   658→   658→          {alerts.map((a) => (
   659→   659→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   660→   660→              <div className="flex min-w-0 items-center gap-2">
   661→   661→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   662→   662→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   663→   663→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   664→   664→                <Badge
   665→   665→                  variant="outline"
   666→   666→                  className={
   667→   667→                    a.status === 'triggered'
   668→   668→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   669→   669→                      : a.status === 'paused'
   670→   670→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   671→   671→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   672→   672→                  }
   673→   673→                >
   674→   674→                  {a.status}
   675→   675→                </Badge>
   676→   676→              </div>
   677→   677→              <div className="flex shrink-0 items-center gap-1">
   678→   678→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>
   679→   679→                  {a.status === 'paused' ? 'Resume' : 'Pause'}
   680→   680→                </Button>
   681→   681→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   682→   682→                  ✕
   683→   683→                </Button>
   684→   684→              </div>
   685→   685→            </div>
   686→   686→          ))}
   687→   687→        </div>
   688→   688→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   689→   689→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   690→   690→          <span className="ml-1 inline-flex items-center align-middle">
   691→   691→            <Settings2 className="h-3 w-3" />
   692→   692→          </span>
   693→   693→        </div>
   694→   694→      </div>
   695→   695→    </div>
   696→   696→  )
   697→   697→}
