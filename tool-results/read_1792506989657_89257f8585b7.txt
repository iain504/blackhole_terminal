    56→/** Resolves the pane list for a tab (grid config or the single legacy pane). */
    57→export function tabPanes(tab: Tab): Pane[] {
    58→  if (tab.grid?.panes?.length) return tab.grid.panes
    59→  return [{ symbol: tab.symbol, timeframe: tab.timeframe }]
    60→}
    61→
    62→/** Resolves the active grid layout for a tab ('1' when unset). */
    63→export function tabLayout(tab: Tab): GridLayoutId {
    64→  return tab.grid?.layout ?? '1'
    65→}
    66→
    67→/** Pane count for a layout id. */
    68→export function layoutPaneCount(id: GridLayoutId): number {
    69→  return GRID_LAYOUTS.find((l) => l.id === id)?.panes ?? 1
    70→}

   269→let idCounter = 0

   271→  idCounter += 1
   272→  return `${prefix}-${Date.now().toString(36)}-${idCounter}`

   276→  v: 1,

   278→  activeTabId: null,

   282→export const useTerminal = create<TerminalState>((set, get) => ({
   283→  user: null,
   284→  hydrated: false,
   285→  setUser: (u) => set({ user: u }),
   286→  setHydrated: (v) => set({ hydrated: v }),
   287→
   288→  tabs: [{ id: uid('tab'), ...DEFAULT_TAB }],
   289→  activeTabId: '',
   290→
   291→  addTab: (type) => {
   292→    const t: Tab =
   293→      type === 'chart'
   294→        ? { id: uid('tab'), type, symbol: get().tabs.find((x) => x.id === get().activeTabId)?.symbol || 'AAPL', timeframe: get().tabs.find((x) => x.id === get().activeTabId)?.timeframe || '1D' }
   295→        : { id: uid('tab'), type, symbol: 'AAPL', timeframe: '1D' }
   296→    set((s) => ({ tabs: [...s.tabs, t], activeTabId: t.id, layoutDirty: true }))
   297→  },
   298→
   299→  closeTab: (id) => {
   300→    const { tabs, activeTabId } = get()
   301→    if (tabs.length <= 1) return
   302→    const idx = tabs.findIndex((t) => t.id === id)
   303→    const remaining = tabs.filter((t) => t.id !== id)
   304→    const nextActive = activeTabId === id ? remaining[Math.max(0, idx - 1)].id : activeTabId
   305→    set({ tabs: remaining, activeTabId: nextActive, layoutDirty: true })
   306→  },
   307→
   308→  setActiveTab: (id) => set({ activeTabId: id, layoutDirty: true }),
   309→
   310→  moveTab: (dragId, toIndex) =>
   311→    set((s) => {
   312→      const from = s.tabs.findIndex((t) => t.id === dragId)
   313→      if (from < 0) return {}
   314→      // remove first, then clamp the insertion point into the shortened array
   315→      const next = s.tabs.filter((t) => t.id !== dragId)
   316→      const idx = Math.max(0, Math.min(toIndex, next.length))
   317→      next.splice(idx, 0, s.tabs[from])
   318→      if (next.every((t, i) => t.id === s.tabs[i].id)) return {} // no-op order
   319→      return { tabs: next, layoutDirty: true }
   320→    }),
   321→
   322→  setTabSymbol: (tabId, symbol) =>
   323→    set((s) => ({
   324→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, symbol } : t)),
   325→      layoutDirty: true,
   326→    })),
   327→
   328→  setTabTimeframe: (tabId, tf) =>
   329→    set((s) => ({
   330→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, timeframe: tf } : t)),
   331→      layoutDirty: true,
   332→    })),
   333→
   334→  // ---------- multi-chart grid ----------
   335→  setGridLayout: (tabId, layout) =>
   336→    set((s) => ({
   337→      tabs: s.tabs.map((t) => {
   338→        if (t.id !== tabId || t.type !== 'chart') return t
   339→        const panes = tabPanes(t)
   340→        const count = layoutPaneCount(layout)
   341→        const next = [...panes]
   342→        while (next.length < count) next.push({ ...panes[0] })
   343→        next.length = count
   344→        if (layout === '1') {
   345→          // single pane mirrors the legacy tab fields so other panels keep working
   346→          return { ...t, symbol: next[0].symbol, timeframe: next[0].timeframe, grid: { layout, panes: next } }
   347→        }
   348→        // new arrangement → start from the default proportions (stale size
   349→        // keys from the previous arrangement would fight the new groups)
   350→        return { ...t, grid: { layout, panes: next } }
   351→      }),
   352→      layoutDirty: true,
   353→    })),
   354→
   355→  setPaneSymbol: (tabId, paneIdx, symbol) =>
   356→    set((s) => ({
   357→      tabs: s.tabs.map((t) => {
   358→        if (t.id !== tabId) return t
   359→        const panes = tabPanes(t)
   360→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   361→        if (panes[i].symbol === symbol) return t
   362→        const next = panes.map((p, j) => (j === i ? { ...p, symbol } : p))
   363→        // legacy single-chart mode keeps tab fields in sync
   364→        if (!t.grid || t.grid.layout === '1') return { ...t, symbol, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   365→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   366→      }),
   367→      layoutDirty: true,
   368→    })),
   369→
   370→  setPaneTimeframe: (tabId, paneIdx, tf) =>
   371→    set((s) => ({
   372→      tabs: s.tabs.map((t) => {
   373→        if (t.id !== tabId) return t
   374→        const panes = tabPanes(t)
   375→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   376→        if (panes[i].timeframe === tf) return t
   377→        const next = panes.map((p, j) => (j === i ? { ...p, timeframe: tf } : p))
   378→        if (!t.grid || t.grid.layout === '1') return { ...t, timeframe: tf, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   379→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   380→      }),
   381→      layoutDirty: true,
   382→    })),
   383→
   384→  setPaneChartType: (tabId, paneIdx, chartType) =>
   385→    set((s) => ({
   386→      tabs: s.tabs.map((t) => {
   387→        if (t.id !== tabId) return t
   388→        const panes = tabPanes(t)
   389→        const i = Math.max(0, Math.min(paneIdx, panes.length - 1))
   390→        if (panes[i].chartType === chartType) return t
   391→        const next = panes.map((p, j) => (j === i ? { ...p, chartType } : p))
   392→        if (!t.grid || t.grid.layout === '1') return { ...t, grid: { layout: t.grid?.layout ?? '1', panes: next } }
   393→        return { ...t, grid: { layout: t.grid.layout, panes: next } }
   394→      }),
   395→      layoutDirty: true,
   396→    })),
   397→
   398→  setGridSizes: (tabId, key, sizes) =>
   399→    set((s) => ({
   400→      tabs: s.tabs.map((t) => {
   401→        if (t.id !== tabId || t.type !== 'chart') return t
   402→        const clean = sizes.slice(0, 8).map((n) => (Number.isFinite(n) ? Math.min(97, Math.max(3, Math.round(n * 10) / 10)) : 50))
   403→        if (clean.length < 2) return t
   404→        const prev = t.grid?.sizes?.[key]
   405→        if (prev && prev.length === clean.length && prev.every((v, i) => Math.abs(v - clean[i]) < 0.05)) return t
   406→        return {
   407→          ...t,
   408→          grid: { layout: t.grid?.layout ?? '1', panes: tabPanes(t), sizes: { ...(t.grid?.sizes ?? {}), [key]: clean } },
   409→        }
   410→      }),
   411→      layoutDirty: true,
   412→    })),
   413→
   414→  setFocusedPane: (tabId, paneIdx) =>
   415→    set((s) => ({
   416→      tabs: s.tabs.map((t) => (t.id === tabId && t.focusedPane !== paneIdx ? { ...t, focusedPane: paneIdx } : t)),
   417→      layoutDirty: true,
   418→    })),
   419→
   420→  setSyncCrosshair: (tabId, v) =>
   421→    set((s) => ({
   422→      tabs: s.tabs.map((t) => (t.id === tabId ? { ...t, syncCrosshair: v } : t)),
   423→      layoutDirty: true,
   424→    })),
   425→
   426→  selectSymbol: (symbol) => {
   427→    const { activeTabId, tabs } = get()
   428→    const active = tabs.find((t) => t.id === activeTabId)
   429→    if (active && active.type === 'chart') {
   430→      // target the focused pane so multi-chart grids follow watchlist clicks
   431→      const panes = tabPanes(active)
   432→      const idx = Math.min(active.focusedPane ?? 0, panes.length - 1)
   433→      get().setPaneSymbol(active.id, idx, symbol)
   434→    } else {
   435→      const chartTab = tabs.find((t) => t.type === 'chart')
   436→      if (chartTab) {
   437→        // Activate the first chart tab and route through setPaneSymbol so the
   438→        // tab's PANES are updated too — setting only `symbol` left
   439→        // grid.panes[0] on the old symbol (tab label said XOM, chart showed
   440→        // AAPL after clicking a heatmap tile from the heatmap tab).
   441→        get().setPaneSymbol(chartTab.id, chartTab.focusedPane ?? 0, symbol)
   442→        set((s) => ({ activeTabId: chartTab.id, layoutDirty: true }))
   443→      }
   444→    }
   445→  },
   446→
   447→  panelOrder: DEFAULT_LAYOUT.sidebar.panelOrder,
   448→  panelCollapsed: {},
   449→  mainSizes: DEFAULT_LAYOUT.sidebar.mainSizes,
   450→
   451→  setPanelOrder: (order) => set({ panelOrder: order, layoutDirty: true }),
   452→
   453→  togglePanelCollapsed: (id) =>
   454→    set((s) => ({
   455→      panelCollapsed: { ...s.panelCollapsed, [id]: !s.panelCollapsed[id] },
   456→      layoutDirty: true,
   457→    })),
   458→
   459→  setMainSizes: (sizes) => {
   460→    const prev = get().mainSizes
   461→    if (Math.abs(prev[0] - sizes[0]) < 0.1 && Math.abs(prev[1] - sizes[1]) < 0.1) return
   462→    set({ mainSizes: sizes, layoutDirty: true })
   463→  },
   464→
   465→  layoutDirty: false,
   466→  markLayoutDirty: () => set({ layoutDirty: true }),
   467→  markLayoutClean: () => set({ layoutDirty: false }),
   468→
   469→  layoutSnapshot: () => {
   470→    const s = get()
   471→    return {
   472→      v: 1 as const,
   473→      tabs: s.tabs,
   474→      activeTabId: s.activeTabId,
   475→      sidebar: {
   476→        panelOrder: s.panelOrder,
   477→        panelCollapsed: s.panelCollapsed,
   478→        mainSizes: s.mainSizes,
   479→      },
   480→      // indicator settings travel WITH the layout so the autosave (and the
   481→      // beforeunload beacon) carry them server-side, not just the export file
   482→      indicators: loadIndicators(),
   483→    }
   484→  },
   485→
   486→  hydrateFromServer: (data) => {
   487→    if (!data || typeof data !== 'object') return
   488→    const blob = data as Partial<LayoutBlob>
   489→    if (!blob.tabs || !Array.isArray(blob.tabs) || blob.tabs.length === 0) return

   491→    const VALID_LAYOUTS: GridLayoutId[] = ['1', '2h', '2v', '3s', '4', '6', '8']

   497→        if (!VALID_SIZE_KEYS.includes(k) || !Array.isArray(arr)) continue
   498→        const clean = arr
   499→          .slice(0, 8)
   500→          .map((n) => (typeof n === 'number' && Number.isFinite(n) ? Math.min(97, Math.max(3, n)) : NaN))
   501→        if (clean.length >= 2 && clean.every((n) => Number.isFinite(n))) out[k] = clean.map((n) => Math.round(n * 10) / 10)
   502→      }
   503→      return Object.keys(out).length ? out : undefined
   504→    }
   505→    const tabs = blob.tabs
   506→      .filter((t) => t && typeof t.id === 'string' && ['chart', 'calendar', 'heatmap', 'positions', 'strategy'].includes(t.type))

   509→          id: t.id,
   510→          type: t.type as TabType,
   511→          symbol: (t.symbol || 'AAPL').toUpperCase(),
   512→          timeframe: (VALID_TFS.includes(t.timeframe as string) ? t.timeframe : '1D') as Timeframe,
   513→        }

   515→        // restore multi-chart grid (sanitized; legacy blobs get a single pane)
   516→        const layout: GridLayoutId =
   517→          t.grid && VALID_LAYOUTS.includes(t.grid.layout) ? t.grid.layout : '1'
   518→        let panes: Pane[] = Array.isArray(t.grid?.panes)
   519→          ? t.grid.panes
   520→              .filter((p) => p && typeof p.symbol === 'string')
   521→              .map((p) => ({
   522→                symbol: p.symbol.toUpperCase(),
   523→                timeframe: (VALID_TFS.includes(p.timeframe as string) ? p.timeframe : '1D') as Timeframe,
   524→                chartType: p.chartType === 'line' ? 'line' : 'candles',
   525→              }))
   526→          : []
   527→        if (panes.length === 0) panes = [{ symbol: base.symbol, timeframe: base.timeframe }]
   528→        const count = layoutPaneCount(layout)
   529→        while (panes.length < count) panes.push({ ...panes[0] })
   530→        panes.length = count
   531→        const gridSizes = sizes(t.grid?.sizes)
   532→        return {
   533→          ...base,
   534→          grid: { layout, panes, ...(gridSizes ? { sizes: gridSizes } : {}) },
   535→          focusedPane: Math.max(0, Math.min(typeof t.focusedPane === 'number' ? t.focusedPane : 0, count - 1)),
   536→          syncCrosshair: Boolean(t.syncCrosshair),
   537→        }
   538→      })

   540→    // indicator settings persisted server-side (autosave) — apply the stored
   541→    // copy so they follow the user across browsers/machines like the rest
   542→    if (blob.indicators && typeof blob.indicators === 'object') {
   543→      try { saveIndicators(normalizeIndicators(blob.indicators)) } catch { /* private mode */ }
   544→    }
   545→    const activeTabId = tabs.some((t) => t.id === blob.activeTabId) ? (blob.activeTabId as string) : tabs[0].id
   546→    const sb = blob.sidebar
   547→    const mainSizes: [number, number] =
   548→      Array.isArray(sb?.mainSizes) && sb.mainSizes.length === 2 ? [sb.mainSizes[0], 100 - sb.mainSizes[0]] : [72, 28]
   549→    set({
   550→      tabs,
   551→      activeTabId,
   552→      panelOrder: sb?.panelOrder?.length === 3 ? sb.panelOrder : DEFAULT_LAYOUT.sidebar.panelOrder,
   553→      panelCollapsed: sb?.panelCollapsed || {},
   554→      mainSizes,
   555→      layoutDirty: false,
   556→    })

   678→  // drag & drop reorder — the items ARRAY order IS the persisted sort (the
   679→  // PATCH endpoint stores it verbatim; export/import carries the same array)
   680→  // `from`/`to` are ORIGINAL-space indices; `to` is an insertion boundary
   681→  // (drop on a row's top half → its index, bottom half → index+1)
   682→  reorderWatchlistSymbol: async (watchlistId, from, to) => {
   683→    const list = get().watchlists.find((l) => l.id === watchlistId)
   684→    if (!list) return
   685→    if (from < 0 || from >= list.items.length) return
   686→    const to2 = Math.max(0, Math.min(list.items.length, to))
   687→    if (to2 === from || to2 === from + 1) return // same place — no-op
   688→    const isSep = (s: string) => s.startsWith('#')
   689→    const items = [...list.items]
   690→    if (isSep(items[from])) {
   691→      // dragging a SEPARATOR moves the whole group block (marker + its symbols)
   692→      let end = from + 1
   693→      while (end < items.length && !isSep(items[end])) end++
   694→      const block = items.splice(from, end - from)
   695→      // map the boundary into the array AFTER the block was extracted, then
   696→      // apply the same moving-down offset as single-row moves
   697→      const b = to2 > from ? Math.max(from, to2 - block.length) : to2
   698→      const ins = Math.max(0, Math.min(items.length, from < to2 ? b - 1 : b))
   699→      items.splice(ins, 0, ...block)
   700→    } else {
   701→      const [sym] = items.splice(from, 1)
   702→      // moving DOWN: removing the item first shifts the target left by one
   703→      items.splice(from < to2 ? to2 - 1 : to2, 0, sym)
   704→    }
   705→    set((s) => ({ watchlists: s.watchlists.map((l) => (l.id === watchlistId ? { ...l, items } : l)) }))
   706→    try {
   707→      const res = await fetch(`/api/watchlists/${watchlistId}`, {
   708→        method: 'PATCH',
   709→        headers: { 'content-type': 'application/json' },
   710→        body: JSON.stringify({ items }),
   711→      })
   712→      if (!res.ok) await get().loadWatchlists() // resync to source of truth
   713→    } catch {
   714→      /* offline — optimistic order stays; next action resyncs */
   715→    }
   716→  },
