   302→// Ticking local clock for the footer (updates every second).
   303→// Renders a placeholder until mounted so SSR/CSR markup always matches.
   304→function LiveClock() {
   305→  const [now, setNow] = useState<Date | null>(null)
   306→  useEffect(() => {
   307→    setNow(new Date())
   308→    const timer = setInterval(() => setNow(new Date()), 1000)
   309→    return () => clearInterval(timer)
   310→  }, [])
   311→  const tz = useMemo(() => {
   312→    try {
   313→      return Intl.DateTimeFormat().resolvedOptions().timeZone || 'Local'
   314→    } catch {
   315→      return 'Local'
   316→    }
   317→  }, [])
   318→  const city = tz.includes('/') ? tz.split('/').pop()!.replace(/_/g, ' ') : tz
   319→  const time = now
   320→    ? now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })
   321→    : '--:--:--'
   322→  return (
   323→    <span
   324→      className="flex items-center gap-1.5 tabular-nums text-zinc-300"
   325→      title={now ? `${now.toLocaleDateString('en-GB', { weekday: 'short', day: '2-digit', month: 'short' })} ${time} · ${tz}` : tz}
   326→    >
   327→      <Clock className="h-3 w-3 text-emerald-500" aria-hidden="true" />
   328→      <span>{time}</span>
   329→      <span className="hidden text-[9px] uppercase tracking-wider text-zinc-600 sm:inline">{city}</span>
   330→    </span>
   331→  )
   332→}

   358→  const [meta, setMeta] = useState<{ demo: boolean; reason?: string }>({ demo: false })
   359→  useEffect(() => {
   360→    let alive = true

   369→        const json = await res.json()

   402→/** One-line human summary of an alarm (any kind) for the list. */
   403→function alertSummary(a: Alert): string {
   404→  if (a.kind === 'rsi') {
   405→    return `RSI(${a.params?.period ?? 14}) crosses ${a.direction === 'above' ? 'above' : 'below'} ${a.price} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   406→  }
   407→  if (a.kind === 'macd') {
   408→    const p = a.params ?? {}
   409→    const base = `MACD(${p.fast ?? 12},${p.slow ?? 26},${p.signal ?? 9})`
   410→    const cond =
   411→      a.condition === 'bull_cross' ? 'bullish crossover'
   412→      : a.condition === 'bear_cross' ? 'bearish crossover'
   413→      : a.condition === 'cross_above' ? `crosses above ${a.price}`
   414→      : a.condition === 'cross_below' ? `crosses below ${a.price}`
   415→      : 'crossover'
   416→    return `${base} ${cond} · ${TF_LABEL[a.timeframe] ?? a.timeframe}`
   417→  }
   418→  if (a.kind === 'trend') {
   419→    // the level MOVES with the bound trend line — show the crossing direction
   420→    // plus the level the line had when the alarm was created
   421→    return `Trend line · crosses ${a.direction === 'above' ? 'up' : 'down'} · level was ${fmtPrice(a.price)}`
   422→  }
   423→  return `${a.direction === 'above' ? '≥' : '≤'} ${fmtPrice(a.price)}`
   424→}

   431→  const activeTab = useTerminal((s) => s.tabs.find((t) => t.id === s.activeTabId) ?? s.tabs[0])
   432→
   433→  // ---- creation form state ----
   434→  const [kind, setKind] = useState<'price' | 'rsi' | 'macd'>('rsi')
   435→  const [symbol, setSymbol] = useState('')
   436→  const [tf, setTf] = useState('1D')

   438→  const [price, setPrice] = useState('')

   440→  // rsi fields
   441→  const [rsiPeriod, setRsiPeriod] = useState('14')
   442→  const [rsiLevel, setRsiLevel] = useState('70')

   446→  const [signal, setSignal] = useState('9')
   447→  const [cond, setCond] = useState<'bull_cross' | 'bear_cross' | 'cross_above' | 'cross_below'>('bull_cross')
   448→  const [macdLevel, setMacdLevel] = useState('0')
   449→  const [creating, setCreating] = useState(false)
   450→  const symTouched = useRef(false)
   451→
   452→  // prefill symbol/timeframe from the active chart tab (once)
   453→  useEffect(() => {
   454→    if (symTouched.current) return
   455→    if (activeTab?.symbol) {
   456→      setSymbol(activeTab.symbol)
   457→      if (activeTab.timeframe) setTf(activeTab.timeframe)
   458→    }
   459→  }, [activeTab?.symbol, activeTab?.timeframe])
   460→
   461→  // RSI level default flips with the direction (70 overbought / 30 oversold)
   462→  const flipRsiLevel = (d: 'above' | 'below') => {
   463→    setDir(d)
   464→    if (!symTouched.current || rsiLevel === '70' || rsiLevel === '30') setRsiLevel(d === 'above' ? '70' : '30')
   465→  }
   466→
   467→  const create = async () => {
   468→    const sym = symbol.trim().toUpperCase()
   469→    if (!sym) return
   470→    setCreating(true)
   471→    try {
   472→      const body: Record<string, unknown> =
   473→        kind === 'price'
   474→          ? {
   475→              symbol: sym, kind, price: parseFloat(price),
   476→              direction: dir, color: '#f59e0b',
   477→            }
   478→          : kind === 'rsi'
   479→            ? {
   480→                symbol: sym, kind, price: parseFloat(rsiLevel),
   481→                direction: dir, timeframe: tf,
   482→                params: { period: parseInt(rsiPeriod, 10) }, color: '#f472b6',
   483→              }
   484→            : {
   485→                symbol: sym, kind, condition: cond, timeframe: tf,
   486→                price: parseFloat(macdLevel) || 0,
   487→                params: { fast: parseInt(fast, 10), slow: parseInt(slow, 10), signal: parseInt(signal, 10) },
   488→                color: '#f59e0b',
   489→              }
   490→      const res = await fetch('/api/alerts', {
   491→        method: 'POST',
   492→        headers: { 'content-type': 'application/json' },
   493→        body: JSON.stringify(body),
   494→      })
   495→      const json = await res.json().catch(() => null)
   496→      if (res.ok && json?.alert) {
   497→        addAlertLocal(json.alert as Alert)
   498→      } else {
   499→        ttLog('alerts', `Create alarm failed: ${json?.error || res.status}`, 'warn')
   500→      }
   501→    } catch { /* offline */ } finally {
   502→      setCreating(false)
   503→    }
   504→  }
   505→
   506→  const toggle = async (id: string, status: string) => {
   507→    const next = status === 'paused' ? 'active' : 'paused'
   508→    patchAlertLocal(id, { status: next })
   509→    await fetch(`/api/alerts/${id}`, {
   510→      method: 'PATCH',
   511→      headers: { 'content-type': 'application/json' },
   512→      body: JSON.stringify({ status: next }),
   513→    })
   514→  }
   515→  const remove = async (id: string) => {
   516→    removeAlertLocal(id)
   517→    await fetch(`/api/alerts/${id}`, { method: 'DELETE' })
   518→  }
   519→
   520→  const formValid =
   521→    kind === 'price'
   522→      ? Number.isFinite(parseFloat(price)) && parseFloat(price) > 0 && symbol.trim().length > 0
   523→      : kind === 'rsi'
   524→        ? Number.isFinite(parseFloat(rsiLevel)) && parseFloat(rsiLevel) > 0 && symbol.trim().length > 0
   525→        : cond === 'cross_above' || cond === 'cross_below'
   526→          ? Number.isFinite(parseFloat(macdLevel)) && symbol.trim().length > 0
   527→          : symbol.trim().length > 0
   528→
   529→  return (
   530→    <div className="fixed inset-0 z-50" role="dialog" aria-label="Alerts">
   531→      <button className="absolute inset-0 bg-black/50" onClick={onClose} aria-label="Close alerts" />
   532→      <div className="absolute right-3 top-12 w-[340px] max-w-[calc(100vw-24px)] rounded-lg border border-zinc-800 bg-zinc-900 shadow-xl">
   533→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-2.5 text-sm font-medium">
   534→          <span className="flex items-center gap-2">
   535→            <BellRing className="h-4 w-4 text-amber-400" /> Alerts
   536→          </span>
   537→          <Button variant="ghost" size="sm" onClick={onClose} className="h-7 px-2 text-xs">
   538→            Close
   539→          </Button>
   540→        </div>
   541→
   542→        {/* ---- new alarm form ---- */}
   543→        <div className="border-b border-zinc-800 p-2.5">
   544→          <div className="mb-2 flex items-center gap-1.5">
   545→            {(['price', 'rsi', 'macd'] as const).map((k) => (
   546→              <button
   547→                key={k}
   548→                onClick={() => setKind(k)}
   549→                aria-pressed={kind === k}
   550→                className={cn(
   551→                  'rounded px-2 py-1 text-[11px] font-semibold uppercase tracking-wide transition-colors',
   552→                  kind === k
   553→                    ? k === 'rsi'
   554→                      ? 'bg-pink-500/15 text-pink-300'
   555→                      : k === 'macd'
   556→                        ? 'bg-amber-500/15 text-amber-300'
   557→                        : 'bg-emerald-500/15 text-emerald-300'
   558→                    : 'text-zinc-500 hover:bg-zinc-800 hover:text-zinc-300',
   559→                )}
   560→              >
   561→                {k}
   562→              </button>
   563→            ))}
   564→            <Input
   565→              value={symbol}
   566→              onChange={(e) => { symTouched.current = true; setSymbol(e.target.value.toUpperCase()) }}
   567→              placeholder="SYMBOL"
   568→              aria-label="Alarm symbol"
   569→              className="ml-auto h-7 w-[110px] border-zinc-700 bg-zinc-950 text-xs uppercase"
   570→            />
   571→          </div>
   572→
   573→          {kind === 'price' && (
   574→            <div className="flex items-center gap-1.5">
   575→              <select
   576→                value={dir}
   577→                onChange={(e) => setDir(e.target.value as 'above' | 'below')}
   578→                aria-label="Trigger direction"
   579→                className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   580→              >
   581→                <option value="above">crosses above</option>
   582→                <option value="below">falls below</option>
   583→              </select>
   584→              <Input
   585→                type="number" step="any" value={price}
   586→                onChange={(e) => setPrice(e.target.value)}
   587→                onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   588→                placeholder="Price" aria-label="Alert price"
   589→                className="h-7 flex-1 border-zinc-700 bg-zinc-950 text-xs"
   590→              />
   591→            </div>
   592→          )}
   593→
   594→          {kind === 'rsi' && (
   595→            <div className="space-y-1.5">
   596→              <div className="flex items-center gap-1.5">
   597→                <label className="text-[10px] uppercase tracking-wider text-zinc-500">Len</label>
   598→                <Input
   599→                  type="number" value={rsiPeriod} min={2} max={200}
   600→                  onChange={(e) => setRsiPeriod(e.target.value)}
   601→                  aria-label="RSI period"
   602→                  className="h-7 w-14 border-zinc-700 bg-zinc-950 text-xs"
   603→                />
   604→                <select
   605→                  value={dir}
   606→                  onChange={(e) => flipRsiLevel(e.target.value as 'above' | 'below')}
   607→                  aria-label="RSI trigger direction"
   608→                  className="h-7 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   609→                >
   610→                  <option value="above">crosses above</option>
   611→                  <option value="below">crosses below</option>
   612→                </select>
   613→                <Input
   614→                  type="number" step="any" value={rsiLevel} min={1} max={99}
   615→                  onChange={(e) => setRsiLevel(e.target.value)}
   616→                  onKeyDown={(e) => e.key === 'Enter' && formValid && void create()}
   617→                  aria-label="RSI level"
   618→                  className="h-7 w-16 border-zinc-700 bg-zinc-950 text-xs"
   619→                />
   620→              </div>
   621→              <TfRow tf={tf} setTf={setTf} />
   622→            </div>
   623→          )}
   624→
   625→          {kind === 'macd' && (
   626→            <div className="space-y-1.5">
   627→              <div className="flex items-center gap-1.5">
   628→                <span className="text-[10px] uppercase tracking-wider text-zinc-500">F/S/G</span>
   629→                <Input type="number" value={fast} min={1} onChange={(e) => setFast(e.target.value)} aria-label="MACD fast" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   630→                <Input type="number" value={slow} min={2} onChange={(e) => setSlow(e.target.value)} aria-label="MACD slow" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   631→                <Input type="number" value={signal} min={1} onChange={(e) => setSignal(e.target.value)} aria-label="MACD signal" className="h-7 w-12 border-zinc-700 bg-zinc-950 text-xs" />
   632→                <select
   633→                  value={cond}
   634→                  onChange={(e) => setCond(e.target.value as typeof cond)}
   635→                  aria-label="MACD condition"
   636→                  className="h-7 min-w-0 flex-1 rounded border border-zinc-700 bg-zinc-950 px-1 text-xs text-zinc-200"
   637→                >
   638→                  <option value="bull_cross">bull crossover</option>
   639→                  <option value="bear_cross">bear crossover</option>
   640→                  <option value="cross_above">crosses above</option>
   641→                  <option value="cross_below">crosses below</option>
   642→                </select>
   643→              </div>
   644→              <div className="flex items-center gap-1.5">
   645→                {(cond === 'cross_above' || cond === 'cross_below') && (
   646→                  <Input
   647→                    type="number" step="any" value={macdLevel}
   648→                    onChange={(e) => setMacdLevel(e.target.value)}
   649→                    aria-label="MACD level"
   650→                    className="h-7 w-20 border-zinc-700 bg-zinc-950 text-xs"
   651→                  />
   652→                )}
   653→                <TfRow tf={tf} setTf={setTf} right />
   654→              </div>
   655→            </div>
   656→          )}
   657→
   658→          <Button
   659→            size="sm" disabled={!formValid || creating} onClick={() => void create()}
   660→            className="mt-2 h-7 w-full bg-amber-500 text-xs text-zinc-950 hover:bg-amber-400"
   661→          >
   662→            {creating ? <Loader2 className="h-3 w-3 animate-spin" /> : '🔔 Create alarm'}
   663→          </Button>
   664→        </div>
   665→
   666→        {/* ---- alarm list ---- */}
   667→        <div className="max-h-[45vh] overflow-y-auto p-2">
   668→          {alerts.length === 0 && (
   669→            <p className="px-2 py-6 text-center text-xs text-zinc-500">
   670→              No alarms yet — create one above or right-click the chart.
   671→            </p>
   672→          )}
   673→          {alerts.map((a) => (
   674→            <div key={a.id} className="flex items-center justify-between rounded-md px-2 py-1.5 text-xs hover:bg-zinc-800/60">
   675→              <div className="flex min-w-0 items-center gap-2">
   676→                <span className="inline-block h-2 w-2 shrink-0 rounded-full" style={{ backgroundColor: a.color }} />
   677→                <span className="font-medium text-zinc-200">{a.symbol}</span>
   678→                <span className="truncate text-zinc-400">{alertSummary(a)}</span>
   679→                <Badge
   680→                  variant="outline"
   681→                  className={
   682→                    a.status === 'triggered'
   683→                      ? 'border-emerald-500/40 text-[10px] text-emerald-400'
   684→                      : a.status === 'paused'
   685→                        ? 'border-zinc-600 text-[10px] text-zinc-400'
   686→                        : 'border-amber-500/40 text-[10px] text-amber-400'
   687→                  }
   688→                >
   689→                  {a.status}
   690→                </Badge>
   691→              </div>
   692→              <div className="flex shrink-0 items-center gap-1">
   693→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px]" onClick={() => toggle(a.id, a.status)}>
   694→                  {a.status === 'paused' ? 'Resume' : 'Pause'}
   695→                </Button>
   696→                <Button variant="ghost" size="sm" className="h-6 px-1.5 text-[11px] text-rose-400 hover:text-rose-300" onClick={() => remove(a.id)}>
   697→                  ✕
   698→                </Button>
   699→              </div>
   700→            </div>
   701→          ))}
   702→        </div>
   703→        <div className="border-t border-zinc-800 px-4 py-2 text-[11px] text-zinc-500">
   704→          Checked every 30s · indicator alarms fire on a real crossing of the chosen timeframe · Telegram via Settings
   705→          <span className="ml-1 inline-flex items-center align-middle">
   706→            <Settings2 className="h-3 w-3" />
   707→          </span>
   708→        </div>
   709→      </div>
   710→    </div>
   711→  )
   712→}
